   269→function ModulesTab() {
   270→  const { tenant, setTenant, runtime, user } = usePlatform();
   271→  const allModules = moduleRegistry.getAll().filter((m) => {
   272→    // Only show modules that support the current application
   273→    if (!m.manifest.supportedApplications || m.manifest.supportedApplications.length === 0) return true;
   274→    return m.manifest.supportedApplications.includes(user.application);
   275→  });
   276→  const isPlatform = tenant.id === "platform";
   277→
   278→  const toggle = (moduleId: string) => {
   279→    const enabled = new Set(tenant.enabledModules);
   280→    if (enabled.has(moduleId)) enabled.delete(moduleId);
   281→    else enabled.add(moduleId);
   282→    setTenant({ ...tenant, enabledModules: Array.from(enabled) });
   283→    const mod = allModules.find((m) => m.manifest.id === moduleId);
   284→    toast({
   285→      title: enabled.has(moduleId) ? "Module enabled" : "Module disabled",
   286→      description: `${mod?.manifest.name} is now ${enabled.has(moduleId) ? "visible" : "hidden"} in navigation.`,
   287→    });
   288→  };
   289→
   290→  return (
   291→    <Card>
   292→      <CardHeader>
   293→        <div className="flex items-center justify-between">
   294→          <span className="text-sm font-medium">Module Catalog</span>
   295→          <Badge variant="secondary">{tenant.enabledModules.length} enabled</Badge>
   296→        </div>
   297→      </CardHeader>
   298→      <CardContent className="space-y-2">
   299→        <p className="text-xs text-muted-foreground">Toggle modules to see the sidebar and dashboard re-compose live. This demonstrates the plug-and-play architecture (spec section 67).</p>
   300→        <Separator />
   301→        {allModules.map((m) => {
   302→          const enabled = isPlatform || tenant.enabledModules.includes(m.manifest.id);
   303→          const Icon = m.manifest.icon;
   304→          return (
   305→            <div key={m.manifest.id} className="flex items-center gap-3 rounded-md border p-3">
   306→              <div className="rounded-md p-2" style={{ background: `${m.manifest.accentColor}1a` }}>
   307→                {Icon ? <Icon className="h-4 w-4" style={{ color: m.manifest.accentColor }} /> : null}
   308→              </div>
   309→              <div className="min-w-0 flex-1">
   310→                <div className="flex items-center gap-2">
   311→                  <p className="text-sm font-medium">{m.manifest.name}</p>
   312→                  {m.manifest.optional ? <Badge variant="outline" className="text-[9px]">optional</Badge> : null}
   313→                  <Badge variant="outline" className="text-[9px]">{m.manifest.category}</Badge>
   314→                </div>
   315→                <p className="text-xs text-muted-foreground">{m.manifest.description}</p>
   316→                {m.manifest.dependencies?.length ? (
   317→                  <p className="text-[10px] text-muted-foreground/70">Depends on: {m.manifest.dependencies.join(", ")}</p>
   318→                ) : null}
   319→              </div>
   320→              <Switch checked={enabled} onCheckedChange={() => toggle(m.manifest.id)} disabled={isPlatform} />
   321→            </div>
   322→          );
   323→        })}
   324→      </CardContent>
   325→    </Card>
   326→  );
   327→}
