   305→function couponStatusTone(status: CouponStatus): "success" | "muted" | "danger" | "info" {
   306→  switch (status) {
   307→    case "active":
   308→      return "success";
   309→    case "scheduled":
   310→      return "info";
   311→    case "disabled":
   312→      return "danger";
   313→    case "expired":
   314→    default:
   315→      return "muted";
   316→  }
   317→}
   318→
   319→function discountTypeIcon(type: DiscountType) {
   320→  switch (type) {
   321→    case "Percentage":
   322→      return Percent;
   323→    case "Flat":
   324→      return DollarSign;
   325→    case "Free Trial":
   326→      return Gift;
   327→    case "Bonus Credit":
   328→      return Coins;
   329→  }
   330→}
   331→
   332→function formatDiscountValue(type: DiscountType, value: number, currency = "USD"): string {
   333→  switch (type) {
   334→    case "Percentage":
   335→      return `${value}%`;
   336→    case "Flat":
   337→      return formatCurrency(value, currency);
   338→    case "Free Trial":
   339→      return `${value} days`;
   340→    case "Bonus Credit":
   341→      return formatCurrency(value, currency);
   342→  }
   343→}
   344→
   345→function formatDate(iso: string): string {
   346→  const d = new Date(iso);
   347→  if (Number.isNaN(d.getTime())) return iso;
   348→  return d.toLocaleDateString("en-US", { year: "numeric", month: "short", day: "numeric" });
   349→}
   350→
   351→/* Deterministic 8-char code generator — LCG seeded by an integer
   352→ * counter. No Math.random — same seed always produces same code. */
   353→const CODE_ALPHABET = "ABCDEFGHJKMNPQRSTUVWXYZ23456789"; // no I O 0 1
   354→function generateCode(seed: number): string {
   355→  let v = seed * 31 + 7;
   356→  let out = "";
   357→  for (let i = 0; i < 8; i++) {
   358→    v = (v * 1103515245 + 12345) & 0x7fffffff;
   359→    out += CODE_ALPHABET[v % CODE_ALPHABET.length];
   360→  }
   361→  return out;
   362→}
   363→
   364→function copyToClipboard(text: string, label = "Code") {
   365→  if (typeof navigator !== "undefined" && navigator.clipboard?.writeText) {
   366→    navigator.clipboard.writeText(text).then(
   367→      () => toast({ title: `${label} copied`, description: text }),
   368→      () => toast({ title: "Copy failed", description: "Clipboard unavailable", variant: "destructive" }),
   369→    );
   370→  } else {
   371→    toast({ title: "Copy failed", description: "Clipboard unavailable", variant: "destructive" });
   372→  }
   373→}

   929→              <Input
   930→                id="cpn-code"
