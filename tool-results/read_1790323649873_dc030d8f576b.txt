   116→export function SettingsPage() {
   117→  const { router, navigate, tenant, runtime } = usePlatform();
   118→  const [query, setQuery] = useState("");
   119→  const [showQuickEdit, setShowQuickEdit] = useState(false);
   120→
   121→  // Quick edit's active tab — initialized from the URL `?tab=` param so
   122→  // deep-links like `navigate("settings", { tab: "modules" })` land on
   123→  // the right tab when the page first mounts. Subsequent tab clicks
   124→  // are pure local state — we don't push back to the URL (avoids the
   125→  // setState-in-effect anti-pattern).
   126→  const [activeTab, setActiveTab] = useState<string>(
   127→    (router.params.tab as string) || "general",
   128→  );
   129→
   130→  const openQuickEdit = (tab: string) => {
   131→    setActiveTab(tab);
   132→    setShowQuickEdit(true);
   133→  };
   134→
   135→  const filtered = useMemo(() => {
   136→    if (!query.trim()) return SETTINGS_CARDS;
   137→    const q = query.toLowerCase();
   138→    return SETTINGS_CARDS.filter(
   139→      (c) =>
   140→        `${c.title} ${c.description} ${c.category}`.toLowerCase().includes(q),
   141→    );
   142→  }, [query]);
   143→
   144→  const grouped = useMemo(() => {
   145→    const map = new Map<SettingsCardSpec["category"], SettingsCardSpec[]>();
   146→    for (const cat of CATEGORY_ORDER) map.set(cat, []);
   147→    for (const c of filtered) map.get(c.category)!.push(c);
   148→    return map;
   149→  }, [filtered]);
   150→
   151→  const totalSections = SETTINGS_CARDS.length;
   152→  const activeModules = tenant.enabledModules.length;
   153→  const recentlyModified = new Date(tenant.createdAt).toLocaleDateString(
   154→    "en-US",
   155→    { year: "numeric", month: "short", day: "numeric" },
   156→  );
   157→
   158→  return (
   159→    <Page>
   160→      <PageHeader
   161→        title="Settings"
   162→        description="Configure branding, security, communications, and system."
   163→        icon={SettingsIcon}
   164→        actions={
   165→          <Button
   166→            variant={showQuickEdit ? "default" : "outline"}
   167→            size="sm"
   168→            className="gap-1.5"
   169→            onClick={() => setShowQuickEdit((s) => !s)}
   170→          >
   171→            <Pencil className="h-3.5 w-3.5" />
   172→            {showQuickEdit ? "Hide quick edit" : "Quick edit"}
   173→          </Button>
   174→        }
   175→      />
   176→
   177→      <PageContent>
   178→        {/* KPI row — §9 KPI rules (value + meaning, not bare metrics) */}
   179→        <div className="grid grid-cols-2 gap-3 md:grid-cols-4">
   180→          <MetricCard
   181→            label="Settings Sections"
   182→            value={totalSections}
   183→            deltaLabel="across 5 categories"
   184→            icon={SettingsIcon}
   185→          />
   186→          <MetricCard
   187→            label="Recently Modified"
   188→            value={recentlyModified}
   189→            deltaLabel="tenant creation date"
   190→            icon={RefreshCw}
   191→          />
   192→          <MetricCard
   193→            label="Active Modules"
   194→            value={activeModules}
   195→            deltaLabel={`${runtime.enabledFeatures.length} feature flags`}
   196→            icon={Package}
   197→            tone="positive"
   198→          />
   199→          <MetricCard
   200→            label="Platform Status"
   201→            value={tenant.status.charAt(0).toUpperCase() + tenant.status.slice(1)}
   202→            deltaLabel={`${tenant.plan} plan`}
   203→            icon={ShieldCheck}
   204→            tone={
   205→              tenant.status === "active"
   206→                ? "positive"
   207→                : tenant.status === "trial"
   208→                  ? "warning"
   209→                  : "default"
   210→            }
   211→          />
   212→        </div>
   213→
   214→        {/* Search bar */}
   215→        <div className="relative">
   216→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   217→          <Input
   218→            value={query}
   219→            onChange={(e) => setQuery(e.target.value)}
   220→            placeholder="Search settings sections by title or description…"
   221→            className="pl-9"
   222→            aria-label="Search settings sections"
   223→          />
   224→        </div>
   225→
   226→        {/* Landing grid — grouped by category */}
   227→        {filtered.length === 0 ? (
   228→          <EmptyState
   229→            title="No settings match"
   230→            description={`No settings sections match "${query}". Try a different term.`}
   231→            icon={Search}
   232→            hint="Settings are grouped under Branding, Security, Communications, Certificates, and System."
   233→          />
   234→        ) : (
   235→          <div className="space-y-6">
   236→            {CATEGORY_ORDER.map((cat) => {
   237→              const items = grouped.get(cat) ?? [];
   238→              if (items.length === 0) return null;
   239→              return (
   240→                <section key={cat} aria-label={`${cat} settings`}>
   241→                  <div className="mb-3 flex items-center gap-2">
   242→                    <h2 className="text-sm font-semibold tracking-tight text-foreground">
   243→                      {cat}
   244→                    </h2>
   245→                    <Badge variant="outline" className="text-[10px]">
   246→                      {items.length}
   247→                    </Badge>
   248→                  </div>
   249→                  <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   250→                    {items.map((card) => {
   251→                      const Icon = card.icon;
   252→                      const isQuickEdit = card.viewId === "settings";
   253→                      return (
   254→                        <button
   255→                          key={card.id}
   256→                          onClick={() =>
   257→                            isQuickEdit
   258→                              ? openQuickEdit(card.id)
   259→                              : navigate(card.viewId)
   260→                          }
   261→                          className="group flex h-full flex-col gap-2 rounded-lg border bg-card p-4 text-left transition-colors hover:border-primary/40 hover:bg-accent/40"
   262→                          aria-label={`Open ${card.title}`}
   263→                        >
   264→                          <div className="flex items-start justify-between">
   265→                            <div className="rounded-md border bg-muted p-1.5">
   266→                              <Icon className="h-4 w-4 text-foreground" />
   267→                            </div>
   268→                            <ChevronRight className="h-4 w-4 text-muted-foreground opacity-0 transition-opacity group-hover:opacity-100" />
   269→                          </div>
   270→                          <div className="flex-1">
   271→                            <p className="text-sm font-medium text-foreground">
   272→                              {card.title}
   273→                            </p>
   274→                            <p className="mt-0.5 text-xs text-muted-foreground">
   275→                              {card.description}
   276→                            </p>
   277→                          </div>
   278→                          <span className="text-[10px] font-medium uppercase tracking-wide text-muted-foreground/70">
   279→                            Open →
   280→                          </span>
   281→                        </button>
   282→                      );
   283→                    })}
   284→                  </div>
   285→                </section>
   286→              );
   287→            })}
   288→          </div>
   289→        )}
   290→
   291→        {/* Quick edit (collapsible 7-tab editor) */}
   292→        {showQuickEdit ? (
   293→          <Card>
   294→            <CardHeader className="pb-3">
   295→              <div className="flex items-center justify-between">
   296→                <div>
   297→                  <span className="text-sm font-medium">Quick edit</span>
   298→                  <p className="text-xs text-muted-foreground">
   299→                    Inline editors for the 7 most-changed settings groups.
   300→                    Each card above links to its full-page editor.
   301→                  </p>
   302→                </div>
   303→                <Button
   304→                  variant="ghost"
   305→                  size="sm"
   306→                  onClick={() => setShowQuickEdit(false)}
   307→                  aria-label="Hide quick edit"
   308→                >
   309→                  Hide
   310→                </Button>
   311→              </div>
   312→            </CardHeader>
   313→            <CardContent>
   314→              <Tabs
   315→                value={activeTab}
   316→                onValueChange={setActiveTab}
   317→                className="w-full"
   318→              >
   319→                <TabsList className="flex flex-wrap justify-start">
   320→                  <TabsTrigger value="general" className="gap-1">
   321→                    <SettingsIcon className="h-3 w-3" /> General
   322→                  </TabsTrigger>
   323→                  <TabsTrigger value="branding" className="gap-1">
   324→                    <Palette className="h-3 w-3" /> Branding
   325→                  </TabsTrigger>
   326→                  <TabsTrigger value="terminology" className="gap-1">

   329→                  <TabsTrigger value="modules" className="gap-1">
   330→                    <Package className="h-3 w-3" /> Modules
   331→                  </TabsTrigger>
   332→                  <TabsTrigger value="roles" className="gap-1">
   333→                    <Users className="h-3 w-3" /> Roles
   334→                  </TabsTrigger>
   335→                  <TabsTrigger value="integrations" className="gap-1">
   336→                    <Plug className="h-3 w-3" /> Integrations
   337→                  </TabsTrigger>
   338→                  <TabsTrigger value="notifications" className="gap-1">
   339→                    <Bell className="h-3 w-3" /> Notifications
   340→                  </TabsTrigger>
   341→                </TabsList>
   342→                <TabsContent value="general">

   346→                  <BrandingTab />
   347→                </TabsContent>

   353→                </TabsContent>
   354→                <TabsContent value="roles">
   355→                  <RolesTab />
   356→                </TabsContent>

   358→                  <IntegrationsTab />
   359→                </TabsContent>

   361→                  <NotificationsTab />
   362→                </TabsContent>
   363→              </Tabs>
   364→            </CardContent>
   365→          </Card>
   366→        ) : null}
   367→      </PageContent>
   368→    </Page>

   372→function GeneralTab() {
   373→  const { tenant, setTenant, runtime } = usePlatform();
   374→  const [name, setName] = useState(tenant.branding.name);
   375→  const [tagline, setTagline] = useState(tenant.branding.tagline ?? "");
   376→  const [currency, setCurrency] = useState(tenant.currency);
   377→  const [timezone, setTimezone] = useState(tenant.timezone);
   378→
   379→  const save = () => {
   380→    setTenant({
   381→      ...tenant,
   382→      name,
   383→      branding: { ...tenant.branding, name, tagline },
   384→      currency,
   385→      timezone,
   386→    });
   387→    toast({ title: "Settings saved", description: "General settings updated." });
   388→  };
   389→
   390→  return (
   391→    <Card>
   392→      <CardHeader><span className="text-sm font-medium">General</span></CardHeader>
   393→      <CardContent className="space-y-4">
   394→        <div className="grid gap-4 md:grid-cols-2">
   395→          <div className="space-y-1.5">
   396→            <Label htmlFor="name">Tenant name</Label>
   397→            <Input id="name" value={name} onChange={(e) => setName(e.target.value)} />
   398→          </div>

   400→            <Label htmlFor="tagline">Tagline</Label>

   403→          <div className="space-y-1.5">

   570→    else enabled.add(moduleId);

   573→    toast({
   574→      title: enabled.has(moduleId) ? "Module enabled" : "Module disabled",
   575→      description: `${mod?.manifest.name} is now ${enabled.has(moduleId) ? "visible" : "hidden"} in navigation.`,
   576→    });
   577→  };
   578→
   579→  return (
   580→    <Card>
   581→      <CardHeader>
   582→        <div className="flex items-center justify-between">
   583→          <span className="text-sm font-medium">Module Catalog</span>
   584→          <Badge variant="secondary">{tenant.enabledModules.length} enabled</Badge>
   585→        </div>
   586→      </CardHeader>
   587→      <CardContent className="space-y-2">
