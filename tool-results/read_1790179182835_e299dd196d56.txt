    99→    99→/** Account (trader) options for the Account dropdown. */
   100→   100→function getAccountOptions(tid: string): Trader[] {
   101→   101→  const tenantTraders = getTenantTraders(tid);
   102→   102→  const pool = tenantTraders.length > 0 ? tenantTraders : allTraders;
   103→   103→  // De-duplicate by email and cap at 20 for the dropdown.
   104→   104→  const seen = new Set<string>();
   105→   105→  const out: Trader[] = [];
   106→   106→  for (const t of pool) {
   107→   107→    if (seen.has(t.email)) continue;
   108→   108→    seen.add(t.email);
   109→   109→    out.push(t);
   110→   110→    if (out.length >= 20) break;
   111→   111→  }
   112→   112→  return out;
   113→   113→}
   114→   114→
   115→   115→/** Mock linked withdrawals list. */
   116→   116→function getWithdrawalOptions(cert: IssuedCertificate) {
   117→   117→  const base = [
   118→   118→    { id: `wd-${1000 + (cert.id.length % 200)}`, amount: cert.withdrawalAmount },
   119→   119→    { id: `wd-${2000 + (cert.id.length % 200)}`, amount: cert.withdrawalAmount + 1250 },
   120→   120→    { id: `wd-${3000 + (cert.id.length % 200)}`, amount: cert.withdrawalAmount + 2750 },
   121→   121→  ];
   122→   122→  // Ensure the certificate's own withdrawal id is always selectable.
   123→   123→  if (cert.withdrawalId && !base.some((w) => w.id === cert.withdrawalId)) {
   124→   124→    base.unshift({ id: cert.withdrawalId, amount: cert.withdrawalAmount });
   125→   125→  }
   126→   126→  return base;
   127→   127→}

   129→   129→/** Tone for the Certificate Type badge (read-only display). */
   130→   130→function certTypeTone(
   131→   131→  certType: CertType,
   132→   132→): "info" | "warning" | "success" | "muted" {
   133→   133→  switch (certType) {
   134→   134→    case "Challenge Passed":
   135→   135→      return "success";
   136→   136→    case "Funded Trader":
   137→   137→      return "info";
   138→   138→    case "Competition Winner":
   139→   139→      return "warning";
   140→   140→    default:
   141→   141→      return "muted";
   142→   142→  }
   143→   143→}

   167→   167→  const id = router.params.id ?? "";

   169→   169→
   170→   170→  // Working copy — local state so edits don't mutate mock data.
   171→   171→  const [working, setWorking] = useState<IssuedCertificate>(seedCert);
   172→   172→  const [editing, setEditing] = useState(false);
   173→   173→  const [deleteOpen, setDeleteOpen] = useState(false);

   175→   175→  // Recompute working when the seed changes (e.g. navigating to another id).

   178→   178→  const accounts = useMemo(() => getAccountOptions(tid), [tid]);
   179→   179→  const templates = useMemo(() => getCertificateTemplates(), []);
   180→   180→  const withdrawals = useMemo(() => getWithdrawalOptions(seedCert), [seedCert]);
   181→   181→
   182→   182→  const update = (patch: Partial<IssuedCertificate>) =>
   183→   183→    setWorking((w) => ({ ...w, ...patch }));
   184→   184→
   185→   185→  const onToggleEdit = () => {
   186→   186→    if (editing) {
   187→   187→      // Discard changes on cancel-style toggle off.
   188→   188→      setWorking(seedCert);
   189→   189→    }
   190→   190→    setEditing((e) => !e);
   191→   191→  };
   192→   192→
   193→   193→  const onSave = () => {
   194→   194→    setEditing(false);

   196→   196→      title: "Certificate updated",
   197→   197→      description: `Changes to ${working.id} were saved.`,
   198→   198→    });
   199→   199→  };
   200→   200→
   201→   201→  const onSaveAndContinue = () => {
   202→   202→    toast({
   203→   203→      title: "Changes saved",
   204→   204→      description: `Certificate ${working.id} updated. Continuing edits.`,
   205→   205→    });

   216→   216→    navigate("certificates-issued");
   217→   217→  };
   218→   218→
   219→   219→  const onDelete = () => {
   220→   220→    setDeleteOpen(false);
   221→   221→    toast({
   222→   222→      title: "Certificate deleted",
   223→   223→      description: `${working.id} was permanently deleted.`,
   224→   224→      // tone: "destructive" — the platform toast API doesn't expose a tone
   225→   225→      // prop in this version; the description carries the gravity.
   226→   226→    });
   227→   227→    navigate("certificates-issued");
   228→   228→  };
   229→   229→
   230→   230→  const onCopyUrl = async () => {
   231→   231→    try {
   232→   232→      if (navigator?.clipboard?.writeText) {
   233→   233→        await navigator.clipboard.writeText(working.certUrl);

   235→   235→          title: "URL copied",
   236→   236→          description: "Certificate URL copied to clipboard.",
   237→   237→        });

   239→   239→        throw new Error("Clipboard API unavailable");
   240→   240→      }
   241→   241→    } catch {
   242→   242→      toast({
   243→   243→        title: "Couldn't copy automatically",
   244→   244→        description: "Copy the URL manually from the field below.",
   245→   245→      });
   246→   246→    }
   247→   247→  };
   248→   248→
   249→   249→  const onAccountChange = (email: string) => {
   250→   250→    const t = accounts.find((a) => a.email === email);
   251→   251→    if (!t) return;
   252→   252→    update({
   253→   253→      accountEmail: t.email,
   254→   254→      traderName: t.name,
   255→   255→      traderId: t.id,
   256→   256→    });
   257→   257→  };
   258→   258→
   259→   259→  const onWithdrawalChange = (wdId: string) => {
   260→   260→    const wd = withdrawals.find((w) => w.id === wdId);
   261→   261→    if (!wd) return;
   262→   262→    update({ withdrawalId: wd.id, withdrawalAmount: wd.amount });
   263→   263→  };
   264→   264→

   274→   274→            >

   278→   278→              >

   294→   294→      {/* Header row */}

   302→   302→              {working.traderName}

   349→   349→          currency={currency}
