   100→const daysAgoShort = (n: number) => {

   291→/** Heatmap cell background — interpolates green → amber → rose by % value. */
   292→function retentionBg(pct: number): string {
   293→  if (pct === 0) return "transparent";
   294→  if (pct >= 85) return "rgba(22,163,74,0.92)";
   295→  if (pct >= 70) return "rgba(22,163,74,0.72)";
   296→  if (pct >= 55) return "rgba(217,119,6,0.72)";
   297→  if (pct >= 40) return "rgba(217,119,6,0.92)";
   298→  return "rgba(225,29,72,0.82)";
   299→}
   300→
   301→function retentionText(pct: number): string {
   302→  if (pct === 0) return "var(--muted-foreground)";
   303→  return "#ffffff";
   304→}
   305→
   306→function pctRate(pass: number | null, fail: number | null): string {
   307→  if (pass === null || fail === null) return "—";
   308→  const total = pass + fail;
   309→  if (total === 0) return "—";
   310→  return `${Math.round((fail / total) * 100)}%`;
   311→}

   317→/** Mock: daily pass/fail counts over the last 30 days. */
   318→function buildPassFailSeries() {
   319→  return Array.from({ length: 30 }, (_, i) => {
   320→    const n = 29 - i;
   321→    const pass = Math.max(
   322→      2,
   323→      Math.round(8 + Math.sin(n / 3) * 4 + Math.cos(n / 7) * 2 + n * 0.1),
   324→    );
   325→    const fail = Math.max(
   326→      1,
   327→      Math.round(6 + Math.cos(n / 4) * 3 + n * 0.08),
   328→    );
   329→    return { date: daysAgoShort(n), pass, fail };
   330→  });
   331→}
