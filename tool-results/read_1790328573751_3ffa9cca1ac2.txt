   115→function relativeTime(iso: string): string {
   116→  const diff = Date.now() - new Date(iso).getTime();
   117→  const hours = Math.floor(diff / (1000 * 60 * 60));
   118→  if (hours < 1) return "just now";
   119→  if (hours < 24) return `${hours}h ago`;
   120→  const days = Math.floor(hours / 24);
   121→  if (days < 30) return `${days}d ago`;
   122→  const months = Math.floor(days / 30);
   123→  return `${months}mo ago`;
   124→}

   141→function activityFor(c: CrmContact): ActivityEntry[] {
   142→  const seed = hashStr(c.id);
   143→  // Pick 3-5 entries deterministically based on the contact hash + stage.
   144→  const count = 3 + (seed % 3);
   145→  const created = new Date(c.lastInteraction).getTime();
   146→  const out: ActivityEntry[] = [];
   147→  for (let i = 0; i < count; i++) {
   148→    const tpl = ACTIVITY_TEMPLATES[(seed + i) % ACTIVITY_TEMPLATES.length];
   149→    // Spread timestamps backwards from last interaction.
   150→    const ts = new Date(created - i * 86_400_000).toISOString();
   151→    out.push({ ...tpl, timestamp: ts });
   152→  }
   153→  return out;
   154→}

   165→function dealsFor(c: CrmContact): MockDeal[] {
   166→  const seed = hashStr(c.id);
   167→  // Stage 'opportunity' or 'customer' contacts get deals; 'lead'/'churned' don't.
   168→  if (c.stage === "lead" || c.stage === "churned") return [];
   169→  const count = (seed % 3) + 1; // 1–3 deals
   170→  const out: MockDeal[] = [];
   171→  for (let i = 0; i < count; i++) {
   172→    const stages: MockDeal["stage"][] = ["qualified", "negotiation", "closed-won", "closed-lost"];
   173→    out.push({
   174→      id: `deal-${c.id}-${i}`,
   175→      name: DEAL_NAMES[(seed + i) % DEAL_NAMES.length],
   176→      amount: 500 + ((seed >>> (i + 1)) % 8) * 500,
   177→      stage: c.stage === "customer" ? "closed-won" : stages[(seed >>> (i + 2)) % stages.length],
   178→    });
   179→  }
   180→  return out;
   181→}

   198→/**
   199→ * Deterministic lead score (0–100) derived from contact attributes.
   200→ * No Math.random — every render yields the same score for the same contact.
   201→ *
   202→ * Factors (cumulative):
   203→ *   +50  base
   204→ *   +20  pipeline value > $5,000
   205→ *   +10  pipeline value > $15,000  (stacks with the +20 above)
   206→ *   +15  source is referral or affiliate (case-insensitive)
   207→ *   +15  last interaction < 3 days
   208→ *   +8   last interaction < 7 days  (mutually exclusive with the +15)
   209→ *   -10  last interaction > 30 days
   210→ *   +12  contact has at least one associated deal
   211→ *
   212→ * Final value clamped to 0–100.
   213→ */
   214→function computeLeadScore(contact: CrmContact): number {
   215→  let score = 50;
   216→  if (contact.value && contact.value > 5000) score += 20;
   217→  if (contact.value && contact.value > 15000) score += 10;
   218→  const src = (contact.source ?? "").toLowerCase();
   219→  if (src === "referral" || src === "affiliate") score += 15;
   220→  if (contact.lastInteraction) {
   221→    const daysSince =
   222→      (Date.now() - new Date(contact.lastInteraction).getTime()) /
   223→      (1000 * 60 * 60 * 24);
   224→    if (daysSince < 3) score += 15;
   225→    else if (daysSince < 7) score += 8;
   226→    else if (daysSince > 30) score -= 10;
   227→  }
   228→  if (dealsFor(contact).length > 0) score += 12;
   229→  return Math.max(0, Math.min(100, score));
   230→}
   231→
   232→type LeadScoreTone = "hot" | "warm" | "cold";
   233→
   234→function leadScoreTone(score: number): LeadScoreTone {
   235→  if (score >= 80) return "hot";
   236→  if (score >= 50) return "warm";
   237→  return "cold";
   238→}
   239→

   242→  warm: "border-transparent bg-amber-100 text-amber-700 dark:bg-amber-950 dark:text-amber-400",
   243→  cold: "border-transparent bg-rose-100 text-rose-700 dark:bg-rose-950 dark:text-rose-400",
   244→};

   246→const LEAD_SCORE_TONE_LABEL: Record<LeadScoreTone, string> = {
