   160→
   161→function resolveChallengeType(id: string | undefined): {
   162→  type: ChallengeType;
   163→  index: number;
   164→} {
   165→  const types = getChallengeTypes();
   166→  if (!id) return { type: types[0], index: 0 };
   167→  const numeric = Number(id);
   168→  if (Number.isInteger(numeric) && numeric >= 0 && numeric < types.length) {
   169→    return { type: types[numeric], index: numeric };
   170→  }
   171→  const byId = types.findIndex((t) => t.id === id);
   172→  if (byId >= 0) return { type: types[byId], index: byId };
   173→  return { type: types[0], index: 0 };
   174→}
