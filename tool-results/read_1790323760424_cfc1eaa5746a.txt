    90→const DATE_RANGES: Record<string, number> = {
    91→  "24h": 24 * 60 * 60 * 1000,
    92→  "7d": 7 * 24 * 60 * 60 * 1000,
    93→  "30d": 30 * 24 * 60 * 60 * 1000,
    94→  "90d": 90 * 24 * 60 * 60 * 1000,
    95→};

    97→/** Deterministic time delta in seconds from a seed index. */
    98→function deltaFromIndex(i: number): number {
    99→  const cycle = [2, 5, 11, 23, 47, 91, 137, 213];
   100→  return cycle[i % cycle.length];
   101→}
   102→
   103→/** Build a deterministic list of copy-trading event pairs from positions. */
   104→function buildCopyEvents(tenantId: string): CopyEvent[] {
   105→  const positions = getTenantPositions(tenantId);
   106→  const accounts = getTenantAccounts(tenantId);
   107→  const accountById = new Map(accounts.map((a) => [a.id, a]));
   108→  if (positions.length < 2) return [];
   109→
   110→  // Pair positions sequentially: (0,1), (2,3), (4,5), … when symbols match.
   111→  const pairs: Array<[Position, Position]> = [];
   112→  for (let i = 0; i + 1 < positions.length; i += 2) {
   113→    const a = positions[i];
   114→    const b = positions[i + 1];
   115→    // Use the same symbol for both sides (typical copy pattern); pick from a.
   116→    pairs.push([a, b]);
   117→  }
   118→
   119→  const now = Date.now();
   120→  return pairs.map(([p1, p2], idx) => {
   121→    const acct1 = accountById.get(p1.accountId);
   122→    const acct2 = accountById.get(p2.accountId);
   123→    const openDelta = deltaFromIndex(idx);
   124→    const closeDelta = deltaFromIndex(idx + 3);
   125→    const createdAtMs =
   126→      now - ((idx % 14) + 1) * 24 * 60 * 60 * 1000 - (idx % 7) * 60 * 60 * 1000;
   127→    return {
   128→      id: `cte-${tenantId}-${idx + 1}`,
   129→      tenantId,
   130→      position1Id: p1.id,
   131→      position2Id: p2.id,
   132→      symbol: p1.symbol,
   133→      side1: p1.side,
   134→      side2: p2.side,
   135→      accountLogin1: acct1?.login ?? p1.accountId,
   136→      accountLogin2: acct2?.login ?? p2.accountId,
   137→      accountId1: p1.accountId,
   138→      accountId2: p2.accountId,
   139→      traderId1: p1.traderId,
   140→      traderId2: p2.traderId,
   141→      openTimeDeltaSec: openDelta,
   142→      closeTimeDeltaSec: closeDelta,
   143→      // Alternate expired state deterministically — about 1 in 4 expired.
   144→      expired: idx % 4 === 3,
   145→      createdAt: new Date(createdAtMs).toISOString(),
   146→      reasons: `Synchronized open/close timing on ${p1.symbol}; ${openDelta}s open delta; ${closeDelta}s close delta; both accounts share identical volume profile.`,
   147→    };
   148→  });
   149→}

   171→
   172→function formatDelta(seconds: number): string {
   173→  if (seconds < 60) return `+${seconds}s`;
   174→  const m = Math.floor(seconds / 60);
   175→  const s = seconds % 60;
   176→  if (m < 60) return `+${m}m ${s}s`;
   177→  const h = Math.floor(m / 60);
   178→  return `+${h}h ${m % 60}m`;
   179→}

   615→          {activeFilters > 0 ? (

   620→        </div>
