   155→/** Deterministic pseudo-random generator from a numeric seed. */
   156→function seededRandom(seed: number): number {
   157→  const x = Math.sin(seed * 9999.1) * 10000;
   158→  return x - Math.floor(x);
   159→}
   160→
   161→/** Hash a string id to a stable integer seed. */
   162→function hashSeed(id: string): number {
   163→  let h = 0;
   164→  for (let i = 0; i < id.length; i++) {
   165→    h = (h * 31 + id.charCodeAt(i)) | 0;
   166→  }
   167→  return Math.abs(h) || 1;
   168→}

   170→/**
   171→ * Generate a single deterministic closed position from an id. The shape
   172→ * mirrors `generateClosedPositions` on the list page so opening a row
   173→ * there would show matching data.
   174→ */
   175→function generateClosedPositionDetail(
   176→  id: string,
   177→  tenantId: string,
   178→): ClosedPositionDetail {
   179→  const accounts = getTenantAccounts(tenantId);
   180→  const traders = getTenantTraders(tenantId);
   181→  const openPositions = getTenantPositions(tenantId);
   182→
   183→  const seed = hashSeed(id);
   184→  const idx = seed % Math.max(openPositions.length, 1);
   185→  const base = openPositions[idx] ?? openPositions[0];
   186→
   187→  const [sym, basePrice, mult, description] =
   188→    BASE_SYMBOLS[(seed - 1) % BASE_SYMBOLS.length];
   189→
   190→  const r1 = seededRandom(seed + 0.1);
   191→  const r2 = seededRandom(seed + 0.5);
   192→  const r3 = seededRandom(seed + 1.7);
   193→  const r4 = seededRandom(seed + 3.3);
   194→
   195→  const side: ClosedSide = base.side;
   196→  const entry = base.entryPrice;
   197→  const movePct = (r1 - 0.5) * 0.04; // ±2%
   198→  const close = entry * (1 + movePct);
   199→  const pnlPerUnit = side === "buy" ? close - entry : entry - close;
   200→  const volume = base.volume;
   201→  // JPY-quoted pairs (USDJPY, …) need /currentPrice conversion to land
   202→  // in USD — otherwise P&L was 150x too high for USDJPY (the raw value
   203→  // is in JPY, not USD).
   204→  const isJpyQuote = sym.endsWith("JPY");
   205→  const grossPnlRaw = pnlPerUnit * mult * volume;
   206→  const grossPnl = isJpyQuote ? grossPnlRaw / close : grossPnlRaw;
   207→  const commission = Math.round((volume * basePrice * 0.0004) * 100) / 100;
   208→  const swap = Math.round(Math.sin(seed) * 5 * 100) / 100;
   209→  // Net profit = gross - commission - swap. Previously `profit` already
   210→  // subtracted commission+swap, then `netProfit` subtracted them AGAIN —
   211→  // displaying a value that was off by exactly (commission + swap).
   212→  const profit = Math.round((grossPnl - commission - swap) * 100) / 100;
   213→  const netProfit = profit;
   214→
   215→  // Determine close reason from P&L direction
   216→  let closeReason: CloseReason;
   217→  if (profit >= Math.abs(grossPnl) * 0.5 && profit > 0) closeReason = "TP";
   218→  else if (profit <= -Math.abs(grossPnl) * 0.3 && profit < 0) closeReason = "SL";
   219→  else if (r4 > 0.92) closeReason = "Liquidation";
   220→  else if (r4 > 0.8) closeReason = "System";
   221→  else closeReason = "Manual";
   222→
   223→  // SL / TP — derive from side and entry
   224→  const slDistance = entry * 0.012;
   225→  const tpDistance = entry * 0.024;
   226→  const stopLoss =
   227→    side === "buy" ? entry - slDistance : entry + slDistance;
   228→  const takeProfit =
   229→    side === "buy" ? entry + tpDistance : entry - tpDistance;
   230→  const rrRatio =
   231→    Math.abs(takeProfit - entry) > 0 && Math.abs(entry - stopLoss) > 0
   232→      ? Math.round(
   233→          (Math.abs(takeProfit - entry) / Math.abs(entry - stopLoss)) * 100,
   234→        ) / 100
   235→      : null;
   236→
   237→  // Open / close timestamps
   238→  const now = Date.now();
   239→  const openedMsAgo = Math.floor((1 + r2 * 90 * 24 * 60)) * 60 * 1000;
   240→  const durationMs = Math.floor((r3 * 48 * 60 + 5) * 60 * 1000);
   241→  const closedMsAgo = Math.max(openedMsAgo - durationMs, 60 * 1000);
   242→  const openTime = new Date(now - openedMsAgo).toISOString();
   243→  const closeTime = new Date(now - closedMsAgo).toISOString();
   244→
   245→  const trader = traders.find((t) => t.id === base.traderId);
   246→  const account = accounts.find((a) => a.id === base.accountId);
   247→
   248→  return {
   249→    id,
   250→    uid: `uid-${seed.toString(16).padStart(8, "0")}`,
   251→    direction: side,
   252→    state: "CLOSED",
   253→    symbol: sym,
   254→    symbolDescription: description,
   255→    positionType: "Market",
   256→    entryType: "In",
   257→    volume,
   258→    openPrice: Math.round(entry * 100000) / 100000,
   259→    closePrice: Math.round(close * 100000) / 100000,
   260→    currentPrice: Math.round(close * 100000) / 100000,
   261→    openTime,
   262→    closeTime,
   263→    durationMs,
   264→    openOrderId: `ord-${(seed % 90000) + 1000}`,
   265→    closeOrderId: `ord-${(seed % 90000) + 1000 + 7}`,
   266→    profit,
   267→    commission,
   268→    swap,
   269→    netProfit,
   270→    stopLoss: Math.round(stopLoss * 100000) / 100000,
   271→    takeProfit: Math.round(takeProfit * 100000) / 100000,
   272→    rrRatio,
   273→    isPartial: r4 > 0.85,
   274→    closeReason,
   275→    traderId: trader?.id ?? "trader-unknown",
   276→    traderName: trader?.name ?? "Unknown trader",
   277→    accountId: account?.id ?? base.accountId,
   278→    accountLogin: account?.login ?? "—",
   279→  };
   280→}

   286→function formatDuration(ms: number): string {
   287→  const totalMinutes = Math.round(ms / 60000);
   288→  const h = Math.floor(totalMinutes / 60);
   289→  const m = totalMinutes % 60;
   290→  if (h >= 24) {
   291→    const d = Math.floor(h / 24);
   292→    const rh = h % 24;
   293→    return `${d}d ${rh}h`;
   294→  }
   295→  if (h === 0) return `${m}m`;
   296→  return `${h}h ${m}m`;
   297→}

   299→function reasonTone(reason: CloseReason): "success" | "danger" | "warning" | "muted" {
   300→  switch (reason) {
   301→    case "TP":
   302→      return "success";
   303→    case "SL":
   304→    case "Liquidation":
   305→      return "danger";
   306→    case "System":
   307→      return "warning";
   308→    case "Manual":
   309→    default:
   310→      return "muted";
   311→  }
   312→}

   322→

   324→  const seed = useMemo(
   325→    () => generateClosedPositionDetail(id, tid),
   326→    [id, tid],

   328→
   329→  // Working copy — local state so edits don't mutate mock data.
   330→  const [working, setWorking] = useState<ClosedPositionDetail>(seed);
   331→  const [editing, setEditing] = useState(false);
   332→  const [deleteOpen, setDeleteOpen] = useState(false);
   333→
   334→  // Render-time resync — when the URL `id` changes (in-app navigation
   335→  // between two closed positions), the useState initial value above is
   336→  // already stale. Without this guard, the page briefly renders the
   337→  // previous position's data until the user edits a field. Mirrors the
   338→  // pattern in profile-page.tsx.
   339→  const [lastId, setLastId] = useState(id);
   340→  if (lastId !== id) {
   341→    setLastId(id);
   342→    setWorking(seed);
   343→  }
   344→
   345→  // Recompute working when the seed changes (navigating to another id).
   346→  // Use a key-based remount of the inner form below to flush stale edits.
   347→  const update = (patch: Partial<ClosedPositionDetail>) =>
   348→    setWorking((w) => ({ ...w, ...patch }));
   349→
   350→  const onToggleEdit = () => {
   351→    if (editing) {
   352→      // Discard local edits on toggle-off.
   353→      setWorking(seed);
   354→    }
   355→    setEditing((e) => !e);
   356→  };
   357→
   358→  const onSave = () => {
   359→    setEditing(false);
   360→    toast({
   361→      title: "Position updated",
   362→      description: `Changes to ${working.id} were saved.`,
   363→    });
   364→  };
   365→
   366→  const onSaveAndContinue = () => {
   367→    toast({
   368→      title: "Changes saved",
   369→      description: `Position ${working.id} updated. Continuing edits.`,
   370→    });
   371→  };
   372→
   373→  const onDelete = () => {
   374→    setDeleteOpen(false);
   375→    toast({
   376→      title: "Closed position deleted",
   377→      description: `${working.id} was permanently deleted along with its P&L and audit trail.`,
   378→    });
   379→    navigate("closed-positions");
   380→  };
   381→
   382→  return (

   389→              asChild
   390→              className="cursor-pointer text-muted-foreground"

   392→              <button
   393→                type="button"
   394→                onClick={() => navigate("closed-positions")}
   395→              >
   396→                Closed Positions

   399→          </BreadcrumbItem>

   403→          <BreadcrumbItem>
   404→            <BreadcrumbPage className="font-mono text-xs">
   405→              {working.id}

   407→          </BreadcrumbItem>
   408→        </BreadcrumbList>
   409→      </Breadcrumb>
   410→
   411→      {/* Header row */}
   412→      <div className="flex flex-col gap-3 md:flex-row md:items-start md:justify-between">
