   122→export function SettingsPage() {
   123→  const { router, navigate, tenant, runtime } = usePlatform();
   124→  const [query, setQuery] = useState("");
   125→  const [showQuickEdit, setShowQuickEdit] = useState<boolean>(
   126→    Boolean(router.params.tab),
   127→  );
   128→
   129→  // Quick edit's active tab — initialized from the URL `?tab=` param so
   130→  // deep-links like `navigate("settings", { tab: "modules" })` land on
   131→  // the right tab when the page first mounts. Subsequent tab clicks
   132→  // are pure local state — we don't push back to the URL (avoids the
   133→  // setState-in-effect anti-pattern).
   134→  const [activeTab, setActiveTab] = useState<string>(
   135→    (router.params.tab as string) || "general",
   136→  );
   137→
   138→  const openQuickEdit = (tab: string) => {
   139→    setActiveTab(tab);
   140→    setShowQuickEdit(true);
   141→  };
   142→
   143→  const filtered = useMemo(() => {
   144→    if (!query.trim()) return SETTINGS_CARDS;
   145→    const q = query.toLowerCase();
   146→    return SETTINGS_CARDS.filter(
   147→      (c) =>
   148→        `${c.title} ${c.description} ${c.category}`.toLowerCase().includes(q),
   149→    );
   150→  }, [query]);
   151→
   152→  const grouped = useMemo(() => {
   153→    const map = new Map<SettingsCardSpec["category"], SettingsCardSpec[]>();
   154→    for (const cat of CATEGORY_ORDER) map.set(cat, []);
   155→    for (const c of filtered) map.get(c.category)!.push(c);
   156→    return map;
   157→  }, [filtered]);
   158→
   159→  const totalSections = SETTINGS_CARDS.length;
   160→  const activeModules = tenant.enabledModules.length;
   161→  const recentlyModified = new Date(tenant.createdAt).toLocaleDateString(
   162→    "en-US",
   163→    { year: "numeric", month: "short", day: "numeric" },
   164→  );
   165→
   166→  return (
   167→    <Page>
   168→      <PageHeader
   169→        title="Settings"
   170→        description="Configure branding, security, communications, and system."
   171→        icon={SettingsIcon}
   172→        actions={
   173→          <Button
   174→            variant={showQuickEdit ? "default" : "outline"}
   175→            size="sm"
   176→            className="gap-1.5"
   177→            onClick={() => setShowQuickEdit((s) => !s)}
   178→          >
   179→            <Pencil className="h-3.5 w-3.5" />
   180→            {showQuickEdit ? "Hide quick edit" : "Quick edit"}
   181→          </Button>
   182→        }
   183→      />
   184→
   185→      <PageContent>
   186→        {/* KPI row — §9 KPI rules (value + meaning, not bare metrics) */}
   187→        <div className="grid grid-cols-2 gap-3 md:grid-cols-4">
   188→          <MetricCard
   189→            label="Settings Sections"
   190→            value={totalSections}
   191→            deltaLabel="across 5 categories"
   192→            icon={SettingsIcon}
   193→          />
   194→          <MetricCard
   195→            label="Tenant Age"
   196→            value={recentlyModified}
   197→            deltaLabel="since tenant created"
   198→            icon={RefreshCw}
   199→          />
   200→          <MetricCard
   201→            label="Active Modules"
   202→            value={activeModules}
   203→            deltaLabel={`${runtime.enabledFeatures.length} feature flags`}
   204→            icon={Package}
   205→            tone="positive"
   206→          />
   207→          <MetricCard
   208→            label="Firm Status"
   209→            value={tenant.status.charAt(0).toUpperCase() + tenant.status.slice(1)}
   210→            deltaLabel={`${tenant.plan} plan`}
   211→            icon={ShieldCheck}
   212→            tone={
   213→              tenant.status === "active"
   214→                ? "positive"
   215→                : tenant.status === "trial"
   216→                  ? "warning"
   217→                  : "default"
   218→            }
   219→          />
   220→        </div>
   221→
   222→        {/* Search bar */}
   223→        <div className="relative">
   224→          <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   225→          <Input
   226→            value={query}
   227→            onChange={(e) => setQuery(e.target.value)}
   228→            placeholder="Search settings sections by title or description…"
   229→            className="pl-9"
   230→            aria-label="Search settings sections"
   231→          />
   232→        </div>
   233→
   234→        {/* Landing grid — grouped by category */}
   235→        {filtered.length === 0 ? (
   236→          <EmptyState
   237→            title="No settings match"
   238→            description={`No settings sections match "${query}". Try a different term.`}
   239→            icon={Search}
   240→            hint="Settings are grouped under Branding, Security, Communications, Certificates, and System."
   241→          />
   242→        ) : (
   243→          <div className="space-y-6">
   244→            {CATEGORY_ORDER.map((cat) => {
   245→              const items = grouped.get(cat) ?? [];
   246→              if (items.length === 0) return null;
   247→              return (
   248→                <section key={cat} aria-label={`${cat} settings`}>
   249→                  <div className="mb-3 flex items-center gap-2">
   250→                    <h2 className="text-sm font-semibold tracking-tight text-foreground">
   251→                      {cat}
   252→                    </h2>
   253→                    <Badge variant="outline" className="text-[10px]">
   254→                      {items.length}
   255→                    </Badge>
   256→                  </div>
   257→                  <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   258→                    {items.map((card) => {
   259→                      const Icon = card.icon;
   260→                      const isQuickEdit = card.viewId === "settings";
   261→                      return (
   262→                        <button
   263→                          key={card.id}
   264→                          onClick={() =>
   265→                            isQuickEdit
   266→                              ? openQuickEdit(card.id)
   267→                              : navigate(card.viewId)
   268→                          }
   269→                          className="group flex h-full flex-col gap-2 rounded-lg border bg-card p-4 text-left transition-colors hover:border-primary/40 hover:bg-accent/40"
   270→                          aria-label={`Open ${card.title}`}
   271→                        >
   272→                          <div className="flex items-start justify-between">
   273→                            <div className="rounded-md border bg-muted p-1.5">
   274→                              <Icon className="h-4 w-4 text-foreground" />
   275→                            </div>
   276→                            <ChevronRight className="h-4 w-4 text-muted-foreground opacity-0 transition-opacity group-hover:opacity-100" />
   277→                          </div>
   278→                          <div className="flex-1">
   279→                            <p className="text-sm font-medium text-foreground">
   280→                              {card.title}
   281→                            </p>
   282→                            <p className="mt-0.5 text-xs text-muted-foreground">
   283→                              {card.description}
   284→                            </p>
   285→                          </div>
   286→                          <span className="text-[10px] font-medium uppercase tracking-wide text-muted-foreground/70">
   287→                            Open →
   288→                          </span>
   289→                        </button>
   290→                      );
   291→                    })}
   292→                  </div>
   293→                </section>
   294→              );
   295→            })}
   296→          </div>
   297→        )}
   298→
   299→        {/* Quick edit (collapsible 7-tab editor) */}
   300→        {showQuickEdit ? (
   301→          <Card>
   302→            <CardHeader className="pb-3">
   303→              <div className="flex items-center justify-between">
   304→                <div>
   305→                  <span className="text-sm font-medium">Quick edit</span>
   306→                  <p className="text-xs text-muted-foreground">
   307→                    Inline editors for the 7 most-changed settings groups.
   308→                    Each card above links to its full-page editor.
   309→                  </p>
   310→                </div>
   311→                <Button
   312→                  variant="ghost"
   313→                  size="sm"
   314→                  onClick={() => setShowQuickEdit(false)}
   315→                  aria-label="Hide quick edit"
   316→                >
   317→                  Hide
   318→                </Button>
   319→              </div>
   320→            </CardHeader>
   321→            <CardContent>
   322→              <Tabs
   323→                value={activeTab}
   324→                onValueChange={setActiveTab}
   325→                className="w-full"
   326→              >
   327→                <TabsList className="flex flex-wrap justify-start">
   328→                  <TabsTrigger value="general" className="gap-1">
   329→                    <SettingsIcon className="h-3 w-3" /> General
   330→                  </TabsTrigger>

   332→                    <Palette className="h-3 w-3" /> Branding

   339→                  </TabsTrigger>
   340→                  <TabsTrigger value="roles" className="gap-1">
   341→                    <Users className="h-3 w-3" /> Roles

   344→                    <Plug className="h-3 w-3" /> Integrations

   380→function GeneralTab() {
   381→  const { tenant, setTenant, runtime } = usePlatform();
   382→  const [name, setName] = useState(tenant.branding.name);
   383→  const [tagline, setTagline] = useState(tenant.branding.tagline ?? "");
   384→  const [currency, setCurrency] = useState(tenant.currency);
   385→  const [timezone, setTimezone] = useState(tenant.timezone);
   386→
   387→  const save = () => {
   388→    setTenant({
   389→      ...tenant,
   390→      name,
   391→      branding: { ...tenant.branding, name, tagline },
   392→      currency,
   393→      timezone,
   394→    });
   395→    toast({ title: "Settings saved", description: "General settings updated." });
   396→  };
   397→
   398→  return (
   399→    <Card>
   400→      <CardHeader><span className="text-sm font-medium">General</span></CardHeader>
   401→      <CardContent className="space-y-4">
   402→        <div className="grid gap-4 md:grid-cols-2">
   403→          <div className="space-y-1.5">

   405→            <Input id="name" value={name} onChange={(e) => setName(e.target.value)} />
   406→          </div>

   411→          <div className="space-y-1.5">
