    76→function relativeTime(iso: string): string {
    77→  const diff = Date.now() - new Date(iso).getTime();
    78→  const hours = Math.floor(diff / (1000 * 60 * 60));
    79→  if (hours < 1) return "just now";
    80→  if (hours < 24) return `${hours}h ago`;
    81→  const days = Math.floor(hours / 24);
    82→  if (days < 30) return `${days}d ago`;
    83→  const months = Math.floor(days / 30);
    84→  return `${months}mo ago`;
    85→}

    94→const ACTIVITY_TEMPLATES: Omit<ActivityEntry, "timestamp">[] = [
    95→  { label: "Joined via landing page", detail: "Organic search — 'prop firm challenge' campaign", icon: "join" },
    96→  { label: "Opened email campaign", detail: "Q4 challenge promotion — opened 3 times", icon: "email" },
    97→  { label: "Clicked affiliate link", detail: "Affiliate code 'BONUS50' applied at checkout", icon: "click" },
    98→  { label: "Requested demo", detail: "Booked a 15-minute onboarding call with Sarah", icon: "demo" },

   102→function activityFor(c: CrmContact): ActivityEntry[] {
   103→  const seed = hashStr(c.id);
   104→  // Pick 3-5 entries deterministically based on the contact hash + stage.
   105→  const count = 3 + (seed % 3);
   106→  const created = new Date(c.lastInteraction).getTime();
   107→  const out: ActivityEntry[] = [];
   108→  for (let i = 0; i < count; i++) {
   109→    const tpl = ACTIVITY_TEMPLATES[(seed + i) % ACTIVITY_TEMPLATES.length];
   110→    // Spread timestamps backwards from last interaction.
   111→    const ts = new Date(created - i * 86_400_000).toISOString();
   112→    out.push({ ...tpl, timestamp: ts });
   113→  }
   114→  return out;
   115→}

   117→interface MockDeal {
   118→  id: string;
   119→  name: string;
   120→  amount: number;
   121→  stage: "qualified" | "negotiation" | "closed-won" | "closed-lost";
   122→}
   123→
   124→const DEAL_NAMES = ["100k Challenge", "50k Reset Bundle", "Affiliate Boost Pack", "Annual Pro Tier"];
   125→
   126→function dealsFor(c: CrmContact): MockDeal[] {
   127→  const seed = hashStr(c.id);
   128→  // Stage 'opportunity' or 'customer' contacts get deals; 'lead'/'churned' don't.
   129→  if (c.stage === "lead" || c.stage === "churned") return [];
   130→  const count = (seed % 3) + 1; // 1–3 deals
   131→  const out: MockDeal[] = [];
   132→  for (let i = 0; i < count; i++) {
   133→    const stages: MockDeal["stage"][] = ["qualified", "negotiation", "closed-won", "closed-lost"];
   134→    out.push({
   135→      id: `deal-${c.id}-${i}`,
   136→      name: DEAL_NAMES[(seed + i) % DEAL_NAMES.length],
   137→      amount: 500 + ((seed >>> (i + 1)) % 8) * 500,
   138→      stage: c.stage === "customer" ? "closed-won" : stages[(seed >>> (i + 2)) % stages.length],
   139→    });
   140→  }
   141→  return out;
   142→}

   163→export function CrmOverviewPage() {
   164→  const { runtime, tenant } = usePlatform();
   165→  const term = makeTermResolver(tenant);
   166→  const tid = runtime.tenant?.id ?? "platform";
   167→  const currency = runtime.tenant?.currency ?? "USD";
   168→  const contacts = getTenantContacts(tid);
   169→  const leads = contacts.filter((c) => c.stage === "lead").length;
   170→  const qualified = contacts.filter((c) => c.stage === "qualified").length;
   171→  const customers = contacts.filter((c) => c.stage === "customer").length;
   172→  const churned = contacts.filter((c) => c.stage === "churned").length;
   173→  const pipelineValue = contacts.filter((c) => c.stage !== "churned").reduce((s, c) => s + c.value, 0);
   174→
   175→  const stages = ["lead", "qualified", "opportunity", "customer", "churned"] as const;
   176→  const pipelineData = stages.map((stage) => ({
   177→    name: stage,
   178→    value: contacts.filter((c) => c.stage === stage).length,
   179→  }));
   180→
   181→  const handleExport = () => {
   182→    exportToCsv(
   183→      contacts,
   184→      [
   185→        { key: "name", header: "Name", value: (c) => c.name },
   186→        { key: "email", header: "Email", value: (c) => c.email },
   187→        { key: "phone", header: "Phone", value: (c) => c.phone ?? "" },
   188→        { key: "source", header: "Source", value: (c) => c.source },
   189→        { key: "stage", header: "Stage", value: (c) => c.stage },
   190→        { key: "owner", header: "Owner", value: (c) => c.owner },
   191→        { key: "value", header: "Value", value: (c) => c.value },
   192→        { key: "lastInteraction", header: "Last Interaction", value: (c) => c.lastInteraction },
   193→      ],
   194→      `crm-contacts-${Date.now()}.csv`,
   195→    );
   196→  };
   197→
   198→  return (
   199→    <Page>
   200→      <PageHeader
   201→        title="CRM"
   202→        description={`Contacts, leads, and pipeline for this ${term("trader").toLowerCase()} tenant.`}
   203→        icon={Contact}
   204→        actions={
   205→          <Button
   206→            size="sm"
   207→            variant="outline"
   208→            onClick={handleExport}
   209→          >
   210→            <Download className="mr-1 h-4 w-4" /> Export
   211→          </Button>
   212→        }
   213→      />
   214→      <PageContent>
   215→        <div className="grid grid-cols-2 gap-3 lg:grid-cols-5">
   216→          <MetricCard label="Total Contacts" value={contacts.length} delta={5} icon={Users} tone="positive" />
   217→          <MetricCard label="Leads" value={leads} delta={9} icon={UserPlus} />
   218→          <MetricCard label="Qualified" value={qualified} delta={7} icon={CheckCircle2} tone="positive" />
   219→          <MetricCard label="Customers" value={customers} delta={3} icon={Crown} tone="positive" />
   220→          <MetricCard label="Churned" value={churned} delta={-2} icon={UserMinus} tone="negative" />
   221→        </div>
   222→        <div className="grid gap-4 lg:grid-cols-2">
   223→          <div className="rounded-lg border bg-card p-4">
   224→            <p className="mb-2 text-sm font-medium">Pipeline by Stage</p>
   225→            <BarSeries data={pipelineData} xKey="name" yKey="value" color="#0891b2" />
   226→          </div>
   227→          <div className="rounded-lg border bg-card p-4">
   228→            <div className="mb-2 flex items-center justify-between">
   229→              <p className="text-sm font-medium">Pipeline Value</p>
   230→              <Badge variant="secondary" className="text-xs">{formatCurrency(pipelineValue, currency)}</Badge>
   231→            </div>
   232→            <div className="flex flex-col gap-3 pt-2">
   233→              {stages.map((stage) => {
   234→                const count = contacts.filter((c) => c.stage === stage).length;
   235→                const value = contacts.filter((c) => c.stage === stage).reduce((s, c) => s + c.value, 0);
   236→                const total = contacts.length || 1;
   237→                return (
   238→                  <div key={stage} className="flex flex-col gap-1">
   239→                    <div className="flex items-center justify-between text-xs">
   240→                      <span className="capitalize text-muted-foreground">{stage}</span>
   241→                      <span className="font-medium text-foreground">
   242→                        {count} · {formatCurrency(value, currency)}
   243→                      </span>
   244→                    </div>
   245→                    <div className="h-2 w-full overflow-hidden rounded-full bg-muted">
   246→                      <div
   247→                        className="h-full rounded-full"
   248→                        style={{ width: `${(count / total) * 100}%`, background: STAGE_COLOR[stage] }}
   249→                      />
   250→                    </div>
   251→                  </div>
   252→                );
   253→              })}
   254→            </div>
   255→          </div>
   256→        </div>
   257→      </PageContent>
   258→    </Page>
   259→  );
   260→}
