   250→/** Format a relative-time string from an ISO timestamp. */
   251→function formatRelativeTime(iso: string): string {
   252→  if (iso === "—" || !iso) return "—";
   253→  const then = new Date(iso).getTime();
   254→  const now = Date.now();
   255→  const diffMs = now - then;
   256→  if (diffMs < 0) return "just now";
   257→  const minutes = Math.floor(diffMs / 1000 / 60);
   258→  if (minutes < 1) return "just now";
   259→  if (minutes < 60) return `${minutes}m ago`;
   260→  const hours = Math.floor(minutes / 60);
   261→  if (hours < 24) return `${hours}h ago`;
   262→  const days = Math.floor(hours / 24);
   263→  if (days < 30) return `${days}d ago`;
   264→  const months = Math.floor(days / 30);
   265→  return `${months}mo ago`;
   266→}

   268→/** Deterministic trend arrow renderer. */
   269→function TrendIndicator({ delta }: { delta: number }) {
   270→  if (delta === 0) {
   271→    return <span className="text-muted-foreground/70">—</span>;
   272→  }
   273→  const up = delta > 0;
   274→  return (
   275→    <span
   276→      className={cn(
   277→        "inline-flex items-center gap-0.5 text-xs font-semibold",
   278→        up ? "text-emerald-600" : "text-rose-600",
   279→      )}
   280→    >
   281→      {up ? <TrendingUp className="h-3 w-3" /> : <TrendingDown className="h-3 w-3" />}
   282→      {up ? "+" : ""}
   283→      {delta.toFixed(1)}%
   284→    </span>
   285→  );
   286→}
