   207→/** Deterministic pseudo-random generator from a numeric seed. */
   208→function seededRandom(seed: number): number {
   209→  const x = Math.sin(seed * 9999.1) * 10000;
   210→  return x - Math.floor(x);
   211→}
   212→
   213→/** Hash a string id to a stable integer seed. */
   214→function hashSeed(id: string): number {
   215→  let h = 0;
   216→  for (let i = 0; i < id.length; i++) {
   217→    h = (h * 31 + id.charCodeAt(i)) | 0;
   218→  }
   219→  return Math.abs(h) || 1;
   220→}
   221→
   222→const BROKERS = ["MT5", "MT4", "DXTrade"];
   223→const ACCOUNT_STATUSES: LinkedAccount["status"][] = [
   224→  "active",
   225→  "passed",
   226→  "pending",
   227→  "breached",
   228→];
   229→
   230→function generateOrderDetail(id: string, tenantId: string): OrderDetail {
   231→  const seed = hashSeed(id);
   232→  const traders = getTenantTraders(tenantId);
   233→  const trader = traders[seed % Math.max(traders.length, 1)] ?? traders[0];
   234→
   235→  const r1 = seededRandom(seed + 0.1);
   236→  const r2 = seededRandom(seed + 0.5);
   237→  const r3 = seededRandom(seed + 1.7);
   238→  const r4 = seededRandom(seed + 3.3);
   239→
   240→  const orderType = ORDER_TYPES[seed % ORDER_TYPES.length];
   241→  const challenge = CHALLENGES[seed % CHALLENGES.length];
   242→  const competition = r4 > 0.7 ? COMPETITIONS[seed % COMPETITIONS.length] : null;
   243→  const accountBalance = [10000, 25000, 50000, 100000, 200000][
   244→    seed % 5
   245→  ];
   246→  const amountPaid =
   247→    orderType === "Refund"
   248→      ? -Math.round(accountBalance * 0.15 * 100) / 100
   249→      : Math.round(accountBalance * (0.05 + r1 * 0.03) * 100) / 100;
   250→  const quantity = 1 + (seed % 3);
   251→  const paymentMethod = PAYMENT_METHODS[seed % PAYMENT_METHODS.length];
   252→  const couponCode = r2 > 0.5 ? `SAVE${10 + (seed % 30)}` : "";
   253→  const bundleId = r3 > 0.6 ? `bundle-${(seed % 100) + 1}` : null;
   254→  const customerIp = `${10 + (seed % 200)}.${seed % 255}.${(seed * 7) % 255}.${(seed * 13) % 255}`;
   255→
   256→  // Addons — deterministic small object
   257→  const addons: Record<string, unknown> = {};
   258→  if (r4 > 0.4) addons["priority_withdrawal"] = true;
   259→  if (r4 > 0.6) addons["extended_kyc"] = false;
   260→  if (r4 > 0.8) addons["addon_quantity"] = 1 + (seed % 3);
   261→  if (Object.keys(addons).length === 0) addons["none"] = true;
   262→
   263→  // Date created — deterministic window from the past 60 days
   264→  const now = Date.now();
   265→  const createdMsAgo = Math.floor((1 + r2 * 60 * 24 * 60)) * 60 * 1000;
   266→  const dateCreated = new Date(now - createdMsAgo).toISOString();
   267→
   268→  // Attribution — deterministic UTM values
   269→  const utm = {
   270→    source: ["twitter", "google", "affiliate", "telegram", "direct"][seed % 5],
   271→    campaign:
   272→      challenge.toLowerCase().replace(/\s+/g, "_") + `_${(seed % 12) + 1}`,
   273→    medium: ["cpc", "social", "email", "referral"][seed % 4],
   274→    term: ["forex", "funded", "evaluation", ""][seed % 4],
   275→    content: `ad_${(seed % 6) + 1}`,
   276→    referrer: REFERRERS[seed % REFERRERS.length],
   277→    landing: LANDING_PAGES[seed % LANDING_PAGES.length],
   278→  };
   279→
   280→  // Linked accounts — 1..3 deterministic accounts
   281→  const accountCount = 1 + (seed % 3);
   282→  const accounts: LinkedAccount[] = Array.from(
   283→    { length: accountCount },
   284→    (_, i) => {
   285→      const s = seed + i * 17;
   286→      return {
   287→        id: `acct-${s}`,
   288→        login: `${100000 + (s % 90000)}`,
   289→        phase: ["phase-1", "phase-2", "funded"][i % 3],
   290→        broker: BROKERS[i % BROKERS.length],
   291→        initialBalance: accountBalance,
   292→        status: ACCOUNT_STATUSES[s % ACCOUNT_STATUSES.length],
   293→      };
   294→    },
   295→  );
   296→
   297→  // Subscription rows — only meaningful for Subscription type but always
   298→  // generated so the accordion is never empty.
   299→  const subCount = orderType === "Subscription" ? 1 + (seed % 3) : 1;
   300→  const subscriptions: SubscriptionRow[] = Array.from(
   301→    { length: subCount },
   302→    (_, i) => {
   303→      const s = seed + i * 23;
   304→      const billingOffset = (s % 90) * 24 * 60 * 60 * 1000;
   305→      const status: SubscriptionStatus =
   306→        i === 0
   307→          ? "Active"
   308→          : (["Active", "Cancelled", "Expired"][s % 3] as SubscriptionStatus);
   309→      const next =
   310→        status === "Active"
   311→          ? new Date(now + (30 - (s % 30)) * 24 * 60 * 60 * 1000).toISOString()
   312→          : null;
   313→      return {
   314→        id: `sub-${s}`,
   315→        billingDate: new Date(now - billingOffset).toISOString(),
   316→        amount: Math.round(accountBalance * 0.05 * 100) / 100,
   317→        status,
   318→        nextBilling: next,
   319→      };
   320→    },
   321→  );
   322→
   323→  return {
   324→    id,
   325→    userId: trader?.id ?? "usr-unknown",
   326→    userEmail: trader?.email ?? "trader@example.com",
   327→    addons,
   328→    dateCreated,
   329→    orderType,
   330→    notes:
   331→      orderType === "Refund"
   332→        ? "Refund issued for duplicate challenge purchase (CS ticket #4231)."
   333→        : `Customer purchased ${quantity} × ${challenge}.`,
   334→    challenge,
   335→    competition,
   336→    accountBalance,
   337→    amountPaid,
   338→    quantity,
   339→    paymentMethod,
   340→    couponCode,
   341→    bundleId,
   342→    customerIp,
   343→    utm,
   344→    accounts,
   345→    subscriptions,
   346→  };
   347→}
