   648→function seedTraders(tenantId: string, count: number): Trader[] {
   649→  const out: Trader[] = [];
   650→  for (let i = 0; i < count; i++) {
   651→    const name = `${traderFirstNames[i % traderFirstNames.length]} ${traderLastNames[(i * 3) % traderLastNames.length]}`;
   652→    const phaseRoll = (i % 7);
   653→    const phase =
   654→      phaseRoll === 0 ? "funded" :
   655→      phaseRoll === 1 ? "none" :
   656→      phaseRoll <= 3 ? "phase-1" : "phase-2";
