   391→/* Deterministic per-link click series (30d) for the Sheet drawer's
   392→ * mini AreaSeries. Same link id → same series every render. */
   393→function clickSeriesForLink(link: AffiliateLink) {
   394→  // Seed from the link id (deterministic).
   395→  let seed = 0;
   396→  for (let i = 0; i < link.id.length; i++) seed = (seed * 31 + link.id.charCodeAt(i)) & 0x7fffffff;
   397→  const dailyAvg = Math.max(8, Math.round(link.clicks / 30));
   398→  return Array.from({ length: 30 }, (_, i) => {
   399→    const v = (seed + i * 7) & 0x7fffffff;
   400→    const wave = Math.sin((i + (seed % 12)) / 3) * 0.4 + 0.7; // 0.3..1.1
   401→    return {
   402→      date: `D${i + 1}`,
   403→      value: Math.max(2, Math.round(dailyAvg * wave + (v % 5))),
   404→    };
   405→  });
   406→}
   407→
   408→/* Deterministic per-link recent clicks (10 entries) for the Sheet
   409→ * drawer's Recent Clicks DataTable. */
   410→const COUNTRIES = ["US", "GB", "AE", "SG", "DE", "AU", "CA", "FR", "JP", "IN"];
   411→const DEVICES: Array<"Desktop" | "Mobile" | "Tablet"> = ["Desktop", "Mobile", "Tablet"];
   412→function recentClicksForLink(link: AffiliateLink): RecentClick[] {
   413→  let seed = 7;
   414→  for (let i = 0; i < link.id.length; i++) seed = (seed * 17 + link.id.charCodeAt(i)) & 0x7fffffff;
   415→  const baseConv = convRate(link.clicks, link.conversions) / 100;
   416→  const out: RecentClick[] = [];
   417→  for (let i = 0; i < 10; i++) {
   418→    const v = (seed + i * 13) & 0x7fffffff;
   419→    const dayOffset = i + 1;
   420→    const d = new Date();
   421→    d.setDate(d.getDate() - dayOffset);
   422→    d.setHours(8 + (v % 14), (v * 3) % 60, 0, 0);
   423→    out.push({
   424→      timestamp: d.toISOString(),
   425→      ip: `${100 + (v % 150)}.${(v * 7) % 250}.${(v * 11) % 250}.${(v * 13) % 250}`,
   426→      country: COUNTRIES[v % COUNTRIES.length],
   427→      device: DEVICES[v % DEVICES.length],
   428→      converted: (v % 100) / 100 < baseConv,
   429→    });
   430→  }
   431→  return out;
   432→}
