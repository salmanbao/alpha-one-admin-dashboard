    99→/** Account (trader) options for the Account dropdown. */
   100→function getAccountOptions(tid: string): Trader[] {
   101→  const tenantTraders = getTenantTraders(tid);
   102→  const pool = tenantTraders.length > 0 ? tenantTraders : allTraders;
   103→  // De-duplicate by email and cap at 20 for the dropdown.
   104→  const seen = new Set<string>();
   105→  const out: Trader[] = [];
   106→  for (const t of pool) {
   107→    if (seen.has(t.email)) continue;
   108→    seen.add(t.email);
   109→    out.push(t);
   110→    if (out.length >= 20) break;
   111→  }
   112→  return out;
   113→}
   114→
   115→/** Mock linked withdrawals list. */
   116→function getWithdrawalOptions(cert: IssuedCertificate) {
   117→  const base = [
   118→    { id: `wd-${1000 + (cert.id.length % 200)}`, amount: cert.withdrawalAmount },
   119→    { id: `wd-${2000 + (cert.id.length % 200)}`, amount: cert.withdrawalAmount + 1250 },
   120→    { id: `wd-${3000 + (cert.id.length % 200)}`, amount: cert.withdrawalAmount + 2750 },
   121→  ];
   122→  // Ensure the certificate's own withdrawal id is always selectable.
   123→  if (cert.withdrawalId && !base.some((w) => w.id === cert.withdrawalId)) {
   124→    base.unshift({ id: cert.withdrawalId, amount: cert.withdrawalAmount });
   125→  }
   126→  return base;
   127→}
   128→
   129→/** Tone for the Certificate Type badge (read-only display). */
   130→function certTypeTone(
   131→  certType: CertType,
   132→): "info" | "warning" | "success" | "muted" {
   133→  switch (certType) {
   134→    case "Challenge Passed":
   135→      return "success";
   136→    case "Funded Trader":
   137→      return "info";
   138→    case "Competition Winner":
   139→      return "warning";
   140→    default:
   141→      return "muted";
   142→  }
   143→}
   144→
   145→/** Icon for a Certificate Type. */
   146→function certTypeIcon(certType: CertType) {
   147→  switch (certType) {
   148→    case "Funded Trader":
   149→      return Award;
   150→    case "Competition Winner":
   151→      return Trophy;
   152→    case "Challenge Passed":
   153→    default:
   154→      return Check;
   155→  }
   156→}

   161→
   162→export function CertificateDetailPage() {
   163→  const { runtime, navigate, router } = usePlatform();
   164→  const tid = runtime.tenant?.id ?? "platform";
   165→  const currency = runtime.tenant?.currency ?? "USD";
   166→
   167→  const id = router.params.id ?? "";
   168→  const seedCert = useMemo(() => getIssuedCertificate(id, tid), [id, tid]);
   169→
   170→  // Working copy — local state so edits don't mutate mock data.
   171→  const [working, setWorking] = useState<IssuedCertificate>(seedCert);
   172→  const [editing, setEditing] = useState(false);
   173→  const [deleteOpen, setDeleteOpen] = useState(false);
   174→
   175→  // Recompute working when the seed changes (e.g. navigating to another id).
   176→  // useEffect-less sync via key prop on the inner form below avoids stale
   177→  // state when the user navigates between certificates.
   178→  const accounts = useMemo(() => getAccountOptions(tid), [tid]);
   179→  const templates = useMemo(() => getCertificateTemplates(), []);
   180→  const withdrawals = useMemo(() => getWithdrawalOptions(seedCert), [seedCert]);
   181→
   182→  const update = (patch: Partial<IssuedCertificate>) =>
   183→    setWorking((w) => ({ ...w, ...patch }));
   184→
   185→  const onToggleEdit = () => {
   186→    if (editing) {
   187→      // Discard changes on cancel-style toggle off.
   188→      setWorking(seedCert);
   189→    }
   190→    setEditing((e) => !e);
   191→  };
   192→
   193→  const onSave = () => {
   194→    setEditing(false);
   195→    toast({
   196→      title: "Certificate updated",
   197→      description: `Changes to ${working.id} were saved.`,
   198→    });
   199→  };
   200→
   201→  const onSaveAndContinue = () => {
   202→    toast({
   203→      title: "Changes saved",
   204→      description: `Certificate ${working.id} updated. Continuing edits.`,
   205→    });
   206→  };
   207→
   208→  const onSaveAndAdd = () => {
   209→    toast({
   210→      title: "Certificate saved",
   211→      description: "Redirecting to a fresh certificate form (demo).",
   212→    });
   213→    // The "new certificate form" view isn't implemented yet — the Issued
   214→    // Certificates list page is the closest parent. We navigate there so the
   215→    // operator can use the "Issue Certificate" button to start a new record.
   216→    navigate("certificates-issued");
   217→  };
   218→
   219→  const onDelete = () => {
   220→    setDeleteOpen(false);
   221→    toast({
   222→      title: "Certificate deleted",
   223→      description: `${working.id} was permanently deleted.`,
   224→      // tone: "destructive" — the platform toast API doesn't expose a tone
   225→      // prop in this version; the description carries the gravity.
   226→    });
   227→    navigate("certificates-issued");
   228→  };
   229→
   230→  const onCopyUrl = async () => {
   231→    try {
   232→      if (navigator?.clipboard?.writeText) {
   233→        await navigator.clipboard.writeText(working.certUrl);
   234→        toast({
   235→          title: "URL copied",
   236→          description: "Certificate URL copied to clipboard.",
   237→        });
   238→      } else {
   239→        throw new Error("Clipboard API unavailable");
   240→      }
   241→    } catch {
   242→      toast({
   243→        title: "Couldn't copy automatically",
   244→        description: "Copy the URL manually from the field below.",
   245→      });
   246→    }
   247→  };
   248→
   249→  const onAccountChange = (email: string) => {
   250→    const t = accounts.find((a) => a.email === email);
   251→    if (!t) return;
   252→    update({
   253→      accountEmail: t.email,
   254→      traderName: t.name,
   255→      traderId: t.id,
   256→    });
   257→  };
   258→
   259→  const onWithdrawalChange = (wdId: string) => {
   260→    const wd = withdrawals.find((w) => w.id === wdId);
   261→    if (!wd) return;
   262→    update({ withdrawalId: wd.id, withdrawalAmount: wd.amount });
   263→  };
   264→
   265→  return (
   266→    <Page>
   267→      {/* Breadcrumb */}
   268→      <Breadcrumb>
   269→        <BreadcrumbList>
   270→          <BreadcrumbItem>
   271→            <BreadcrumbLink
   272→              asChild
   273→              className="cursor-pointer text-muted-foreground"
   274→            >
   275→              <button
   276→                type="button"
   277→                onClick={() => navigate("certificates-issued")}
   278→              >
   279→                Certificates
   280→              </button>
   281→            </BreadcrumbLink>
   282→          </BreadcrumbItem>
   283→          <BreadcrumbSeparator>
   284→            <ChevronRight className="h-3.5 w-3.5" />
   285→          </BreadcrumbSeparator>
   286→          <BreadcrumbItem>
   287→            <BreadcrumbPage className="font-mono text-xs">
   288→              {working.id}
   289→            </BreadcrumbPage>
   290→          </BreadcrumbItem>
   291→        </BreadcrumbList>
   292→      </Breadcrumb>
   293→
   294→      {/* Header row */}
   295→      <div className="flex flex-col gap-3 md:flex-row md:items-start md:justify-between">
   296→        <div className="flex items-start gap-3">
   297→          <div className="rounded-lg border bg-muted p-2">
   298→            <Award className="h-5 w-5 text-foreground" />
   299→          </div>
   300→          <div>
   301→            <h1 className="text-xl font-semibold tracking-tight text-foreground">
   302→              {working.traderName}
   303→              <span className="ml-2 text-sm font-normal text-muted-foreground">
   304→                · {working.id}
   305→              </span>
   306→            </h1>
   307→            <p className="mt-1 text-sm text-muted-foreground">
   308→              {working.challengeName} · {working.certType}
   309→            </p>
   310→          </div>
   311→        </div>
   312→        <div className="flex flex-wrap items-center gap-2">
   313→          <ExplainableStateBadge
   314→            status={working.status}
   315→            entityType="certificate"
   316→            detail={`Certificate URL: ${working.certUrl}`}
   317→          />
   318→          <Badge variant="outline" className="capitalize">
   319→            {working.certType}
   320→          </Badge>
   321→          <Button
   322→            size="sm"
   323→            variant={editing ? "default" : "outline"}
   324→            onClick={onToggleEdit}
   325→            aria-pressed={editing}
   326→          >
   327→            {editing ? (
   328→              <>
   329→                <Check className="h-3.5 w-3.5" /> Done
   330→              </>
   331→            ) : (
   332→              <>
   333→                <Pencil className="h-3.5 w-3.5" /> Edit
   334→              </>
   335→            )}
   336→          </Button>
   337→        </div>
   338→      </div>
   339→
   340→      <PageContent>
   341→        {/* Key remount on id change so local working state resets */}
   342→        <CertificateForm
   343→          key={seedCert.id}
   344→          working={working}
   345→          editing={editing}
   346→          accounts={accounts}
   347→          templates={templates}
   348→          withdrawals={withdrawals}
   349→          currency={currency}
   350→          onAccountChange={onAccountChange}
   351→          onWithdrawalChange={onWithdrawalChange}
   352→          onFieldChange={update}
   353→          onCopyUrl={onCopyUrl}
   354→        />
   355→
   356→        <Separator />
   357→
   358→        {/* Bottom action bar — one primary action + destructive w/ confirm */}
   359→        <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   360→          <AlertDialog open={deleteOpen} onOpenChange={setDeleteOpen}>
   361→            <AlertDialogTrigger asChild>
   362→              <Button
   363→                size="sm"
   364→                variant="outline"
   365→                className="text-rose-700 hover:bg-rose-50 hover:text-rose-800 dark:text-rose-400 dark:hover:bg-rose-950"
   366→              >
   367→                <Trash2 className="h-3.5 w-3.5" /> Delete Certificate
   368→              </Button>
   369→            </AlertDialogTrigger>
   370→            <AlertDialogContent>
   371→              <AlertDialogHeader>
   372→                <AlertDialogTitle className="flex items-center gap-2">
   373→                  <ShieldAlert className="h-5 w-5 text-rose-600" />
   374→                  Delete certificate?
   375→                </AlertDialogTitle>
   376→                <AlertDialogDescription>
   377→                  The certificate will be permanently deleted. The trader
   378→                  will lose access to their certificate URL.
   379→                </AlertDialogDescription>
   380→              </AlertDialogHeader>
   381→              <AlertDialogFooter>
   382→                <AlertDialogCancel>Cancel</AlertDialogCancel>
   383→                <AlertDialogAction
   384→                  className={cn(
   385→                    "bg-rose-600 text-white hover:bg-rose-700 dark:bg-rose-700 dark:hover:bg-rose-800",
   386→                  )}
   387→                  onClick={onDelete}
   388→                >
   389→                  Delete permanently
   390→                </AlertDialogAction>
   391→              </AlertDialogFooter>
   392→            </AlertDialogContent>
   393→          </AlertDialog>
   394→
   395→          <div className="flex flex-wrap items-center gap-2">
   396→            <Button size="sm" variant="outline" onClick={onSaveAndAdd}>
   397→              <Plus className="h-3.5 w-3.5" /> Save and add another
   398→            </Button>
   399→            <Button size="sm" variant="outline" onClick={onSaveAndContinue}>
   400→              <Save className="h-3.5 w-3.5" /> Save and continue editing
   401→            </Button>
   402→            <Button size="sm" onClick={onSave}>
   403→              <Save className="h-3.5 w-3.5" /> Save
   404→            </Button>
   405→          </div>
   406→        </div>
   407→      </PageContent>
   408→    </Page>
   409→  );
   410→}

   630→                value={working.certUrl}
   631→                readOnly
   632→                className="flex-1 font-mono text-xs"
   633→                aria-label="Certificate URL"
   634→              />
   635→              <Button
   636→                size="sm"
   637→                variant="outline"
   638→                className="h-8 shrink-0 px-2"
   639→                onClick={onCopyUrl}
   640→                aria-label="Copy certificate URL"
   641→              >
   642→                <Copy className="h-3.5 w-3.5" />
   643→              </Button>
   644→              <a
   645→                href={working.certUrl}
   646→                target="_blank"

   650→              >
   651→                <ExternalLink className="h-3.5 w-3.5" />

   654→          </div>
   655→
   656→          <Separator />
   657→
   658→          <div className="space-y-1.5">
   659→            <LabelWithHelp
   660→              help="Valid = active & verifiable. Expired = past its validity period. Revoked = manually invalidated by an admin; the URL stops working."
   661→              className="text-sm font-medium"
   662→            >
   663→              Status
   664→            </LabelWithHelp>
   665→            <Select
   666→              value={working.status}
   667→              onValueChange={(v) =>
   668→                onFieldChange({ status: v as CertStatus })
   669→              }
   670→              disabled={!editing}
   671→            >
   672→              <SelectTrigger className="w-full" disabled={!editing}>
   673→                <SelectValue placeholder="Select status…" />
   674→              </SelectTrigger>
   675→              <SelectContent>
   676→                <SelectItem value="valid">Valid</SelectItem>
   677→                <SelectItem value="expired">Expired</SelectItem>
   678→                <SelectItem value="revoked">Revoked</SelectItem>
   679→              </SelectContent>
   680→            </Select>
   681→            <div className="pt-1">
   682→              <ExplainableStateBadge
   683→                status={working.status}
   684→                entityType="certificate"
   685→                detail={`Certificate URL: ${working.certUrl}`}
   686→              />
   687→            </div>
   688→          </div>
   689→        </div>
   690→      </FormSection>
   691→    </div>
   692→  );
   693→}
