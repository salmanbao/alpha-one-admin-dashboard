    33→function FilterBar({
    34→  label,
    35→  activeCount,
    36→  onClear,
    37→  resultCount,
    38→  totalCount,
    39→  children,
    40→}: {
    41→  label: string;
    42→  activeCount: number;
    43→  onClear: () => void;
    44→  resultCount: number;
    45→  totalCount: number;
    46→  children: React.ReactNode;
    47→}) {
    48→  return (
    49→    <div className="flex flex-col gap-3 rounded-lg border bg-card p-3 lg:flex-row lg:items-center lg:justify-between">
    50→      <div className="flex flex-wrap items-center gap-2">
    51→        <span className="inline-flex items-center gap-1.5 text-xs font-medium text-muted-foreground">
    52→          <Filter className="h-3.5 w-3.5" aria-hidden />
    53→          <span>{label}</span>
    54→          {activeCount > 0 ? (
    55→            <Badge variant="secondary" className="text-[9px]">
    56→              {activeCount}
    57→            </Badge>
    58→          ) : null}
    59→        </span>
    60→        {children}
    61→        {activeCount > 0 ? (
    62→          <Button
    63→            size="sm"
    64→            variant="ghost"
    65→            onClick={onClear}
    66→            className="h-8 gap-1 px-2 text-xs"
    67→          >
    68→            <X className="h-3 w-3" /> Clear all
    69→          </Button>
    70→        ) : null}
    71→      </div>
    72→      <span className="text-xs text-muted-foreground" aria-live="polite">
    73→        {resultCount} of {totalCount} shown
    74→      </span>
    75→    </div>
    76→  );
    77→}

    79→function FilterSelect<T extends string>({
    80→  value,
    81→  onChange,
    82→  placeholder,
    83→  options,
    84→  width = "w-[140px]",
    85→}: {
    86→  value: string;
    87→  onChange: (v: T) => void;
    88→  placeholder: string;
    89→  options: { value: string; label: string }[];
    90→  width?: string;
    91→}) {
    92→  return (
    93→    <Select value={value} onValueChange={(v) => onChange(v as T)}>
    94→      <SelectTrigger className={`h-8 ${width}`}>
    95→        <SelectValue placeholder={placeholder} />
    96→      </SelectTrigger>
    97→      <SelectContent>
    98→        {options.map((opt) => (
    99→          <SelectItem key={opt.value} value={opt.value}>
   100→            {opt.label}
   101→          </SelectItem>
   102→        ))}
   103→      </SelectContent>
   104→    </Select>
   105→  );
