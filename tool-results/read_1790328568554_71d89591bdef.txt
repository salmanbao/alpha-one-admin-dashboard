    54→function relativeTime(iso: string): string {
    55→  const diff = Date.now() - new Date(iso).getTime();
    56→  const hours = Math.floor(diff / (1000 * 60 * 60));
    57→  if (hours < 1) return "just now";
    58→  if (hours < 24) return `${hours}h ago`;
    59→  const days = Math.floor(hours / 24);
    60→  return `${days}d ago`;
    61→}
    62→
    63→function avgResponseHours(tickets: SupportTicket[]): number {
    64→  const withReplies = tickets.filter((t) => t.lastReplyAt);
    65→  if (!withReplies.length) return 0;
    66→  let total = 0;
    67→  for (const t of withReplies) {
    68→    const created = new Date(t.createdAt).getTime();
    69→    const replied = new Date(t.lastReplyAt!).getTime();
    70→    total += Math.max(1, (replied - created) / (1000 * 60 * 60));
    71→  }
    72→  return Math.round(total / withReplies.length);
    73→}

    75→/**
    76→ * Derive SLA target (in hours) from priority — used by the Ticket Detail
    77→ * Sheet KPI strip. The mock SupportTicket has no `slaHours` field, so we
    78→ * project from the existing `priority` field deterministically.
    79→ */
    80→function slaHoursFor(priority: SupportTicket["priority"]): number {
    81→  switch (priority) {
    82→    case "urgent": return 4;
    83→    case "high": return 8;
    84→    case "medium": return 24;
    85→    case "low": return 48;
    86→  }
    87→}

   101→const TRADER_OPENERS = [
   102→  "Hi team, I'm seeing the issue described above and would appreciate a quick look.",
   103→  "Following up — this is still blocking my trading session today.",
   104→  "Any update on this? I'd like to get back to the challenge before the daily reset.",
   105→  "Thanks for the reply. I've attached the screenshot showing the error.",
   106→  "Could you also confirm whether the payout window re-opens once this is resolved?",
   107→];
   108→
   109→const AGENT_OPENERS = [
   110→  "Thanks for reaching out — I've pulled up your account and I'm investigating now.",
   111→  "I can confirm this is a known issue on our side; engineering is rolling out a fix this afternoon.",
   112→  "Could you share the exact timestamp when this last occurred? I'll cross-reference the logs.",
   113→  "Good news — the rule engine has been recalibrated; please refresh your dashboard in 5 minutes.",
   114→  "I've escalated this to Tier 2 support. You should hear back within the SLA window.",
   115→];

   117→function conversationFor(t: SupportTicket): ConversationMessage[] {
   118→  const seed = hashStr(t.id);
   119→  // Build 3 deterministic messages anchored on createdAt; spacing of 2h, 5h.
   120→  const created = new Date(t.createdAt).getTime();
   121→  const mkTs = (hoursAgo: number) => new Date(created + hoursAgo * 3_600_000).toISOString();
   122→  const traderIdx = seed % TRADER_OPENERS.length;
   123→  const agentIdx = (seed >>> 3) % AGENT_OPENERS.length;
   124→  const traderInitials = t.traderName
   125→    .split(" ")
   126→    .map((p) => p[0])
   127→    .slice(0, 2)
   128→    .join("")
   129→    .toUpperCase();
   130→  return [
   131→    {
   132→      author: t.traderName,
   133→      authorInitials: traderInitials,
   134→      body: TRADER_OPENERS[traderIdx],
   135→      timestamp: mkTs(0),
   136→      fromTrader: true,
   137→    },
   138→    {
   139→      author: t.assignee ?? "Support Agent",
   140→      authorInitials: (t.assignee ?? "SA")
   141→        .split(" ")
   142→        .map((p) => p[0])
   143→        .slice(0, 2)
   144→        .join("")
   145→        .toUpperCase(),
   146→      body: AGENT_OPENERS[agentIdx],
   147→      timestamp: mkTs(2),
   148→      fromTrader: false,
   149→    },
   150→    {
   151→      author: t.traderName,
   152→      authorInitials: traderInitials,
   153→      body: TRADER_OPENERS[(traderIdx + 2) % TRADER_OPENERS.length],
   154→      timestamp: mkTs(5),
   155→      fromTrader: true,
   156→    },
   157→  ];
   158→}
   159→
   160→interface InternalNote {
   161→  author: string;
   162→  body: string;
   163→  timestamp: string;
   164→}
   165→
   166→function internalNotesFor(t: SupportTicket): InternalNote[] {
   167→  const seed = hashStr(t.id);
   168→  const created = new Date(t.createdAt).getTime();
   169→  const mkTs = (h: number) => new Date(created + h * 3_600_000).toISOString();
   170→  const authors = ["Sarah K.", "Marcus T.", "Elena R."];
   171→  const notes = [
   172→    "Triaged to billing tier — confirmed the trader is on the 100k challenge plan.",
   173→    "Cross-referenced with the trading log; the rule-engine recalc is plausible.",
   174→  ];
   175→  return notes.map((body, i) => ({
   176→    author: authors[(seed >>> (i * 2)) % authors.length],
   177→    body,
   178→    timestamp: mkTs(i + 1),
   179→  }));
   180→}

   186→export function SupportOverviewPage() {

   205→      p === "high" ? "#ea580c" :
   206→      p === "medium" ? "#0ea5e9" : "#94a3b8",
   207→  }));

   209→    <Page>
   210→      <PageHeader
   211→        title="Support"
   212→        description={`Ticketing and ${term("trader").toLowerCase()} assistance across this tenant.`}
   213→        icon={LifeBuoy}

   235→                { key: "status", header: "Status", cell: (t) => <StatusBadge tone={ticketStatusTone(t.status)}>{t.status}</StatusBadge> },

   238→              data={recent}
   239→              rowKey={(t) => t.id}
   240→              pageSize={6}

   242→          </div>
