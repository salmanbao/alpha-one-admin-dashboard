   210→/** Deterministic pseudo-random generator from a numeric seed. */
   211→function seededRandom(seed: number): number {
   212→  const x = Math.sin(seed * 9999.1) * 10000;
   213→  return x - Math.floor(x);
   214→}
   215→
   216→/** Hash a string id to a stable integer seed. */
   217→function hashSeed(id: string): number {
   218→  let h = 0;
   219→  for (let i = 0; i < id.length; i++) {
   220→    h = (h * 31 + id.charCodeAt(i)) | 0;
   221→  }
   222→  return Math.abs(h) || 1;
   223→}
   224→
   225→const BROKERS = ["MT5", "MT4", "DXTrade"];
   226→const ACCOUNT_STATUSES: LinkedAccount["status"][] = [
   227→  "active",
   228→  "passed",
   229→  "pending",
   230→  "breached",
   231→];
   232→
   233→function generateOrderDetail(id: string, tenantId: string): OrderDetail {
   234→  const seed = hashSeed(id);
   235→  const traders = getTenantTraders(tenantId);
   236→  const trader = traders[seed % Math.max(traders.length, 1)] ?? traders[0];
   237→
   238→  const r1 = seededRandom(seed + 0.1);
   239→  const r2 = seededRandom(seed + 0.5);
   240→  const r3 = seededRandom(seed + 1.7);
   241→  const r4 = seededRandom(seed + 3.3);
   242→
   243→  const orderType = ORDER_TYPES[seed % ORDER_TYPES.length];
   244→  const challenge = CHALLENGES[seed % CHALLENGES.length];
   245→  const competition = r4 > 0.7 ? COMPETITIONS[seed % COMPETITIONS.length] : null;
   246→  const accountBalance = [10000, 25000, 50000, 100000, 200000][
   247→    seed % 5
   248→  ];
   249→  const amountPaid =
   250→    orderType === "Refund"
   251→      ? -Math.round(accountBalance * 0.15 * 100) / 100
   252→      : Math.round(accountBalance * (0.05 + r1 * 0.03) * 100) / 100;
   253→  const quantity = 1 + (seed % 3);
   254→  const paymentMethod = PAYMENT_METHODS[seed % PAYMENT_METHODS.length];
   255→  const couponCode = r2 > 0.5 ? `SAVE${10 + (seed % 30)}` : "";
   256→  const bundleId = r3 > 0.6 ? `bundle-${(seed % 100) + 1}` : null;
   257→  const customerIp = `${10 + (seed % 200)}.${seed % 255}.${(seed * 7) % 255}.${(seed * 13) % 255}`;
   258→
   259→  // Addons — deterministic small object
   260→  const addons: Record<string, unknown> = {};
   261→  if (r4 > 0.4) addons["priority_withdrawal"] = true;
   262→  if (r4 > 0.6) addons["extended_kyc"] = false;
   263→  if (r4 > 0.8) addons["addon_quantity"] = 1 + (seed % 3);
   264→  if (Object.keys(addons).length === 0) addons["none"] = true;
   265→
   266→  // Date created — deterministic window from the past 60 days
   267→  const now = Date.now();
   268→  const createdMsAgo = Math.floor((1 + r2 * 60 * 24 * 60)) * 60 * 1000;
   269→  const dateCreated = new Date(now - createdMsAgo).toISOString();
   270→
   271→  // Attribution — deterministic UTM values
   272→  const utm = {
   273→    source: ["twitter", "google", "affiliate", "telegram", "direct"][seed % 5],
   274→    campaign:
   275→      challenge.toLowerCase().replace(/\s+/g, "_") + `_${(seed % 12) + 1}`,
   276→    medium: ["cpc", "social", "email", "referral"][seed % 4],
   277→    term: ["forex", "funded", "evaluation", ""][seed % 4],
   278→    content: `ad_${(seed % 6) + 1}`,
   279→    referrer: REFERRERS[seed % REFERRERS.length],
   280→    landing: LANDING_PAGES[seed % LANDING_PAGES.length],
   281→  };
   282→
   283→  // Linked accounts — 1..3 deterministic accounts
   284→  const accountCount = 1 + (seed % 3);
   285→  const accounts: LinkedAccount[] = Array.from(
   286→    { length: accountCount },
   287→    (_, i) => {
   288→      const s = seed + i * 17;
   289→      return {
   290→        id: `acct-${s}`,
   291→        login: `${100000 + (s % 90000)}`,
   292→        phase: ["phase-1", "phase-2", "funded"][i % 3],
   293→        broker: BROKERS[i % BROKERS.length],
   294→        initialBalance: accountBalance,
   295→        status: ACCOUNT_STATUSES[s % ACCOUNT_STATUSES.length],
   296→      };
   297→    },
   298→  );
   299→
   300→  // Subscription rows — only meaningful for Subscription type but always
   301→  // generated so the accordion is never empty.
   302→  const subCount = orderType === "Subscription" ? 1 + (seed % 3) : 1;
   303→  const subscriptions: SubscriptionRow[] = Array.from(
   304→    { length: subCount },
   305→    (_, i) => {
   306→      const s = seed + i * 23;
   307→      const billingOffset = (s % 90) * 24 * 60 * 60 * 1000;
   308→      const status: SubscriptionStatus =
   309→        i === 0
   310→          ? "Active"
   311→          : (["Active", "Cancelled", "Expired"][s % 3] as SubscriptionStatus);
   312→      const next =
   313→        status === "Active"
   314→          ? new Date(now + (30 - (s % 30)) * 24 * 60 * 60 * 1000).toISOString()
   315→          : null;
   316→      return {
   317→        id: `sub-${s}`,
   318→        billingDate: new Date(now - billingOffset).toISOString(),
   319→        amount: Math.round(accountBalance * 0.05 * 100) / 100,
   320→        status,
   321→        nextBilling: next,
   322→      };
   323→    },
   324→  );
   325→
   326→  return {
   327→    id,
   328→    userId: trader?.id ?? "usr-unknown",
   329→    userEmail: trader?.email ?? "trader@example.com",
   330→    userFullName: trader?.name ?? "Unknown trader",
   331→    addons,
   332→    dateCreated,
   333→    orderType,
   334→    notes:
   335→      orderType === "Refund"
   336→        ? "Refund issued for duplicate challenge purchase (CS ticket #4231)."
   337→        : `Customer purchased ${quantity} × ${challenge}.`,
   338→    challenge,
   339→    competition,
   340→    accountBalance,
   341→    amountPaid,
   342→    quantity,
   343→    paymentMethod,
   344→    couponCode,
   345→    bundleId,
   346→    customerIp,
   347→    utm,
   348→    accounts,
   349→    subscriptions,
   350→  };
   351→}

   928→      sortValue: (a) => a.initialBalance,
   929→      numeric: true,
   930→    },
   931→    {
   932→      key: "status",
   933→      header: "Status",
   934→      cell: (a) => (
   935→        <StatusBadge
   936→          tone={
   937→            a.status === "active"
   938→              ? "success"
   939→              : a.status === "breached"
   940→              ? "danger"
   941→              : a.status === "passed"
   942→              ? "info"
   943→              : "warning"

   945→        >
