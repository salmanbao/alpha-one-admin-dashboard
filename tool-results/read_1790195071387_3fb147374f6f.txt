  6094→import { makeTermResolver, plural } from "@/lib/platform/terminology";
  6095→import { usePlatform } from "@/lib/platform/platform-context";
  6096→
  6097→function Page() {

  6102→}

  6105→`makeTermResolver(tenant)` returns a `(key: TermKey) => string` — it has NO plural option, so the `plural()` helper from the same module is composed in for plural labels (e.g., `plural(term("trader"))` → "Traders" / "Participants" / "Candidates").
