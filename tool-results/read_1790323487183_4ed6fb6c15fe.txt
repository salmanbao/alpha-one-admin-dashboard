   205→/** Convert ISO-2 country code (e.g. "US") to a flag emoji (🇺🇸). */
   206→function countryFlagEmoji(country: string): string {
   207→  if (!country || country.length !== 2) return "🏳️";
   208→  const cc = country.toUpperCase();
   209→  const A = 0x1f1e6;
   210→  const base = "A".charCodeAt(0);
   211→  return String.fromCodePoint(
   212→    A + (cc.charCodeAt(0) - base),
   213→    A + (cc.charCodeAt(1) - base),
   214→  );
   215→}

   217→/**
   218→ * Deterministic profit-factor proxy.
   219→ *
   220→ * Real PF = gross profit / gross loss. The mock only exposes signed PnL,
   221→ * so we synthesize a stable per-trader value in [0.6, 2.6] using the
   222→ * trader's `winRate` (acts as the seed) plus a hashStr jitter so two
   223→ * traders with the same winRate still differ — mirroring real variance
   224→ * in risk-reward ratios.
   225→ */
   226→function deriveProfitFactor(trader: Trader): number {
   227→  const h = hashStr(trader.id);
   228→  const wr = trader.winRate / 100;
   229→  const jitter = ((h % 1000) / 1000 - 0.5) * 0.45; // ±0.225
   230→  const pf = 0.6 + wr * 1.8 + jitter; // baseline 0.6 → up to ~2.65
   231→  return Math.max(0.5, Math.round(pf * 100) / 100);
   232→}
   233→
   234→/**
   235→ * Deterministic 30-day PnL for a trader — anchors the leaderboard.
   236→ *
   237→ * Uses the trader's signed totalPnl plus a hashStr-seeded slice so the
   238→ * leaderboard column is stable across reloads and looks distinct from
   239→ * the cumulative equity number.
   240→ */
   241→function derive30dPnl(trader: Trader): number {
   242→  const h = hashStr(trader.id + "30d");
   243→  const slice = 0.25 + ((h % 100) / 100) * 0.4; // 0.25–0.65
   244→  return Math.round(trader.totalPnl * slice);
   245→}

   247→/** Bucket a trader into an activity tier by trade count. */

  1081→  const totalEquity = accounts.reduce((s, a) => s + a.equity, 0);
  1082→  const var95 = Math.round(totalEquity * 0.018 + 8000 + (hashStr(tid + "var95") % 4000));
  1083→  const es = Math.round(var95 * 1.39 + 1500);
  1084→  const maxDrawdownPct = Math.round((12 + ((hashStr(tid + "dd") % 100) / 100) * 8) * 10) / 10; // 12–20%
  1085→  const sharpe = Math.round((1.0 + ((hashStr(tid + "sharpe") % 100) / 100) * 1.0) * 100) / 100; // 1.0–2.0
  1086→

  1088→  const ddBuckets = [
  1089→    { bucket: "0–5%", lo: 0, hi: 5 },

  1097→    const seed = hashStr(tid + "dd" + b.bucket);

  1099→    return { bucket: b.bucket, accounts: base };
  1100→  });
  1101→
  1102→  // ---- VaR confidence curve ---------------------------------------------
  1103→  const varCurve = [
  1104→    { level: "90%", value: Math.round(var95 * 0.82) },

  1107→    { level: "99%", value: Math.round(var95 * 1.35) },

  1110→

  1112→  const challengeSharpe = [
  1113→    { type: "1-Step", sharpe: Math.round((0.8 + (hashStr(tid + "s1") % 100) / 100 * 0.8) * 100) / 100 },

  1115→    { type: "3-Step", sharpe: Math.round((1.2 + (hashStr(tid + "s3") % 100) / 100 * 0.9) * 100) / 100 },
  1116→    { type: "Funded", sharpe: Math.round((1.4 + (hashStr(tid + "s4") % 100) / 100 * 0.9) * 100) / 100 },
  1117→  ];
  1118→
  1119→  // ---- Top 10 highest risk accounts --------------------------------------
  1120→  // Risk score = drawdown % * 0.6 + |open PnL| / equity * 0.4 (mock).
  1121→  const riskAccounts = accounts
  1122→    .map((a) => {
  1123→      const h = hashStr(a.id);
  1124→      const drawdownPct = Math.round((4 + (h % 180) / 10) * 10) / 10; // 4–22%
  1125→      const openPnl = Math.round((Math.sin(h) * a.equity * 0.08));
  1126→      const riskScore = Math.round((drawdownPct * 0.6 + Math.abs(openPnl) / Math.max(1, a.equity) * 100 * 0.4) * 10) / 10;
  1127→      const trader = traderById.get(a.traderId);
  1128→      return { account: a, trader, drawdownPct, openPnl, riskScore };
  1129→    })
  1130→    .sort((a, b) => b.riskScore - a.riskScore)
  1131→    .slice(0, 10);
  1132→
  1133→  const riskCols: Column<(typeof riskAccounts)[number]>[] = [
  1134→    {
  1135→      key: "account",
  1136→      header: "Account",
  1137→      cell: (r) => (
  1138→        <div className="flex flex-col">
  1139→          <span className="font-medium text-foreground">{r.account.login}</span>
  1140→          <span className="text-[10px] text-muted-foreground">{r.account.platform} · {r.account.phase}</span>
  1141→        </div>
  1142→      ),
  1143→      sortValue: (r) => r.account.login,
  1144→    },
  1145→    {
  1146→      key: "trader",
  1147→      header: term("trader"),
  1148→      cell: (r) => (
  1149→        <span className="inline-flex items-center gap-2">
  1150→          {r.trader ? <span aria-hidden>{countryFlagEmoji(r.trader.country)}</span> : null}

  1152→        </span>
  1153→      ),
  1154→      sortValue: (r) => r.account.traderName,
  1155→    },
  1156→    {
  1157→      key: "equity",
  1158→      header: "Equity",

  1190→        return <span className={`font-semibold ${tone}`}>{r.riskScore.toFixed(1)}</span>;
