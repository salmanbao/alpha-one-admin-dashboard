   148→function hashStr(s: string): number {
   149→  let h = 0;
   150→  for (let i = 0; i < s.length; i++) {
   151→    h = (h * 31 + s.charCodeAt(i)) >>> 0;
   152→  }
   153→  return h;
   154→}

   156→function toLocalInputValue(iso: string): string {
   157→  if (!iso) return "";
   158→  // datetime-local expects YYYY-MM-DDTHH:mm
   159→  return iso.slice(0, 16);
   160→}

   162→function deriveAccountConfig(account: TradingAccount): AccountConfig {
   163→  const seed = hashStr(account.id) || 1;
   164→  const userEmail = `${account.traderName.toLowerCase().replace(/\s+/g, ".")}@example.com`;
   165→  const phase = account.phase === "phase-1" ? "Phase 1" : account.phase === "phase-2" ? "Phase 2" : "Funded";
   166→
   167→  const startDaysAgo = (seed % 60) + 5;
   168→  const startDateIso = new Date(Date.now() - startDaysAgo * 24 * 60 * 60 * 1000).toISOString();
   169→  // 60% of accounts have end date, 40% leave empty
   170→  const hasEnd = seed % 5 !== 0;
   171→  const endDays = (seed % 30) + 30;
   172→  const endDateIso = hasEnd
   173→    ? new Date(Date.now() + endDays * 24 * 60 * 60 * 1000).toISOString()
   174→    : "";
   175→
   176→  const profitSplit = 60 + (seed % 21); // 60..80
   177→  const payoutFrequency = PAYOUT_FREQUENCIES[seed % PAYOUT_FREQUENCIES.length];
   178→  const firstWithdrawalDelay = `${7 + (seed % 21)} days`;
   179→  // Order id pattern
   180→  const orderNum = 10000 + (seed % 9999);
   181→  const order = `ORD-${orderNum}`;
   182→  const nextWithdrawalDateIso = new Date(Date.now() + (seed % 14 + 3) * 24 * 60 * 60 * 1000).toISOString();
   183→  const source = SOURCE_OPTIONS[seed % SOURCE_OPTIONS.length];
   184→  const accountLabel = ACCOUNT_LABELS[seed % ACCOUNT_LABELS.length];
   185→
   186→  // Balance & drawdown
   187→  const initialBalance = account.balance;
   188→  const liveBalance = account.balance + Math.round((account.equity - account.balance));
   189→  const liveEquity = account.equity;
   190→  const profitLoss = liveEquity - initialBalance;
   191→  const dailyStartingBalance = initialBalance + Math.round(profitLoss * 0.4);
   192→  const dailyDrawdownAmount = Math.round(initialBalance * 0.03 + (seed % 200));
   193→  const dailyDrawdownPct = Math.round((dailyDrawdownAmount / initialBalance) * 1000) / 10;
   194→  const profitTargetAmount = Math.round(initialBalance * (0.08 + (seed % 5) * 0.005));
   195→  const globalDrawdownAmount = Math.round(initialBalance * 0.05 + (seed % 350));
   196→  const globalDrawdownPct = Math.round((globalDrawdownAmount / initialBalance) * 1000) / 10;
   197→  const expiryMs = Date.now() - ((seed % 8) + 1) * 60 * 60 * 1000;
   198→  const dailyDrawdownExpiry = new Date(expiryMs).toISOString();
   199→  const drawdownLocked = seed % 7 === 0;
   200→
   201→  // Broker details
   202→  const login = account.login;
   203→  const brokerType = BROKER_TYPES[seed % BROKER_TYPES.length];
   204→  const mtTradingAccount = `${account.login}@mt5-acmebroker`;
   205→  const matchTraderTradingAccount = `mt_${account.login}@matchtrader`;
   206→
   207→  // Account status details
   208→  const status = account.status;
   209→  const statusReason = STATUS_REASONS[seed % STATUS_REASONS.length];
   210→  const finalisedMs = Date.now() - ((seed % 24) + 1) * 60 * 60 * 1000;
   211→  const statusFinalisedAt = new Date(finalisedMs).toISOString();
   212→  const customStatusReason = seed % 3 === 0 ? "Trader requested manual review during KYC." : "";
   213→  const copyTradingDetected = seed % 11 === 0;
   214→  const failedReviewReason = seed % 9 === 0 ? "Risk team flagged for inverse-trading pattern with account 530012." : "";
   215→
   216→  // Extra settings
   217→  const hideAccount = seed % 5 === 0;
   218→  const publicTrackRecord = seed % 3 !== 0;
   219→  const publicBalance = seed % 4 !== 0;
   220→  const publicTradeHistory = seed % 6 !== 0;
   221→  const publicLots = seed % 7 !== 0;
   222→
   223→  return {
   224→    userEmail,
   225→    phase,
   226→    startDate: startDateIso,
   227→    endDate: endDateIso,
   228→    profitSplit,
   229→    payoutFrequency,
   230→    firstWithdrawalDelay,
   231→    order,
   232→    nextWithdrawalDate: nextWithdrawalDateIso,
   233→    source,
   234→    accountLabel,
   235→    initialBalance,
   236→    liveBalance,
   237→    liveEquity,
   238→    profitLoss,
   239→    dailyStartingBalance,
   240→    dailyDrawdownAmount,
   241→    dailyDrawdownPct,
   242→    profitTargetAmount,
   243→    globalDrawdownAmount,
   244→    globalDrawdownPct,
   245→    dailyDrawdownExpiry,
   246→    drawdownLocked,
   247→    login,
   248→    brokerType,
   249→    mtTradingAccount,
   250→    matchTraderTradingAccount,
   251→    status,
   252→    statusReason,
   253→    statusFinalisedAt,
   254→    customStatusReason,
   255→    copyTradingDetected,
   256→    failedReviewReason,
   257→    hideAccount,
   258→    publicTrackRecord,
   259→    publicBalance,
   260→    publicTradeHistory,
   261→    publicLots,
   262→  };
   263→}

   362→export function AccountConfigurationPage() {
   363→  const { runtime, router, navigate } = usePlatform();
   364→  const tid = runtime.tenant?.id ?? "platform";
   365→  const accountId = router.params.id;
   366→  const currency = runtime.tenant?.currency ?? "USD";
   367→
   368→  const account = useMemo(
   369→    () => getTenantAccounts(tid).find((a) => a.id === accountId),
   370→    [tid, accountId],
   371→  );
   372→
   373→  const derived = useMemo(
   374→    () => (account ? deriveAccountConfig(account) : null),
   375→    [account],
   376→  );
   377→
   378→  // Order dropdown options (mock with deterministic IDs)
   379→  const orderOptions = useMemo(() => {
   380→    if (!account) return [];
   381→    const seedBase = hashStr(account.id);
   382→    return Array.from({ length: 6 }, (_, i) => {
   383→      const num = 10000 + ((seedBase + i * 137) % 9999);
   384→      return `ORD-${num}`;
   385→    });
   386→  }, [account]);
   387→
   388→  // Local editable draft state — mirrors derived on first paint
   389→  const [draft, setDraft] = useState<AccountConfig | null>(null);
   390→  const draftReady = draft ?? derived;
   391→  const update = <K extends keyof AccountConfig>(key: K, value: AccountConfig[K]) =>
   392→    setDraft((d) => ({ ...(d ?? derived!), [key]: value }));
   393→
   394→  if (!account || !derived || !draftReady) {
   395→    return (
   396→      <Page>
   397→        <Button variant="ghost" size="sm" onClick={() => navigate("trading-accounts")} className="w-fit">
   398→          <ArrowLeft className="mr-1 h-4 w-4" /> Back
   399→        </Button>
   400→        <p className="text-muted-foreground">Account not found.</p>
   401→      </Page>
   402→    );
   403→  }
   404→
   405→  // Days remaining computation
   406→  const daysRemaining = derived.endDate
   407→    ? Math.max(0, Math.ceil((new Date(derived.endDate).getTime() - Date.now()) / (24 * 60 * 60 * 1000)))
   408→    : Infinity;
   409→
   410→  const drawdownPct = derived.globalDrawdownPct;
   411→
   412→  // Toast handlers
   413→  const handleSave = () =>
   414→    toast({
   415→      title: "Account configuration saved",
   416→      description: `Login ${account.login} configuration committed.`,
   417→    });
   418→  const handleSaveContinue = () =>
   419→    toast({
   420→      title: "Saved — keep editing",
   421→      description: "Changes saved; staying on this screen.",
   422→    });
   423→  const handleSync = () =>
   424→    toast({
   425→      title: "Syncing with broker platform",
   426→      description: "Pulling live balance, equity and positions from the bridge.",
   427→    });
   428→  const handleResend = () =>
   429→    toast({
   430→      title: "Credentials resent to trader",
   431→      description: `Login link + temp password sent to ${derived.userEmail}.`,
   432→    });
   433→  const handleReset = () =>
   434→    toast({
   435→      title: "Account reset to initial state",
   436→      description: "All progress lost. Balance restored to initial capital.",
   437→      variant: "destructive",
   438→    });
   439→  const handleBlock = () =>
   440→    toast({
   441→      title: "Account blocked",
   442→      description: `Login ${account.login} can no longer trade. Logged in audit trail.`,
   443→      variant: "destructive",
   444→    });
   445→
   446→  return (
   447→    <Page>
   448→      <Button
   449→        variant="ghost"
   450→        size="sm"
   451→        className="w-fit"
   452→        onClick={() => navigate("trader-detail", { id: account.traderId })}

   454→        <ArrowLeft className="mr-1 h-4 w-4" /> Back to Account

   458→        title="Account Configuration"
   459→        description="Comprehensive configuration, balances, drawdown, broker details, status and extra settings."
   460→        icon={Settings2}
   461→      />
   462→
   463→      {/* Entity header — concise identity */}
   464→      <EntityHeader
   465→        title={`Login ${account.login}`}
   466→        subtitle={`${account.traderName} · ${account.platform} · ${derived.phase}`}
   467→        badges={
   468→          <>
   469→            <Badge variant="outline" className="text-[10px]">{account.type}</Badge>
   470→            <ExplainableStateBadge status={account.status} entityType="account" />
   471→            <StatusBadge
   472→              tone={
   473→                derived.source === "Webhook" ? "info" : "muted"
   474→              }
   475→            >
   476→              {derived.source}

   478→            <StatusBadge tone="default">{derived.accountLabel}</StatusBadge>
