    17→    17→ * Smart defaults (spec §15): selecting a challenge type auto-fills the

    19→    19→ * blank forms (spec §16).

    21→    21→ * Step indicator pattern mirrors OnboardingWizard (done/active/inactive
    22→    22→ * circles + connecting progress bars).
    23→    23→ *
    24→    24→ * All form state is local useState. No persistence — on Create, we show

   164→   164→
   165→   165→export function ChallengeWizardPage() {
   166→   166→  const [state, setState] = useState<WizardState>(INITIAL_STATE);
   167→   167→  const [step, setStep] = useState(0);
   168→   168→
   169→   169→  const challengeTypes = getChallengeTypes();
   170→   170→  const selectedType = useMemo(
   171→   171→    () => challengeTypes.find((t) => t.id === state.challengeTypeId) ?? null,
   172→   172→    [challengeTypes, state.challengeTypeId],
   173→   173→  );
   174→   174→  const hasPhase2 = (selectedType?.phases ?? 0) >= 2;
   175→   175→
   176→   176→  /* ---- Smart defaults: select type → fill phases ---- */
   177→   177→  const selectChallengeType = (typeId: string) => {
   178→   178→    const configs = getChallengePhaseConfigs(typeId);
   179→   179→    // Find first non-funded phase as Phase 1, second non-funded as Phase 2
   180→   180→    const evaluationPhases = configs.filter((p) => !p.isFunded);
   181→   181→    const phase1Cfg = evaluationPhases[0] ?? configs[0];
   182→   182→    const phase2Cfg = evaluationPhases[1] ?? configs[1];
   183→   183→    setState((s) => ({
   184→   184→      ...s,
   185→   185→      challengeTypeId: typeId,
   186→   186→      phase1: phase1Cfg ? phaseConfigFromTemplate(phase1Cfg) : { ...EMPTY_PHASE },
   187→   187→      phase2: phase2Cfg ? phaseConfigFromTemplate(phase2Cfg) : { ...EMPTY_PHASE },
   188→   188→    }));
   189→   189→  };
   190→   190→
   191→   191→  /* ---- Validation per step ---- */
   192→   192→  const stepValid = (i: number): boolean => {
   193→   193→    switch (i) {
   194→   194→      case 0:
   195→   195→        return state.challengeTypeId !== null;
   196→   196→      case 1:
   197→   197→        return phaseValid(state.phase1);
   198→   198→      case 2:
   199→   199→        return !hasPhase2 || phaseValid(state.phase2);
   200→   200→      case 3:
   201→   201→      case 4:
   202→   202→        return true;
   203→   203→      case 5:
   204→   204→        return (
   205→   205→          Number(state.riskRules.maxDailyLossPct) > 0 &&
   206→   206→          Number(state.riskRules.maxOverallLossPct) > 0
   207→   207→        );
   208→   208→      case 6:
   209→   209→        return true;
   210→   210→      default:
   211→   211→        return false;
   212→   212→    }
   213→   213→  };
   214→   214→
   215→   215→  const canNext = stepValid(step) && step < STEPS.length - 1;
   216→   216→  const canBack = step > 0;
   217→   217→
   218→   218→  const next = () => setStep((s) => Math.min(STEPS.length - 1, s + 1));
   219→   219→  const back = () => setStep((s) => Math.max(0, s - 1));
   220→   220→
   221→   221→  const createChallenge = () => {
   222→   222→    toast({
   223→   223→      title: "Challenge created",
   224→   224→      description: `${selectedType?.name ?? "Challenge"} template saved and ready to assign to traders.`,
   225→   225→    });
   226→   226→    setState(INITIAL_STATE);
   227→   227→    setStep(0);
   228→   228→  };
   229→   229→
   230→   230→  /* ---- Step content renderer ---- */
   231→   231→  const renderStep = () => {
   232→   232→    switch (step) {
   233→   233→      case 0:
   234→   234→        return (
   235→   235→          <StepType
   236→   236→            types={challengeTypes}
   237→   237→            selectedId={state.challengeTypeId}
   238→   238→            onSelect={selectChallengeType}
   239→   239→          />
   240→   240→        );
   241→   241→      case 1:
   242→   242→        return (
   243→   243→          <PhaseForm
   244→   244→            title="Phase 1 — Evaluation"
   245→   245→            description="Initial evaluation phase. Traders must reach the profit target while staying within risk limits."
   246→   246→            value={state.phase1}
   247→   247→            onChange={(next) => setState((s) => ({ ...s, phase1: next }))}
   248→   248→          />
   249→   249→        );
   250→   250→      case 2:
   251→   251→        if (!hasPhase2) {
   252→   252→          return (
   253→   253→            <NoticeCard
   254→   254→              title="Phase 2 not applicable"
   255→   255→              description={`"${selectedType?.name ?? "This challenge"}" only has one phase. You can skip this step and continue to Trading Rules.`}
   256→   256→            />
   257→   257→          );
   258→   258→        }
   259→   259→        return (
   260→   260→          <PhaseForm
   261→   261→            title="Phase 2 — Verification"
   262→   262→            description="Second evaluation phase. Typically a lower profit target with the same risk limits as Phase 1."
   263→   263→            value={state.phase2}
   264→   264→            onChange={(next) => setState((s) => ({ ...s, phase2: next }))}
   265→   265→          />
   266→   266→        );
   267→   267→      case 3:
   268→   268→        return (
   269→   269→          <TradingRulesForm
   270→   270→            value={state.tradingRules}
   271→   271→            onChange={(next) => setState((s) => ({ ...s, tradingRules: next }))}
   272→   272→          />
   273→   273→        );
   274→   274→      case 4:
   275→   275→        return (
   276→   276→          <PayoutRulesForm
   277→   277→            value={state.payoutRules}
   278→   278→            onChange={(next) => setState((s) => ({ ...s, payoutRules: next }))}
   279→   279→          />
   280→   280→        );
   281→   281→      case 5:
   282→   282→        return (
   283→   283→          <RiskRulesForm
   284→   284→            value={state.riskRules}
   285→   285→            onChange={(next) => setState((s) => ({ ...s, riskRules: next }))}
   286→   286→          />
   287→   287→        );
   288→   288→      case 6:
   289→   289→        return (
   290→   290→          <ReviewStep
   291→   291→            state={state}
   292→   292→            selectedType={selectedType}
   293→   293→            hasPhase2={hasPhase2}
   294→   294→            onCreate={createChallenge}
   295→   295→          />
   296→   296→        );
   297→   297→      default:
   298→   298→        return null;
   299→   299→    }
   300→   300→  };
   301→   301→
   302→   302→  return (
   303→   303→    <Page>
   304→   304→      <PageHeader
   305→   305→        title="Create Challenge"
   306→   306→        description="Guided wizard to configure a new evaluation challenge template."
   307→   307→        icon={Sparkles}
   308→   308→      />
   309→   309→      <PageContent>
   310→   310→        {/* Step indicator — done / active / inactive + connecting bars (mirrors OnboardingWizard) */}
   311→   311→        <div className="rounded-lg border bg-card p-4">
   312→   312→          <div className="flex items-center justify-between">
   313→   313→            {STEPS.map((s, i) => {
   314→   314→              const Icon = s.icon;
   315→   315→              const done = i < step;
   316→   316→              const active = i === step;
   317→   317→              return (
   318→   318→                <div key={s.id} className="flex flex-1 flex-col items-center gap-1">
   319→   319→                  <div className="flex w-full items-center">
   320→   320→                    {i > 0 ? (
   321→   321→                      <div className={cn("h-0.5 flex-1", i <= step ? "bg-primary" : "bg-muted")} />
   322→   322→                    ) : null}
   323→   323→                    <div
   324→   324→                      className={cn(
   325→   325→                        "flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 transition-all",
   326→   326→                        done
   327→   327→                          ? "border-primary bg-primary text-primary-foreground"
   328→   328→                          : active
   329→   329→                            ? "border-primary text-primary"
   330→   330→                            : "border-muted text-muted-foreground",
   331→   331→                      )}
   332→   332→                    >
   333→   333→                      {done ? <Check className="h-4 w-4" /> : <Icon className="h-4 w-4" />}
   334→   334→                    </div>
   335→   335→                    {i < STEPS.length - 1 ? (
   336→   336→                      <div className={cn("h-0.5 flex-1", i < step ? "bg-primary" : "bg-muted")} />
   337→   337→                    ) : null}
   338→   338→                  </div>
   339→   339→                  <span
   340→   340→                    className={cn(
   341→   341→                      "text-[10px] font-medium",
   342→   342→                      active ? "text-foreground" : "text-muted-foreground",
   343→   343→                    )}
   344→   344→                  >
   345→   345→                    {s.label}
   346→   346→                  </span>
   347→   347→                </div>
   348→   348→              );
   349→   349→            })}
   350→   350→          </div>
   351→   351→        </div>
   352→   352→
   353→   353→        {/* Step content */}
   354→   354→        <div className="rounded-lg border bg-card p-4 md:p-6">
   355→   355→          <div className="min-h-[320px]">{renderStep()}</div>
   356→   356→
   357→   357→          <Separator className="my-4" />
   358→   358→
   359→   359→          {/* Footer nav — one primary action per step (spec §23) */}
   360→   360→          <div className="flex items-center justify-between">
   361→   361→            <Button variant="ghost" size="sm" disabled={!canBack} onClick={back} className="gap-1">
   362→   362→              <ChevronLeft className="h-4 w-4" /> Back
   363→   363→            </Button>
   364→   364→            <span className="text-xs text-muted-foreground">
   365→   365→              Step {step + 1} of {STEPS.length}
   366→   366→            </span>
   367→   367→            {step < STEPS.length - 1 ? (
   368→   368→              <Button size="sm" disabled={!canNext} onClick={next} className="gap-1">
   369→   369→                Next <ChevronRight className="h-4 w-4" />
   370→   370→              </Button>
   371→   371→            ) : (
   372→   372→              <Button size="sm" onClick={createChallenge} className="gap-1.5">
   373→   373→                <Check className="h-4 w-4" /> Create Challenge
   374→   374→              </Button>
   375→   375→            )}
   376→   376→          </div>
   377→   377→        </div>
   378→   378→      </PageContent>
   379→   379→    </Page>
   380→   380→  );
   381→   381→}

   387→   387→function phaseConfigFromTemplate(cfg: {
   388→   388→  accountSize: number;
   389→   389→  profitTargetPct: number;
   390→   390→  maxDrawdownPct: number;
   391→   391→  dailyDrawdownPct: number;
   392→   392→  minTradingDays: number;
   393→   393→  maxDays: number;
   394→   394→}): PhaseConfig {
   395→   395→  return {
   396→   396→    accountSize: String(cfg.accountSize),
   397→   397→    profitTargetPct: String(cfg.profitTargetPct),
   398→   398→    maxDrawdownPct: String(cfg.maxDrawdownPct),
   399→   399→    dailyDrawdownPct: String(cfg.dailyDrawdownPct),
   400→   400→    minTradingDays: String(cfg.minTradingDays),
   401→   401→    maxDays: String(cfg.maxDays),
   402→   402→  };
   403→   403→}
   404→   404→
   405→   405→function phaseValid(p: PhaseConfig): boolean {
   406→   406→  return (
   407→   407→    Number(p.accountSize) > 0 &&
   408→   408→    Number(p.profitTargetPct) >= 0 &&
   409→   409→    Number(p.maxDrawdownPct) > 0 &&
   410→   410→    Number(p.dailyDrawdownPct) > 0 &&
   411→   411→    Number(p.minTradingDays) >= 0 &&
   412→   412→    Number(p.maxDays) >= 0
   413→   413→  );
   414→   414→}
