    96→function buildDailyCostSeries(tid: string) {
    97→  // 30d daily spend — deterministic sin pattern + weekly dip.
    98→  const seed = hashStr(tid + "daily-cost");
    99→  const baseline = 78 + (seed % 20); // $78–$97
   100→  return Array.from({ length: 30 }, (_, i) => {
   101→    const day = i + 1;
   102→    const seasonal = Math.sin((i + (seed % 5)) / 3) * 14;
   103→    const weekly = i % 7 === 5 || i % 7 === 6 ? -18 : 0;
   104→    const growth = i * 0.6;
   105→    const value = Math.max(20, Math.round((baseline + seasonal + weekly + growth) * 100) / 100);
   106→    return { day: `D${day}`, cost: value };
   107→  });
   108→}
   109→
   110→function buildPerTenantCosts(): TenantCostRow[] {
   111→  return ALL_TENANTS.map((t) => {
   112→    const seed = hashStr(t.id + "tenant-cost");
   113→    const requests = 8000 + (seed % 4000);
   114→    const tokens = requests * (140 + (seed % 60)); // 140–199 tokens/req
   115→    const cost = Math.round((tokens / 1000) * (0.6 + (seed % 40) / 100));
   116→    const trend = ((seed % 41) - 20); // -20 to +20
   117→    return {
   118→      tenantId: t.id,
   119→      tenantName: t.name,
   120→      requests,
   121→      tokens,
   122→      cost,
   123→      costPerRequest: Math.round((cost / requests) * 1000) / 1000,
   124→      trend,
   125→    };
   126→  }).sort((a, b) => b.cost - a.cost);
   127→}
   128→
   129→function buildForecastSeries(tid: string, dailySeries: { day: string; cost: number }[]) {
   130→  // 30d forward forecast — deterministic sin pattern + growth.
   131→  const seed = hashStr(tid + "cost-forecast");
   132→  const baseline = dailySeries[dailySeries.length - 1]?.cost ?? 95;
   133→  return Array.from({ length: 30 }, (_, i) => {
   134→    const day = i + 1;
   135→    const seasonal = Math.sin((i + (seed % 4)) / 3) * 11;
   136→    const growth = i * 0.9;
   137→    const predicted = Math.max(30, Math.round((baseline + seasonal + growth) * 100) / 100);
   138→    const band = Math.max(2, Math.round(predicted * 0.18));
   139→    return {
   140→      day: `D${day}`,
   141→      predicted,
   142→      low: Math.max(0, predicted - band),
   143→      high: predicted + band,
   144→    };
   145→  });
   146→}
   147→
   148→function trendTone(trend: number): "default" | "success" | "warning" | "danger" | "info" | "muted" {
   149→  if (trend > 15) return "danger";
   150→  if (trend > 5) return "warning";
   151→  if (trend > 0) return "info";
   152→  if (trend < -10) return "success";
   153→  return "muted";
   154→}
