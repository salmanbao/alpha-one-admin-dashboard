    97→interface AffectedAccountRow {

   107→function pickType(seed: number): AnomalyType {
   108→  return ANOMALY_TYPES[seed % ANOMALY_TYPES.length].key;
   109→}
   110→
   111→function pickSeverity(seed: number): AnomalyRow["severity"] {
   112→  const r = seed % 100;
   113→  if (r < 12) return "critical";
   114→  if (r < 35) return "high";
   115→  if (r < 70) return "medium";
   116→  return "low";
   117→}

   119→function anomalyDescription(type: AnomalyType, trader: string): string {
   120→  const map: Record<AnomalyType, string> = {
   121→    "trade-size": `${trader} placed a trade 4.2× larger than their 30d average`,
   122→    "off-hours": `${trader} traded outside their normal session window (3 AM UTC)`,
   123→    "pattern-break": `${trader}'s typical symbol rotation broke unexpectedly`,
   124→    "volume-spike": `${trader} hit 5.1× normal traded volume in a single hour`,
   125→    spread: `Spread compression detected on ${trader}'s EURUSD position`,
   126→    latency: `Order acknowledgement latency exceeded 850ms for ${trader}`,
   127→  };
   128→  return map[type];
   129→}

   135→function buildRecentAnomalies(
   136→  accounts: TradingAccount[],
   137→  traders: Trader[],
   138→  tid: string,
   139→): AnomalyRow[] {
   140→  if (!accounts.length || !traders.length) return [];
   141→  const traderById = new Map(traders.map((t) => [t.id, t]));
   142→  const out: AnomalyRow[] = [];
   143→  const N = 14;
   144→  for (let i = 0; i < N; i++) {
   145→    const seed = hashStr(tid + "anomaly" + i);
   146→    const acct = accounts[seed % accounts.length];
   147→    const trader = traderById.get(acct.traderId) ?? traders[seed % traders.length];
   148→    const type = pickType(seed);
   149→    const typeMeta = ANOMALY_TYPES.find((t) => t.key === type)!;
   150→    const severity = pickSeverity(seed);
   151→    // Detected between 25h and 5min ago
   152→    const detectedMin = (i * 95) + (seed % 30);
   153→    const detected = new Date(Date.now() - detectedMin * 60 * 1000).toISOString();
   154→    out.push({
   155→      id: `anom-${tid}-${i}`,
   156→      detected,
   157→      typeLabel: typeMeta.label,
   158→      typeColor: typeMeta.color,
   159→      traderId: trader.id,
   160→      traderName: trader.name,
   161→      accountId: acct.id,
   162→      accountLogin: acct.login,
   163→      severity,
   164→      confidence: Math.max(58, Math.min(99, 70 + (seed % 30))),
   165→      description: anomalyDescription(type, trader.name),
   166→    });
   167→  }
   168→  return out.sort((a, b) => new Date(b.detected).getTime() - new Date(a.detected).getTime());
   169→}

   171→function buildTypeDistribution(anomalies: AnomalyRow[]) {
   172→  return ANOMALY_TYPES.map((t) => ({
   173→    label: t.label,
   174→    value: anomalies.filter((a) => a.typeLabel === t.label).length,
   175→    color: t.color,
   176→  })).filter((d) => d.value > 0);
   177→}

   179→function buildHourlyTrend(tid: string) {
   180→  // 24h anomaly count per hour — deterministic sin pattern with peak hours.
   181→  const seed = hashStr(tid + "hourly");
   182→  return Array.from({ length: 24 }, (_, h) => {
   183→    const baseline = 3 + Math.sin((h + (seed % 6)) / 3) * 2;
   184→    const peak = h >= 9 && h <= 11 ? 2 : h >= 14 && h <= 16 ? 1.5 : 0;
   185→    const noise = (seed >> h) & 0x3; // 0–3 deterministic
   186→    return {
   187→      hour: `${h.toString().padStart(2, "0")}h`,
   188→      count: Math.max(0, Math.round(baseline + peak + noise * 0.5)),
   189→    };
   190→  });
   191→}
   192→
   193→function buildHeatmap(tid: string): number[][] {
   194→  // 7 days × 24 hours. Deterministic seeded counts 0–12.
   195→  const out: number[][] = [];
   196→  for (let d = 0; d < 7; d++) {
   197→    const row: number[] = [];
   198→    for (let h = 0; h < 24; h++) {
   199→      const seed = hashStr(tid + "hm" + d + h);
   200→      const peak =
   201→        (d < 5 && h >= 9 && h <= 16) ? 4 :
   202→        (d >= 5 && h >= 22) || (d >= 5 && h <= 2) ? 3 :
   203→        0;
   204→      const base = (seed % 4);
   205→      row.push(Math.min(12, Math.max(0, base + peak)));
   206→    }
   207→    out.push(row);
   208→  }
   209→  return out;
   210→}

   212→function buildAffectedAccounts(
   213→  accounts: TradingAccount[],
   214→  traders: Trader[],
   215→  anomalies: AnomalyRow[],
   216→): AffectedAccountRow[] {
   217→  if (!accounts.length || !traders.length) return [];
   218→  const traderById = new Map(traders.map((t) => [t.id, t]));
   219→  const counts = new Map<string, number>();
   220→  const lastByAccount = new Map<string, number>();
   221→  for (const a of anomalies) {
   222→    counts.set(a.accountId, (counts.get(a.accountId) ?? 0) + 1);
   223→    const t = new Date(a.detected).getTime();
   224→    if (!lastByAccount.has(a.accountId) || t > lastByAccount.get(a.accountId)!) {
   225→      lastByAccount.set(a.accountId, t);
   226→    }
   227→  }
   228→  return accounts
   229→    .map((acct) => {
   230→      const trader = traderById.get(acct.traderId);
   231→      const count = counts.get(acct.id) ?? 0;
   232→      const last = lastByAccount.get(acct.id);
   233→      const riskTier: AffectedAccountRow["riskTier"] =
   234→        count >= 5 ? "Critical" : count >= 3 ? "High" : count >= 1 ? "Medium" : "Low";
   235→      return {
   236→        accountId: acct.id,
   237→        login: acct.login,
   238→        traderId: acct.traderId,
   239→        traderName: trader?.name ?? acct.traderName,
   240→        anomaliesCount: count,
   241→        lastAnomaly: last ? new Date(last).toISOString() : "",
   242→        riskTier,
   243→      };
   244→    })
   245→    .filter((r) => r.anomaliesCount > 0)
   246→    .sort((a, b) => b.anomaliesCount - a.anomaliesCount)
   247→    .slice(0, 8);
   248→}

   250→function riskTierTone(tier: AffectedAccountRow["riskTier"]): "default" | "success" | "warning" | "danger" | "info" | "muted" {
   251→  switch (tier) {
   252→    case "Critical": return "danger";
   253→    case "High": return "warning";
   254→    case "Medium": return "info";
   255→    default: return "muted";
   256→  }
   257→}

   259→function relativeTime(iso: string): string {
   260→  if (!iso) return "—";
   261→  const diff = Date.now() - new Date(iso).getTime();
   262→  const min = Math.floor(diff / (1000 * 60));
   263→  if (min < 1) return "just now";
   264→  if (min < 60) return `${min}m ago`;
   265→  const hours = Math.floor(min / 60);
   266→  if (hours < 24) return `${hours}h ago`;
   267→  const days = Math.floor(hours / 24);
   268→  return `${days}d ago`;
   269→}

   275→function heatmapColor(v: number): string {
   276→  if (v === 0) return "var(--muted)";
   277→  if (v <= 2) return TERRA.emerald;
   278→  if (v <= 5) return TERRA.amber;
   279→  if (v <= 8) return TERRA.rose;
   280→  return TERRA.slate;
   281→}
   282→
   283→function heatmapOpacity(v: number): number {
   284→  if (v === 0) return 0.15;
   285→  if (v <= 2) return 0.25 + v * 0.1;
   286→  if (v <= 5) return 0.4 + (v - 2) * 0.12;
   287→  if (v <= 8) return 0.55 + (v - 5) * 0.1;
   288→  return 0.85;
   289→}

   296→function ExplainableMetricCard({
   297→  label,
   298→  value,
   299→  delta,
   300→  deltaLabel,
   301→  icon: Icon,
   302→  tone = "default",
   303→  help,
   304→}: {
   305→  label: string;
   306→  value: string | number;
   307→  delta?: number;
   308→  deltaLabel?: string;
   309→  icon?: React.ComponentType<{ className?: string; style?: React.CSSProperties }>;
   310→  tone?: "default" | "positive" | "negative" | "warning";
   311→  help?: React.ReactNode;
   312→}) {
   313→  const toneColor =
   314→    tone === "positive" ? "#059669"
   315→    : tone === "negative" ? "#e11d48"
   316→    : tone === "warning" ? "#d97706"
   317→    : "var(--brand-primary)";
   318→  const deltaClass = delta === undefined ? "" : delta >= 0 ? "text-emerald-600" : "text-rose-600";
   319→  return (
   320→    <div className="group relative overflow-hidden rounded-lg border bg-card p-4 transition-shadow hover:shadow-sm">
   321→      <span className="absolute inset-y-0 left-0 w-1" style={{ background: toneColor }} />
   322→      <div className="flex items-start justify-between pl-2">
   323→        <div className="min-w-0 flex-1">
   324→          <p className="flex items-center gap-1 truncate text-[11px] font-medium uppercase tracking-wide text-muted-foreground">
   325→            {help ? <LabelWithHelp help={help}>{label}</LabelWithHelp> : label}
   326→          </p>
   327→          <p className="mt-1.5 text-2xl font-bold tracking-tight text-foreground tabular-nums">{value}</p>
   328→          {delta !== undefined || deltaLabel ? (
   329→            <div className="mt-1.5 flex items-center gap-1.5 text-[11px]">
   330→              {delta !== undefined ? (
   331→                <span className={`inline-flex items-center gap-0.5 font-semibold ${deltaClass}`}>
   332→                  {delta >= 0 ? "▲" : "▼"} {Math.abs(delta)}%
   333→                </span>
   334→              ) : null}
   335→              {deltaLabel ? <span className="text-muted-foreground/80">{deltaLabel}</span> : null}
   336→            </div>
   337→          ) : null}
   338→        </div>
   339→        {Icon ? (
   340→          <div className="ml-2 shrink-0 rounded-md bg-muted/50 p-1.5 transition-colors group-hover:bg-muted">
   341→            <Icon className="h-4 w-4 text-muted-foreground" style={{ color: toneColor }} />
   342→          </div>
   343→        ) : null}
   344→      </div>
   345→    </div>
   346→  );
   347→}
