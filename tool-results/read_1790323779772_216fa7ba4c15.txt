   114→function relativeTime(iso: string): string {
   115→  const diff = Date.now() - new Date(iso).getTime();
   116→  const hours = Math.floor(diff / (1000 * 60 * 60));
   117→  if (hours < 1) return "just now";
   118→  if (hours < 24) return `${hours}h ago`;
   119→  const days = Math.floor(hours / 24);
   120→  if (days < 30) return `${days}d ago`;
   121→  const months = Math.floor(days / 30);
   122→  return `${months}mo ago`;
   123→}

   140→function activityFor(c: CrmContact): ActivityEntry[] {
   141→  const seed = hashStr(c.id);
   142→  // Pick 3-5 entries deterministically based on the contact hash + stage.
   143→  const count = 3 + (seed % 3);
   144→  const created = new Date(c.lastInteraction).getTime();
   145→  const out: ActivityEntry[] = [];
   146→  for (let i = 0; i < count; i++) {
   147→    const tpl = ACTIVITY_TEMPLATES[(seed + i) % ACTIVITY_TEMPLATES.length];
   148→    // Spread timestamps backwards from last interaction.
   149→    const ts = new Date(created - i * 86_400_000).toISOString();
   150→    out.push({ ...tpl, timestamp: ts });
   151→  }
   152→  return out;
   153→}

   164→function dealsFor(c: CrmContact): MockDeal[] {
   165→  const seed = hashStr(c.id);
   166→  // Stage 'opportunity' or 'customer' contacts get deals; 'lead'/'churned' don't.
   167→  if (c.stage === "lead" || c.stage === "churned") return [];
   168→  const count = (seed % 3) + 1; // 1–3 deals
   169→  const out: MockDeal[] = [];
   170→  for (let i = 0; i < count; i++) {
   171→    const stages: MockDeal["stage"][] = ["qualified", "negotiation", "closed-won", "closed-lost"];
   172→    out.push({
   173→      id: `deal-${c.id}-${i}`,
   174→      name: DEAL_NAMES[(seed + i) % DEAL_NAMES.length],
   175→      amount: 500 + ((seed >>> (i + 1)) % 8) * 500,
   176→      stage: c.stage === "customer" ? "closed-won" : stages[(seed >>> (i + 2)) % stages.length],
   177→    });
   178→  }
   179→  return out;
   180→}

   197→/**
   198→ * Deterministic lead score (0–100) derived from contact attributes.
   199→ * No Math.random — every render yields the same score for the same contact.
   200→ *
   201→ * Factors (cumulative):
   202→ *   +50  base
   203→ *   +20  pipeline value > $5,000
   204→ *   +10  pipeline value > $15,000  (stacks with the +20 above)
   205→ *   +15  source is referral or affiliate (case-insensitive)
   206→ *   +15  last interaction < 3 days
   207→ *   +8   last interaction < 7 days  (mutually exclusive with the +15)
   208→ *   -10  last interaction > 30 days
   209→ *   +12  contact has at least one associated deal
   210→ *
   211→ * Final value clamped to 0–100.
   212→ */
   213→function computeLeadScore(contact: CrmContact): number {
   214→  let score = 50;
   215→  if (contact.value && contact.value > 5000) score += 20;
   216→  if (contact.value && contact.value > 15000) score += 10;
   217→  const src = (contact.source ?? "").toLowerCase();
   218→  if (src === "referral" || src === "affiliate") score += 15;
   219→  if (contact.lastInteraction) {
   220→    const daysSince =
   221→      (Date.now() - new Date(contact.lastInteraction).getTime()) /
   222→      (1000 * 60 * 60 * 24);
   223→    if (daysSince < 3) score += 15;
   224→    else if (daysSince < 7) score += 8;
   225→    else if (daysSince > 30) score -= 10;
   226→  }
   227→  if (dealsFor(contact).length > 0) score += 12;
   228→  return Math.max(0, Math.min(100, score));
   229→}

   231→type LeadScoreTone = "hot" | "warm" | "cold";
   232→
   233→function leadScoreTone(score: number): LeadScoreTone {
   234→  if (score >= 80) return "hot";
   235→  if (score >= 50) return "warm";
   236→  return "cold";
   237→}
