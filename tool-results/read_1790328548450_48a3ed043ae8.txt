   464→function OverviewTab({ tenant }: { tenant: TenantContext }) {
   465→  const allModules = moduleRegistry.getAll();
   466→  const summary = [
   467→    { label: "Tenant ID", value: tenant.id, mono: true },
   468→    { label: "Slug", value: tenant.slug },
   469→    { label: "Plan", value: <Badge variant="outline" className="text-[10px] capitalize">{tenant.plan}</Badge> },
   470→    { label: "Status", value: <StatusBadge tone={tenantStatusTone(tenant.status)}>{tenant.status}</StatusBadge> },
   471→    { label: "Currency", value: tenant.currency },
   472→    { label: "Timezone", value: tenant.timezone },
   473→    { label: "Locale", value: tenant.locale },
   474→    { label: "Created", value: new Date(tenant.createdAt).toLocaleDateString() },
   475→    { label: "Last active", value: "2 hours ago" },
   476→    { label: "Application", value: tenant.application },
   477→  ];
   478→
   479→  return (
   480→    <div className="grid gap-4 lg:grid-cols-3">
   481→      <div className="rounded-lg border bg-card p-4 lg:col-span-2">
   482→        <p className="mb-3 text-sm font-medium">Tenant summary</p>
   483→        <dl className="grid grid-cols-2 gap-x-6 gap-y-3 text-sm md:grid-cols-2">
   484→          {summary.map((row) => (
   485→            <div key={row.label}>
   486→              <dt className="text-xs uppercase tracking-wide text-muted-foreground">{row.label}</dt>
   487→              <dd className={cn("mt-0.5", row.mono ? "font-mono text-xs" : "text-sm")}>{row.value}</dd>
   488→            </div>
   489→          ))}
   490→        </dl>
   491→      </div>
   492→      <div className="rounded-lg border bg-card p-4">
   493→        <p className="mb-3 text-sm font-medium">
   494→          <LabelWithHelp help="Modules currently enabled for this tenant. Use the Modules tab to toggle entitlements.">
   495→            Module adoption
   496→          </LabelWithHelp>
   497→        </p>
   498→        <div className="space-y-2">
   499→          {allModules.map((m) => {
   500→            const enabled = tenant.enabledModules.includes(m.manifest.id);
   501→            const Icon = m.manifest.icon;
   502→            return (
   503→              <div key={m.manifest.id} className="flex items-center gap-2">
   504→                {Icon ? <Icon className="h-3.5 w-3.5 text-muted-foreground" /> : null}
   505→                <span className="flex-1 truncate text-xs">{m.manifest.name}</span>
   506→                {enabled ? (
   507→                  <CheckCircle2 className="h-3.5 w-3.5 text-emerald-600" />
   508→                ) : (
   509→                  <span className="h-1.5 w-1.5 rounded-full bg-muted" />
   510→                )}
   511→              </div>
   512→            );
   513→          })}
   514→        </div>
   515→        <Separator className="my-3" />
   516→        <p className="text-xs text-muted-foreground">
   517→          {tenant.enabledModules.length} of {allModules.length} modules enabled
   518→        </p>
   519→        <Progress value={(tenant.enabledModules.length / allModules.length) * 100} className="mt-1.5 h-1.5" />
   520→      </div>
   521→    </div>
   522→  );
   523→}

   526→/* Tab 2 — Modules (KEY feature: toggle tenant entitlements)         */
   527→/* ------------------------------------------------------------------ */
   528→
   529→function ModulesTab({
   530→  tenant,
   531→  onChange,
   532→  onPersist,
   533→}: {
   534→  tenant: TenantContext;
   535→  onChange: (t: TenantContext) => void;
   536→  onPersist: (t: TenantContext) => void;
   537→}) {
   538→  const allModules = moduleRegistry.getAll();
   539→
   540→  const toggle = (moduleId: string) => {
   541→    const enabled = new Set(tenant.enabledModules);
   542→    const willEnable = !enabled.has(moduleId);
   543→    if (willEnable) enabled.add(moduleId);
   544→    else enabled.delete(moduleId);
   545→    const updated: TenantContext = {
   546→      ...tenant,
   547→      enabledModules: Array.from(enabled),
   548→    };
   549→    onChange(updated);
