   139→
   140→function slugify(s: string): string {
   141→  return s
   142→    .toLowerCase()
   143→    .trim()
   144→    .replace(/[^a-z0-9]+/g, "-")
   145→    .replace(/^-+|-+$/g, "")
   146→    .slice(0, 32);
   147→}
   148→
   149→function initialsFromName(name: string): string {
   150→  const parts = name.trim().split(/\s+/).filter(Boolean);
   151→  if (parts.length === 0) return "";
   152→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   153→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();
   154→}
   155→
   156→const REQUIRED_ASTERISK = <span className="text-rose-500" aria-hidden>*</span>;
   157→
   158→function RequiredLabel({ children }: { children: React.ReactNode }) {
   159→  return (
   160→    <Label className="text-xs font-medium text-muted-foreground">
   161→      {children} {REQUIRED_ASTERISK}
   162→    </Label>
   163→  );
   164→}

   186→}

   192→export function CreateTenantPage() {
   193→  const { navigate, registerTenant, pushNotification } = usePlatform();
   194→  const [state, setState] = useState<WizardState>(INITIAL_STATE);
   195→  const [step, setStep] = useState(0);
   196→
   197→  const allModules = moduleRegistry.getAll();
   198→
   199→  /* ---- Validation per step ---- */
   200→  const stepValid = (i: number): boolean => {
   201→    switch (i) {
   202→      case 0:
   203→        return state.name.trim().length > 1 && state.slug.trim().length > 0;
   204→      case 1:
   205→        return state.initials.trim().length > 0 && state.primaryColor.length > 0;
   206→      case 2:
   207→        return state.enabledModules.length > 0;
   208→      case 3:
   209→        return (
   210→          state.adminName.trim().length > 1 &&
   211→          /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(state.adminEmail)
   212→        );
   213→      case 4:
   214→        return true;
   215→      default:
   216→        return false;
   217→    }
   218→  };
   219→
   220→  const canNext = stepValid(step) && step < STEPS.length - 1;
   221→  const canBack = step > 0;
   222→  const next = () => setStep((s) => Math.min(STEPS.length - 1, s + 1));
   223→  const back = () => setStep((s) => Math.max(0, s - 1));
   224→
   225→  /* ---- Create tenant ---- */
   226→  const createTenant = () => {
   227→    const id = `tenant-${state.slug}-${Math.random().toString(36).slice(2, 6)}`;
   228→    const newTenant: TenantContext = {
   229→      id,
   230→      slug: state.slug,
   231→      name: state.name,
   232→      application: "prop-admin",
   233→      branding: {
   234→        name: state.name,
   235→        tagline: state.tagline,
   236→        logoUrl: state.logoUrl || undefined,
   237→        initials: state.initials.toUpperCase(),
   238→        primaryColor: state.primaryColor,
   239→        accentColor: state.accentColor,
   240→        surfaceColor: state.surfaceColor,
   241→        radius: state.radius,
   242→      },
   243→      locale: "en-US",
   244→      timezone: state.timezone,
   245→      currency: state.currency,
   246→      enabledModules: state.enabledModules,
   247→      enabledFeatures: [],
   248→      terminology: {},
   249→      plan: state.plan,
   250→      status: "trial",
   251→      createdAt: new Date().toISOString(),
   252→    };
   253→    // Register the tenant in the platform roster WITHOUT hijacking the
   254→    // operator's context — the super-admin stays in the Platform shell and
   255→    // is taken to the tenant detail page as the wizard promises. Previously
   256→    // this called setTenant(), which swapped the whole sidebar into the new
   257→    // tenant's module set and yanked the operator out of the admin flow.
   258→    registerTenant(newTenant);
   259→    pushNotification({
   260→      title: "Tenant created",
   261→      message: `${state.name} is ready for configuration.`,
   262→      severity: "success",
   263→      module: "super-admin",
   264→    });
   265→    toast({
   266→      title: "Tenant created",
   267→      description: `${state.name} is ready for configuration.`,
   268→    });
   269→    setState(INITIAL_STATE);
   270→    setStep(0);
   271→    navigate("tenant-detail", { id });
   272→  };
   273→
   274→  /* ---- Step content renderer ---- */
   275→  const renderStep = () => {
   276→    switch (step) {
   277→      case 0:
   278→        return (
   279→          <BasicsStep state={state} setState={setState} />
   280→        );
   281→      case 1:
   282→        return <BrandingStep state={state} setState={setState} />;
   283→      case 2:
   284→        return <ModulesStep state={state} setState={setState} allModules={allModules} />;
   285→      case 3:
   286→        return <AdminStep state={state} setState={setState} />;
   287→      case 4:
   288→        return (
   289→          <ReviewStep
   290→            state={state}
   291→            allModules={allModules}
   292→            onCreate={createTenant}
   293→          />
   294→        );
   295→      default:
   296→        return null;
   297→    }
   298→  };
   299→
   300→  return (
   301→    <Page>
   302→      <PageHeader
   303→        title="Create Tenant"
   304→        description="Guided wizard to provision a new tenant on the platform."
   305→        icon={Sparkles}
   306→        actions={
   307→          <Button variant="ghost" size="sm" onClick={() => navigate("tenants")}>
   308→            Cancel
   309→          </Button>
   310→        }
   311→      />
   312→      <PageContent>
   313→        {/* Step indicator */}
   314→        <div className="rounded-lg border bg-card p-4">
   315→          <div className="flex items-center justify-between">
   316→            {STEPS.map((s, i) => {
   317→              const Icon = s.icon;
   318→              const done = i < step;
   319→              const active = i === step;
   320→              return (
   321→                <div key={s.id} className="flex flex-1 flex-col items-center gap-1">
   322→                  <div className="flex w-full items-center">
   323→                    {i > 0 ? (
   324→                      <div className={cn("h-0.5 flex-1", i <= step ? "bg-primary" : "bg-muted")} />
   325→                    ) : null}
   326→                    <div
   327→                      className={cn(
   328→                        "flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 transition-all",
   329→                        done
   330→                          ? "border-primary bg-primary text-primary-foreground"
   331→                          : active
   332→                            ? "border-primary text-primary"
   333→                            : "border-muted text-muted-foreground",
   334→                      )}
   335→                    >
   336→                      {done ? <Check className="h-4 w-4" /> : <Icon className="h-4 w-4" />}
   337→                    </div>
   338→                    {i < STEPS.length - 1 ? (
   339→                      <div className={cn("h-0.5 flex-1", i < step ? "bg-primary" : "bg-muted")} />
   340→                    ) : null}
   341→                  </div>
   342→                  <span
   343→                    className={cn(
   344→                      "text-[10px] font-medium",
   345→                      active ? "text-foreground" : "text-muted-foreground",
   346→                    )}
   347→                  >
   348→                    {s.label}
   349→                  </span>
   350→                </div>
   351→              );
   352→            })}
   353→          </div>
   354→        </div>
   355→
   356→        {/* Step content */}
   357→        <div className="rounded-lg border bg-card p-4 md:p-6">
   358→          <div className="min-h-[320px]">{renderStep()}</div>
   359→
   360→          <Separator className="my-4" />
   361→
   362→          <div className="flex items-center justify-between">
   363→            <Button variant="ghost" size="sm" disabled={!canBack} onClick={back} className="gap-1">
   364→              <ChevronLeft className="h-4 w-4" />Back
   365→            </Button>
   366→            <span className="text-xs text-muted-foreground">
   367→              Step {step + 1} of {STEPS.length}
   368→            </span>
   369→            {step < STEPS.length - 1 ? (
   370→              <Button size="sm" disabled={!canNext} onClick={next} className="gap-1">
   371→                Next <ChevronRight className="h-4 w-4" />
   372→              </Button>
   373→            ) : (
   374→              <Button size="sm" onClick={createTenant} className="gap-1.5">
   375→                <Check className="h-4 w-4" /> Create tenant
   376→              </Button>
   377→            )}
   378→          </div>
   379→        </div>
   380→      </PageContent>
   381→    </Page>
   382→  );
   383→}

   657→/* Step 3 — Modules                                                   */
   658→/* ------------------------------------------------------------------ */
   659→
   660→function ModulesStep({
   661→  state,
   662→  setState,
   663→  allModules,
   664→}: {
   665→  state: WizardState;
   666→  setState: React.Dispatch<React.SetStateAction<WizardState>>;
   667→  allModules: ReturnType<typeof moduleRegistry.getAll>;
   668→}) {
   669→  const toggle = (moduleId: string) => {
   670→    setState((s) => {
   671→      const set = new Set(s.enabledModules);
   672→      if (set.has(moduleId)) set.delete(moduleId);
   673→      else set.add(moduleId);
   674→      return { ...s, enabledModules: Array.from(set) };
   675→    });
   676→  };
   677→
   678→  return (
   679→    <div className="space-y-4">
   680→      <div className="flex items-center justify-between">
   681→        <div>
   682→          <RequiredLabel>Module selection</RequiredLabel>
   683→          <p className="text-xs text-muted-foreground">
   684→            Pick which modules this tenant can access. Core modules are pre-selected.
   685→          </p>
   686→        </div>
   687→        <Badge variant="secondary">
   688→          {state.enabledModules.length} of {allModules.length} selected
   689→        </Badge>
   690→      </div>
   691→      <div className="grid gap-3 sm:grid-cols-2 lg:grid-cols-3">
   692→        {allModules.map((m) => {
   693→          const selected = state.enabledModules.includes(m.manifest.id);
   694→          const Icon = m.manifest.icon;
   695→          const accent = m.manifest.accentColor ?? state.primaryColor;
   696→          const isCore = CORE_MODULE_IDS.includes(m.manifest.id);
   697→          return (
   698→            <button
   699→              key={m.manifest.id}
   700→              type="button"
   701→              onClick={() => toggle(m.manifest.id)}
   702→              className={cn(
   703→                "flex flex-col items-start gap-2 rounded-lg border p-3 text-left transition-all hover:shadow-sm",
   704→                selected
   705→                  ? "border-primary bg-primary/5 ring-1 ring-primary/20"
   706→                  : "border-border hover:border-primary/40",
   707→              )}
   708→              aria-pressed={selected}
   709→            >
   710→              <div className="flex w-full items-start justify-between">
   711→                <div className="flex items-center gap-2">
   712→                  <div className="rounded-md p-1.5" style={{ background: `${accent}1a`, color: accent }}>
   713→                    {Icon ? <Icon className="h-4 w-4" /> : null}
   714→                  </div>
   715→                  <div>
   716→                    <p className="text-sm font-medium leading-tight">{m.manifest.name}</p>
   717→                    <p className="text-[10px] uppercase tracking-wide text-muted-foreground">
   718→                      {m.manifest.category ?? "module"}
   719→                    </p>
   720→                  </div>
   721→                </div>
   722→                <Checkbox checked={selected} className="pointer-events-none" />
   723→              </div>
   724→              <p className="text-xs text-muted-foreground line-clamp-2">
   725→                {m.manifest.description ?? "No description available."}
   726→              </p>
   727→              {isCore ? (
   728→                <Badge variant="secondary" className="text-[9px]">Core</Badge>
   729→              ) : null}
   730→            </button>
   731→          );
   732→        })}
   733→      </div>
   734→    </div>
   735→  );
   736→}
