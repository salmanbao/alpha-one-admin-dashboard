   131→export function TenantDetailPage() {
   132→  const { router, navigate, availableTenants, setTenant, pushNotification } = usePlatform();
   133→  const tenantId = router.params.id;
   134→
   135→  const foundTenant = useMemo(
   136→    () => availableTenants.find((t) => t.id === tenantId && t.id !== "platform"),
   137→    [availableTenants, tenantId],
   138→  );
   139→
   140→  // Local working copy so module toggles / config edits are reflected
   141→  // immediately in this surface and also pushed to global context.
   142→  const [localTenant, setLocalTenant] = useState<TenantContext | null>(foundTenant ?? null);
   143→
   144→  // KPI data — pulled from mock domain functions
   145→  const tenantTraders = useMemo(

   156→  );

   159→    [localTenant],
   160→  );
   161→  const tenantKyc = useMemo(
   162→    () => (localTenant ? getTenantKyc(localTenant.id) : []),
   163→    [localTenant],
   164→  );
   165→  const tenantAudit = useMemo(
   166→    () => (localTenant ? getTenantAudit(localTenant.id).slice(0, 12) : []),

   168→  );

   170→  if (!localTenant) {

   172→      <Page>

   176→        <EmptyState

   199→    setLocalTenant(updated);
   200→    setTenant(updated);

   202→      title: `Tenant ${verb}`,
   203→      message: `${localTenant.name} is now ${status}.`,

   209→      title: `Tenant ${verb}`,
   210→      description: `${localTenant.name} is now ${status}.`,

   216→      <Button

   218→        size="sm"
   219→        className="w-fit"

   407→    { label: "Timezone", value: tenant.timezone },

   409→    { label: "Created", value: new Date(tenant.createdAt).toLocaleDateString() },
   410→    { label: "Last active", value: "2 hours ago" },
   411→    { label: "Application", value: tenant.application },

   415→    <div className="grid gap-4 lg:grid-cols-3">
   416→      <div className="rounded-lg border bg-card p-4 lg:col-span-2">

   418→        <dl className="grid grid-cols-2 gap-x-6 gap-y-3 text-sm md:grid-cols-2">
   419→          {summary.map((row) => (
   420→            <div key={row.label}>
   421→              <dt className="text-xs uppercase tracking-wide text-muted-foreground">{row.label}</dt>

   428→        <p className="mb-3 text-sm font-medium">
   429→          <LabelWithHelp help="Modules currently enabled for this tenant. Use the Modules tab to toggle entitlements.">

   442→                  <CheckCircle2 className="h-3.5 w-3.5 text-emerald-600" />

   444→                  <span className="h-1.5 w-1.5 rounded-full bg-muted" />

   448→          })}
   449→        </div>
   450→        <Separator className="my-3" />
   451→        <p className="text-xs text-muted-foreground">
   452→          {tenant.enabledModules.length} of {allModules.length} modules enabled
   453→        </p>
   454→        <Progress value={(tenant.enabledModules.length / allModules.length) * 100} className="mt-1.5 h-1.5" />
   455→      </div>
   456→    </div>
   457→  );
   458→}

   460→/* ------------------------------------------------------------------ */
   461→/* Tab 2 — Modules (KEY feature: toggle tenant entitlements)         */
   462→/* ------------------------------------------------------------------ */
   463→
   464→function ModulesTab({
   465→  tenant,
   466→  onChange,
   467→  onPersist,
   468→}: {
   469→  tenant: TenantContext;
   470→  onChange: (t: TenantContext) => void;
   471→  onPersist: (t: TenantContext) => void;
   472→}) {
   473→  const allModules = moduleRegistry.getAll();
   474→
   475→  const toggle = (moduleId: string) => {
   476→    const enabled = new Set(tenant.enabledModules);
   477→    const willEnable = !enabled.has(moduleId);
   478→    if (willEnable) enabled.add(moduleId);
   479→    else enabled.delete(moduleId);
   480→    const updated: TenantContext = {
   481→      ...tenant,
   482→      enabledModules: Array.from(enabled),
   483→    };
   484→    onChange(updated);
   485→    onPersist(updated);
   486→    const mod = allModules.find((m) => m.manifest.id === moduleId);
   487→    toast({
   488→      title: willEnable ? "Module enabled" : "Module disabled",
   489→      description: `${mod?.manifest.name ?? moduleId} is now ${willEnable ? "accessible" : "hidden"} for ${tenant.name}.`,
   490→    });
   491→  };
   492→
   493→  return (
   494→    <div className="space-y-3">
   495→      <div className="flex items-center justify-between rounded-lg border bg-card p-3">
   496→        <div>
   497→          <p className="text-sm font-medium">
   498→            <LabelWithHelp help="Toggle modules to control which features this tenant can access. Disabled modules disappear from the tenant's sidebar and dashboard instantly.">
   499→              Module entitlements
   500→            </LabelWithHelp>
   501→          </p>
   502→          <p className="text-xs text-muted-foreground">
   503→            {tenant.enabledModules.length} of {allModules.length} modules enabled
   504→          </p>
   505→        </div>
   506→        <Progress
   507→          value={(tenant.enabledModules.length / allModules.length) * 100}
   508→          className="h-2 w-32"
   509→        />
   510→      </div>
   511→      <div className="grid gap-3 md:grid-cols-2 lg:grid-cols-3">
   512→        {allModules.map((m) => {
   513→          const enabled = tenant.enabledModules.includes(m.manifest.id);
   514→          const Icon = m.manifest.icon;
   515→          const accent = m.manifest.accentColor ?? tenant.branding.primaryColor;
   516→          return (
   517→            <Card key={m.manifest.id} className={cn("gap-4 py-4", enabled && "ring-1 ring-primary/20")}>
   518→              <CardContent className="space-y-3">
   519→                <div className="flex items-start justify-between">
   520→                  <div className="flex items-start gap-2.5">
   521→                    <div className="rounded-md p-2" style={{ background: `${accent}1a`, color: accent }}>
   522→                      {Icon ? <Icon className="h-4 w-4" /> : null}
   523→                    </div>
   524→                    <div>
   525→                      <p className="text-sm font-semibold leading-tight">{m.manifest.name}</p>
   526→                      <p className="text-[10px] uppercase tracking-wide text-muted-foreground">
   527→                        {m.manifest.category ?? "module"}
   528→                      </p>
   529→                    </div>
   530→                  </div>
   531→                  <Switch
   532→                    checked={enabled}
   533→                    onCheckedChange={() => toggle(m.manifest.id)}
   534→                    aria-label={`Toggle ${m.manifest.name}`}
   535→                  />
   536→                </div>
   537→                <p className="text-xs text-muted-foreground leading-relaxed">
   538→                  {m.manifest.description ?? "No description available."}
   539→                </p>
   540→                <div className="flex flex-wrap items-center gap-1.5">
   541→                  <Badge variant="outline" className="text-[9px]">{m.manifest.version}</Badge>
   542→                  {m.manifest.supportedApplications?.map((a) => (
   543→                    <Badge key={a} variant="secondary" className="text-[9px]">{a}</Badge>
   544→                  ))}
   545→                </div>
   546→              </CardContent>
   547→            </Card>
   548→          );
   549→        })}
   550→      </div>
   551→    </div>
