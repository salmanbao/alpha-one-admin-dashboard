    80→/** Trader churn risk score (0–100). Combines inactivity + low winRate + low
    81→ * equity + a deterministic per-trader seed. Higher = more likely to churn. */
    82→function churnScore(trader: Trader): number {
    83→  const seed = hashStr(trader.id + "churn");
    84→  const joinedDays = Math.floor(
    85→    (Date.now() - new Date(trader.joinedAt).getTime()) / (1000 * 60 * 60 * 24),
    86→  );
    87→  const inactivity = Math.min(30, Math.floor((seed % 30) + joinedDays / 12));
    88→  const winPenalty = Math.max(0, 50 - trader.winRate) * 0.6;
    89→  const pnlPenalty = trader.totalPnl < 0 ? 18 : 4;
    90→  const equityLow = trader.equity < 8000 ? 10 : 0;
    91→  const base = 18 + (seed % 22);
    92→  return Math.max(2, Math.min(99, Math.round(base + inactivity * 0.4 + winPenalty + pnlPenalty + equityLow)));
    93→}
    94→
    95→/** Recommended retention action based on churn score. */
    96→function recommendedAction(score: number): string {
    97→  if (score >= 80) return "Schedule check-in call";
    98→  if (score >= 60) return "Offer retention bonus";
    99→  if (score >= 40) return "Send re-engagement email";
   100→  if (score >= 20) return "Send weekly digest";
   101→  return "Monitor only";
   102→}
   103→
   104→/** Deterministic days-since-last-activity proxy (0–45d). */
   105→function daysInactive(trader: Trader): number {
   106→  const seed = hashStr(trader.id + "inactive");
   107→  return Math.min(45, Math.max(0, (seed % 28) + Math.floor(churnScore(trader) / 4)));
   108→}
   109→
   110→/** Payout fraud risk score (0–100). Seeded by payout.id + amount + method. */
   111→function payoutRisk(p: Payout): number {
   112→  const seed = hashStr(p.id + p.amount + p.method);
   113→  const amountPenalty = p.amount > 8000 ? 24 : p.amount > 4000 ? 12 : 4;
   114→  const methodPenalty = p.method === "crypto" ? 14 : p.method === "skrill" ? 8 : 2;
   115→  const recent = Date.now() - new Date(p.createdAt).getTime() < 1000 * 60 * 60 * 24 * 7 ? 6 : 0;
   116→  const base = 22 + (seed % 35);
   117→  return Math.max(3, Math.min(98, Math.round(base + amountPenalty + methodPenalty + recent)));
   118→}
   119→
   120→/** Risk factor badges for a payout. */
   121→function payoutRiskFactors(p: Payout): string[] {
   122→  const factors: string[] = [];
   123→  if (p.amount > 8000) factors.push("High value");
   124→  const seed = hashStr(p.id + "factors");
   125→  if (seed % 3 === 0) factors.push("Multiple IPs");
   126→  if (seed % 4 === 0) factors.push("New account");
   127→  if (seed % 5 === 0) factors.push("Unusual pattern");
   128→  if (p.method === "crypto") factors.push("Crypto method");
   129→  return factors.length ? factors : ["None"];
   130→}
   131→
   132→/** Trader success probability (0–100). Higher = more likely to succeed. */
   133→function successProbability(trader: Trader): number {
   134→  const seed = hashStr(trader.id + "success");
   135→  const win = trader.winRate * 0.8;
   136→  const pnlBonus = trader.totalPnl > 0 ? 18 : trader.totalPnl < 0 ? -10 : 2;
   137→  const tradeBonus = Math.min(15, trader.trades / 20);
   138→  const base = 28 + (seed % 22);
   139→  return Math.max(8, Math.min(98, Math.round(base + win + pnlBonus + tradeBonus)));
   140→}

   169→interface SuccessRow {
   170→  rank: number;
   171→  traderId: string;
   172→  name: string;

   175→  tier: "high" | "medium" | "low";

   178→function buildChurnRows(traders: Trader[]): ChurnRow[] {
   179→  return traders
   180→    .map((t) => ({
   181→      rank: 0,
   182→      id: t.id,
   183→      name: t.name,
   184→      email: t.email,
   185→      churnScore: churnScore(t),
   186→      daysInactive: daysInactive(t),
   187→      equity: t.equity,
   188→      lastActivity: `${daysInactive(t)}d ago`,
   189→      action: recommendedAction(churnScore(t)),
   190→    }))
   191→    .sort((a, b) => b.churnScore - a.churnScore)
   192→    .slice(0, 10)
   193→    .map((r, i) => ({ ...r, rank: i + 1 }));
   194→}
   195→
   196→function buildPayoutRiskRows(payouts: Payout[]): PayoutRiskRow[] {
   197→  const reviewers = ["Sarah Chen", "Marcus Webb", "Priya Nair", "Elena Rossi"];
   198→  return payouts
   199→    .map((p) => {
   200→      const score = payoutRisk(p);
   201→      return {
   202→        payoutId: p.reference || p.id,
   203→        traderId: p.traderId,
   204→        traderName: p.traderName,
   205→        amount: p.amount,
   206→        currency: p.currency,
   207→        riskScore: score,
   208→        riskFactors: payoutRiskFactors(p),
   209→        reviewer: reviewers[hashStr(p.id) % reviewers.length],
   210→      };
   211→    })
   212→    .sort((a, b) => b.riskScore - a.riskScore)
   213→    .slice(0, 8);
   214→}
   215→
   216→function buildSuccessRows(traders: Trader[]): SuccessRow[] {
   217→  return traders
   218→    .filter((t) => t.status === "active")
   219→    .map((t) => {
   220→      const probability = successProbability(t);
   221→      const tier: SuccessRow["tier"] =
   222→        probability >= 70 ? "high" : probability >= 40 ? "medium" : "low";
   223→      return {
   224→        rank: 0,
   225→        traderId: t.id,
   226→        name: t.name,
   227→        email: t.email,
   228→        probability,
   229→        tier,
   230→      };
   231→    })
   232→    .sort((a, b) => b.probability - a.probability)
   233→    .slice(0, 20)
   234→    .map((r, i) => ({ ...r, rank: i + 1 }));
   235→}

   237→function buildChurnBuckets(traders: Trader[]) {
   238→  const buckets = [
   239→    { label: "Very Low", lo: 0, hi: 20, count: 0 },
   240→    { label: "Low", lo: 20, hi: 40, count: 0 },
   241→    { label: "Medium", lo: 40, hi: 60, count: 0 },
   242→    { label: "High", lo: 60, hi: 80, count: 0 },
   243→    { label: "Critical", lo: 80, hi: 101, count: 0 },
   244→  ];
   245→  for (const t of traders) {
   246→    const score = churnScore(t);
   247→    const bucket = buckets.find((b) => score >= b.lo && score < b.hi);
   248→    if (bucket) bucket.count += 1;
   249→  }
   250→  return buckets.map((b) => ({ label: b.label, count: b.count }));
   251→}

   253→function buildSignupForecast(tid: string) {
   254→  // 30d deterministic forecast — daily predicted new signups with a
   255→  // confidence band (±N). Sin-based baseline + weekly seasonality.
   256→  const seed = hashStr(tid + "signup-forecast");
   257→  const baseline = 8 + (seed % 6); // 8–13 baseline
   258→  return Array.from({ length: 30 }, (_, i) => {
   259→    const day = i + 1;
   260→    const seasonal = Math.sin((i + (seed % 7)) / 3) * 3.2;
   261→    const weekly = i % 7 === 0 || i % 7 === 6 ? -2.5 : 0;
   262→    const growth = i * 0.18;
   263→    const predicted = Math.max(2, Math.round(baseline + seasonal + weekly + growth));
   264→    const band = Math.max(1, Math.round(predicted * 0.22));
   265→    return {
   266→      day: `D${day}`,
   267→      predicted,
   268→      low: Math.max(0, predicted - band),
   269→      high: predicted + band,
   270→    };
   271→  });
   272→}

   278→function churnToneFor(score: number): "default" | "success" | "warning" | "danger" | "info" | "muted" {
   279→  if (score >= 80) return "danger";
   280→  if (score >= 60) return "warning";

   285→function riskToneFor(score: number): "default" | "success" | "warning" | "danger" | "info" | "muted" {
   286→  if (score >= 75) return "danger";
   287→  if (score >= 50) return "warning";
   288→  if (score >= 30) return "info";
   289→  return "success";
   290→}
