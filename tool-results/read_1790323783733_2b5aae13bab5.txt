   114→   114→function relativeTime(iso: string): string {
   115→   115→  const diff = Date.now() - new Date(iso).getTime();
   116→   116→  const hours = Math.floor(diff / (1000 * 60 * 60));
   117→   117→  if (hours < 1) return "just now";
   118→   118→  if (hours < 24) return `${hours}h ago`;
   119→   119→  const days = Math.floor(hours / 24);
   120→   120→  if (days < 30) return `${days}d ago`;
   121→   121→  const months = Math.floor(days / 30);
   122→   122→  return `${months}mo ago`;
   123→   123→}

   140→   140→function activityFor(c: CrmContact): ActivityEntry[] {
   141→   141→  const seed = hashStr(c.id);
   142→   142→  // Pick 3-5 entries deterministically based on the contact hash + stage.
   143→   143→  const count = 3 + (seed % 3);
   144→   144→  const created = new Date(c.lastInteraction).getTime();
   145→   145→  const out: ActivityEntry[] = [];
   146→   146→  for (let i = 0; i < count; i++) {
   147→   147→    const tpl = ACTIVITY_TEMPLATES[(seed + i) % ACTIVITY_TEMPLATES.length];
   148→   148→    // Spread timestamps backwards from last interaction.
   149→   149→    const ts = new Date(created - i * 86_400_000).toISOString();
   150→   150→    out.push({ ...tpl, timestamp: ts });
   151→   151→  }
   152→   152→  return out;
   153→   153→}

   197→   197→/**
   198→   198→ * Deterministic lead score (0–100) derived from contact attributes.
   199→   199→ * No Math.random — every render yields the same score for the same contact.
   200→   200→ *
   201→   201→ * Factors (cumulative):
   202→   202→ *   +50  base
   203→   203→ *   +20  pipeline value > $5,000
   204→   204→ *   +10  pipeline value > $15,000  (stacks with the +20 above)
   205→   205→ *   +15  source is referral or affiliate (case-insensitive)
   206→   206→ *   +15  last interaction < 3 days
   207→   207→ *   +8   last interaction < 7 days  (mutually exclusive with the +15)
   208→   208→ *   -10  last interaction > 30 days
   209→   209→ *   +12  contact has at least one associated deal
   210→   210→ *
   211→   211→ * Final value clamped to 0–100.
   212→   212→ */
   213→   213→function computeLeadScore(contact: CrmContact): number {
   214→   214→  let score = 50;
   215→   215→  if (contact.value && contact.value > 5000) score += 20;
   216→   216→  if (contact.value && contact.value > 15000) score += 10;
   217→   217→  const src = (contact.source ?? "").toLowerCase();
   218→   218→  if (src === "referral" || src === "affiliate") score += 15;
   219→   219→  if (contact.lastInteraction) {
   220→   220→    const daysSince =
   221→   221→      (Date.now() - new Date(contact.lastInteraction).getTime()) /
   222→   222→      (1000 * 60 * 60 * 24);
   223→   223→    if (daysSince < 3) score += 15;
   224→   224→    else if (daysSince < 7) score += 8;
   225→   225→    else if (daysSince > 30) score -= 10;
   226→   226→  }
   227→   227→  if (dealsFor(contact).length > 0) score += 12;
   228→   228→  return Math.max(0, Math.min(100, score));
   229→   229→}
