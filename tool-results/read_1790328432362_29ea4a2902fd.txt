     3→/**
     4→ * Challenge Wizard Page (spec §14, §15, §16, §12, §23)
     5→ *
     6→ * Guided 7-step workflow to create a new challenge template. Each step
     7→ * represents a meaningful mental concept:
     8→ *
     9→ *   1. Select challenge type        — what kind of evaluation is this?
    10→ *   2. Phase 1 config               — risk / target / time limits
    11→ *   3. Phase 2 config (if applicable)— verification phase
    12→ *   4. Trading rules                 — what's allowed?
    13→ *   5. Payout rules                  — profit split / frequency / methods
    14→ *   6. Risk rules                    — daily / overall / trailing
    15→ *   7. Review                        — summary + Create Challenge
    16→ *
    17→ * Smart defaults (spec §15): selecting a challenge type auto-fills the
    18→ * phase config forms from getChallengePhaseConfigs(typeId). Templates over
    19→ * blank forms (spec §16).
    20→ *
    21→ * Step indicator pattern mirrors OnboardingWizard (done/active/inactive
    22→ * circles + connecting progress bars).
    23→ *
    24→ * All form state is local useState. No persistence — on Create, we show
    25→ * a toast and reset.
    26→ */

   161→/* ------------------------------------------------------------------ */
   162→/* Component                                                          */
   163→/* ------------------------------------------------------------------ */
   164→
   165→export function ChallengeWizardPage() {
   166→  const [state, setState] = useState<WizardState>(INITIAL_STATE);
   167→  const [step, setStep] = useState(0);
   168→
   169→  const challengeTypes = getChallengeTypes();
   170→  const selectedType = useMemo(
   171→    () => challengeTypes.find((t) => t.id === state.challengeTypeId) ?? null,
   172→    [challengeTypes, state.challengeTypeId],
   173→  );
   174→  const hasPhase2 = (selectedType?.phases ?? 0) >= 2;
   175→
   176→  /* ---- Smart defaults: select type → fill phases ---- */
   177→  const selectChallengeType = (typeId: string) => {
   178→    const configs = getChallengePhaseConfigs(typeId);
   179→    // Find first non-funded phase as Phase 1, second non-funded as Phase 2
   180→    const evaluationPhases = configs.filter((p) => !p.isFunded);
   181→    const phase1Cfg = evaluationPhases[0] ?? configs[0];
   182→    const phase2Cfg = evaluationPhases[1] ?? configs[1];
   183→    setState((s) => ({
   184→      ...s,
   185→      challengeTypeId: typeId,
   186→      phase1: phase1Cfg ? phaseConfigFromTemplate(phase1Cfg) : { ...EMPTY_PHASE },
   187→      phase2: phase2Cfg ? phaseConfigFromTemplate(phase2Cfg) : { ...EMPTY_PHASE },
   188→    }));
   189→  };
   190→
   191→  /* ---- Validation per step ---- */
   192→  const stepValid = (i: number): boolean => {
   193→    switch (i) {
   194→      case 0:
   195→        return state.challengeTypeId !== null;
   196→      case 1:
   197→        return phaseValid(state.phase1);
   198→      case 2:
   199→        return !hasPhase2 || phaseValid(state.phase2);
   200→      case 3:
   201→        return true;
   202→      case 4:
   203→        // Payout step — at least one payout method must be selected and
   204→        // the profit-split % must be a number 0–100. Previously returned
   205→        // `true` unconditionally, so the operator could click Next with
   206→        // zero payout methods and an invalid split.
   207→        return (
   208→          state.payoutRules.payoutMethods.length > 0 &&
   209→          Number(state.payoutRules.profitSplitPct) >= 0 &&
   210→          Number(state.payoutRules.profitSplitPct) <= 100
   211→        );
   212→      case 5:
   213→        return (
   214→          Number(state.riskRules.maxDailyLossPct) > 0 &&
   215→          Number(state.riskRules.maxOverallLossPct) > 0
   216→        );
   217→      case 6:
   218→        return true;
   219→      default:
   220→        return false;
   221→    }
   222→  };
   223→
   224→  const canNext = stepValid(step) && step < STEPS.length - 1;
   225→  const canBack = step > 0;
   226→
   227→  const next = () => setStep((s) => Math.min(STEPS.length - 1, s + 1));
   228→  const back = () => setStep((s) => Math.max(0, s - 1));
   229→
   230→  const createChallenge = () => {
   231→    toast({
   232→      title: "Challenge created",
   233→      description: `${selectedType?.name ?? "Challenge"} template saved and ready to assign to traders.`,
   234→    });
   235→    setState(INITIAL_STATE);
   236→    setStep(0);
   237→  };
   238→
   239→  /* ---- Step content renderer ---- */
   240→  const renderStep = () => {
   241→    switch (step) {
   242→      case 0:
   243→        return (
   244→          <StepType
   245→            types={challengeTypes}
   246→            selectedId={state.challengeTypeId}
   247→            onSelect={selectChallengeType}
   248→          />
   249→        );
   250→      case 1:
   251→        return (
   252→          <PhaseForm
   253→            title="Phase 1 — Evaluation"
   254→            description="Initial evaluation phase. Traders must reach the profit target while staying within risk limits."
   255→            value={state.phase1}
   256→            onChange={(next) => setState((s) => ({ ...s, phase1: next }))}
   257→          />
   258→        );
   259→      case 2:
   260→        if (!hasPhase2) {
   261→          return (
   262→            <NoticeCard
   263→              title="Phase 2 not applicable"
   264→              description={`"${selectedType?.name ?? "This challenge"}" only has one phase. You can skip this step and continue to Trading Rules.`}
   265→            />
   266→          );
   267→        }
   268→        return (
   269→          <PhaseForm
   270→            title="Phase 2 — Verification"
   271→            description="Second evaluation phase. Typically a lower profit target with the same risk limits as Phase 1."
   272→            value={state.phase2}
   273→            onChange={(next) => setState((s) => ({ ...s, phase2: next }))}
   274→          />
   275→        );
   276→      case 3:
   277→        return (
   278→          <TradingRulesForm
   279→            value={state.tradingRules}
   280→            onChange={(next) => setState((s) => ({ ...s, tradingRules: next }))}
   281→          />
   282→        );
   283→      case 4:
   284→        return (
   285→          <PayoutRulesForm
   286→            value={state.payoutRules}
   287→            onChange={(next) => setState((s) => ({ ...s, payoutRules: next }))}
   288→          />
   289→        );
   290→      case 5:
   291→        return (
   292→          <RiskRulesForm
   293→            value={state.riskRules}
   294→            onChange={(next) => setState((s) => ({ ...s, riskRules: next }))}
   295→          />
   296→        );
   297→      case 6:
   298→        return (
   299→          <ReviewStep
   300→            state={state}
   301→            selectedType={selectedType}
   302→            hasPhase2={hasPhase2}
   303→            onCreate={createChallenge}
   304→          />
   305→        );
   306→      default:
   307→        return null;
   308→    }
   309→  };
   310→
   311→  return (
   312→    <Page>
   313→      <PageHeader
   314→        title="Create Challenge"
   315→        description="Guided wizard to configure a new evaluation challenge template."
   316→        icon={Sparkles}
   317→      />
   318→      <PageContent>
   319→        {/* Step indicator — done / active / inactive + connecting bars (mirrors OnboardingWizard) */}
   320→        <div className="rounded-lg border bg-card p-4">
   321→          <div className="flex items-center justify-between">
   322→            {STEPS.map((s, i) => {
   323→              const Icon = s.icon;
   324→              const done = i < step;
   325→              const active = i === step;
   326→              return (
   327→                <div key={s.id} className="flex flex-1 flex-col items-center gap-1">
   328→                  <div className="flex w-full items-center">
   329→                    {i > 0 ? (
   330→                      <div className={cn("h-0.5 flex-1", i <= step ? "bg-primary" : "bg-muted")} />
   331→                    ) : null}
   332→                    <div
   333→                      className={cn(
   334→                        "flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 transition-all",
   335→                        done
   336→                          ? "border-primary bg-primary text-primary-foreground"
   337→                          : active
   338→                            ? "border-primary text-primary"
   339→                            : "border-muted text-muted-foreground",
   340→                      )}
   341→                    >
   342→                      {done ? <Check className="h-4 w-4" /> : <Icon className="h-4 w-4" />}
   343→                    </div>
   344→                    {i < STEPS.length - 1 ? (
   345→                      <div className={cn("h-0.5 flex-1", i < step ? "bg-primary" : "bg-muted")} />
   346→                    ) : null}
   347→                  </div>
   348→                  <span
   349→                    className={cn(
   350→                      "text-[10px] font-medium",
   351→                      active ? "text-foreground" : "text-muted-foreground",
   352→                    )}
   353→                  >
   354→                    {s.label}
   355→                  </span>
   356→                </div>
   357→              );
   358→            })}
   359→          </div>
   360→        </div>
   361→
   362→        {/* Step content */}
   363→        <div className="rounded-lg border bg-card p-4 md:p-6">
   364→          <div className="min-h-[320px]">{renderStep()}</div>
   365→
   366→          <Separator className="my-4" />
   367→
   368→          {/* Footer nav — one primary action per step (spec §23) */}
   369→          <div className="flex items-center justify-between">
   370→            <Button variant="ghost" size="sm" disabled={!canBack} onClick={back} className="gap-1">
   371→              <ChevronLeft className="h-4 w-4" /> Back
   372→            </Button>
   373→            <span className="text-xs text-muted-foreground">
   374→              Step {step + 1} of {STEPS.length}
   375→            </span>
   376→            {step < STEPS.length - 1 ? (
   377→              <Button size="sm" disabled={!canNext} onClick={next} className="gap-1">
   378→                Next <ChevronRight className="h-4 w-4" />
   379→              </Button>
   380→            ) : (
   381→              <Button size="sm" onClick={createChallenge} className="gap-1.5">
   382→                <Check className="h-4 w-4" /> Create Challenge
   383→              </Button>
   384→            )}
   385→          </div>
   386→        </div>
   387→      </PageContent>
   388→    </Page>
   389→  );
   390→}

   395→
   396→function phaseConfigFromTemplate(cfg: {
   397→  accountSize: number;
   398→  profitTargetPct: number;
   399→  maxDrawdownPct: number;
   400→  dailyDrawdownPct: number;
   401→  minTradingDays: number;
   402→  maxDays: number;
   403→}): PhaseConfig {
   404→  return {
   405→    accountSize: String(cfg.accountSize),
   406→    profitTargetPct: String(cfg.profitTargetPct),
   407→    maxDrawdownPct: String(cfg.maxDrawdownPct),
   408→    dailyDrawdownPct: String(cfg.dailyDrawdownPct),
   409→    minTradingDays: String(cfg.minTradingDays),
   410→    maxDays: String(cfg.maxDays),
   411→  };
   412→}
   413→
   414→function phaseValid(p: PhaseConfig): boolean {
   415→  return (
   416→    Number(p.accountSize) > 0 &&
   417→    Number(p.profitTargetPct) >= 0 &&
   418→    Number(p.maxDrawdownPct) > 0 &&
   419→    Number(p.dailyDrawdownPct) > 0 &&
   420→    Number(p.minTradingDays) >= 0 &&
   421→    Number(p.maxDays) >= 0
   422→  );
   423→}
   424→
   425→function FieldRow({
   426→  label,
   427→  hint,
   428→  children,
   429→}: {
   430→  label: string;
   431→  hint?: string;
   432→  children: React.ReactNode;
   433→}) {
   434→  return (
   435→    <div className="flex flex-col gap-1.5">
   436→      <Label className="text-xs font-medium text-muted-foreground">{label}</Label>
   437→      {children}
   438→      {hint ? <p className="text-[10px] text-muted-foreground/80">{hint}</p> : null}
   439→    </div>
   440→  );
   441→}
   442→
   443→const REQUIRED_ASTERISK = <span className="text-rose-500" aria-hidden>*</span>;
   444→
   445→function RequiredLabel({ children }: { children: React.ReactNode }) {
   446→  return (
   447→    <Label className="text-xs font-medium text-muted-foreground">
   448→      {children} {REQUIRED_ASTERISK}
   449→    </Label>
   450→  );
   451→}

   453→/* ------------------------------------------------------------------ */

   456→
   457→function StepType({
   458→  types,
   459→  selectedId,
   460→  onSelect,
   461→}: {
   462→  types: ChallengeType[];
   463→  selectedId: string | null;
   464→  onSelect: (id: string) => void;
   465→}) {
   466→  return (
   467→    <div className="space-y-3">
   468→      <div>
   469→        <RequiredLabel>Select challenge type</RequiredLabel>
   470→        <p className="text-xs text-muted-foreground">
   471→          Choose the evaluation structure. Smart defaults will auto-fill the next steps; you can edit
   472→          them later.
   473→        </p>
   474→      </div>
   475→      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   476→        {types.map((t) => {
   477→          const Icon = TYPE_ICONS[t.icon] ?? Target;
   478→          const selected = t.id === selectedId;
   479→          return (
   480→            <button
   481→              key={t.id}
   482→              type="button"
   483→              onClick={() => onSelect(t.id)}
   484→              className={cn(
   485→                "flex flex-col items-start gap-2 rounded-lg border p-4 text-left transition-all hover:shadow-sm",
   486→                selected
   487→                  ? "border-primary bg-primary/5 ring-1 ring-primary/20"
   488→                  : "border-border hover:border-primary/40",
   489→              )}
   490→              aria-pressed={selected}
   491→            >
   492→              <div className="flex items-center gap-2">
   493→                <div className="rounded-md bg-muted p-1.5">
   494→                  <Icon className="h-4 w-4 text-foreground" />
   495→                </div>
   496→                <span className="text-sm font-medium text-foreground">{t.name}</span>
   497→              </div>
   498→              <p className="text-xs text-muted-foreground">{t.description}</p>
   499→              <div className="flex flex-wrap items-center gap-1.5 pt-1">
   500→                <Badge variant="outline" className="text-[9px]">{t.phases} phase{t.phases !== 1 ? "s" : ""}</Badge>
   501→                {t.hasFreeTrial ? <Badge variant="secondary" className="text-[9px]">Free trial</Badge> : null}
   502→                {t.isCompetition ? <Badge variant="secondary" className="text-[9px]">Competition</Badge> : null}
   503→                {!t.active ? <Badge variant="outline" className="text-[9px] text-muted-foreground">Inactive</Badge> : null}
   504→              </div>
   505→              {selected ? (
   506→                <div className="mt-1 inline-flex items-center gap-1 text-[10px] font-medium text-primary">
   507→                  <Check className="h-3 w-3" /> Selected — defaults applied
   508→                </div>
   509→              ) : null}
   510→            </button>
   511→          );
   512→        })}
   513→      </div>
   514→    </div>
   515→  );
   516→}

   522→function PhaseForm({
   523→  title,
   524→  description,
   525→  value,
   526→  onChange,
   527→}: {
   528→  title: string;
   529→  description: string;
   530→  value: PhaseConfig;
   531→  onChange: (next: PhaseConfig) => void;
   532→}) {
   533→  const update = (field: keyof PhaseConfig, v: string) =>
   534→    onChange({ ...value, [field]: v });
   535→
   536→  return (
   537→    <div className="space-y-4">
   538→      <div>
   539→        <h3 className="text-sm font-semibold text-foreground">{title}</h3>

   541→      </div>
   542→      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   543→        <div className="flex flex-col gap-1.5">
   544→          <RequiredLabel>Account size (USD)</RequiredLabel>
   545→          <Input
   546→            type="number"
   547→            value={value.accountSize}
   548→            onChange={(e) => update("accountSize", e.target.value)}
   549→            placeholder="10000"
   550→            min={0}
   551→          />
   552→        </div>

   554→          <RequiredLabel>Profit target (%)</RequiredLabel>

   556→            type="number"
   557→            value={value.profitTargetPct}
   558→            onChange={(e) => update("profitTargetPct", e.target.value)}
   559→            placeholder="8"
   560→            min={0}
   561→            step={0.1}
   562→          />
   563→        </div>
   564→        <div className="flex flex-col gap-1.5">
   565→          <RequiredLabel>Max drawdown (%)</RequiredLabel>
   566→          <Input
   567→            type="number"
   568→            value={value.maxDrawdownPct}
   569→            onChange={(e) => update("maxDrawdownPct", e.target.value)}
   570→            placeholder="10"
   571→            min={0}
   572→            step={0.1}
   573→          />
   574→        </div>
   575→        <div className="flex flex-col gap-1.5">
   576→          <RequiredLabel>Daily drawdown (%)</RequiredLabel>
   577→          <Input
   578→            type="number"
   579→            value={value.dailyDrawdownPct}
   580→            onChange={(e) => update("dailyDrawdownPct", e.target.value)}
   581→            placeholder="5"
   582→            min={0}
   583→            step={0.1}
   584→          />
   585→        </div>
   586→        <FieldRow label="Min trading days" hint="Optional — set 0 to disable.">
   587→          <Input
   588→            type="number"
   589→            value={value.minTradingDays}
   590→            onChange={(e) => update("minTradingDays", e.target.value)}
   591→            placeholder="0"
   592→            min={0}
   593→          />
   594→        </FieldRow>
   595→        <FieldRow label="Max days" hint="Optional — set 0 for unlimited.">
   596→          <Input
   597→            type="number"
   598→            value={value.maxDays}
   599→            onChange={(e) => update("maxDays", e.target.value)}
   600→            placeholder="30"
   601→            min={0}
   602→          />
   603→        </FieldRow>

   698→          <Input
   699→            type="number"
   700→            value={value.profitSplitPct}
   701→            onChange={(e) => onChange({ ...value, profitSplitPct: e.target.value })}
   702→            placeholder="80"
   703→            min={0}
   704→            max={100}
   705→          />

   738→                  "rounded-md border px-3 py-1.5 text-xs font-medium transition-colors",

   742→                )}
   743→                aria-pressed={selected}
   744→              >
   745→                {selected ? <Check className="mr-1 inline h-3 w-3" /> : null}
   746→                {m}
   747→              </button>
   748→            );
   749→          })}
   750→        </div>
   751→      </div>
   752→    </div>
   753→  );

   978→function Switch({ checked, onChange }: { checked: boolean; onChange: (v: boolean) => void }) {
   979→  // Lightweight inline switch (avoids importing a radix switch wrapper just for one row)
   980→  return (
   981→    <button
   982→      type="button"
   983→      role="switch"
   984→      aria-checked={checked}
   985→      onClick={() => onChange(!checked)}
   986→      className={cn(
   987→        "relative inline-flex h-5 w-9 shrink-0 items-center rounded-full border border-transparent transition-colors",
   988→        checked ? "bg-primary" : "bg-input",
   989→      )}
   990→    >
   991→      <span
   992→        className={cn(
   993→          "pointer-events-none block h-4 w-4 rounded-full bg-background shadow transition-transform",
   994→          checked ? "translate-x-4" : "translate-x-0.5",
   995→        )}
   996→      />
   997→    </button>
   998→  );
