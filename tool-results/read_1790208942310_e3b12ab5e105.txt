    24→ * field without modifying the canonical interface in types.ts. This lets

    34→/**
    35→ * Deterministic small-string hash — used in place of `Math.random()` so
    36→ * mock data is stable across reloads (e.g. tenant scoping, default toggles).
    37→ */
    38→export function hashStr(s: string): number {
    39→  let h = 7;
    40→  for (let i = 0; i < s.length; i++) {
    41→    h = (h * 31 + s.charCodeAt(i)) >>> 0;
    42→  }
    43→  return h;
    44→}
    45→
    46→const now = new Date();
    47→const iso = (d: Date) => d.toISOString();
    48→const daysAgo = (n: number) => {
    49→  const d = new Date(now);
    50→  d.setDate(d.getDate() - n);
    51→  return iso(d);
    52→};
    53→const hoursAgo = (n: number) => {
    54→  const d = new Date(now);
    55→  d.setHours(d.getHours() - n);
    56→  return iso(d);
    57→};
