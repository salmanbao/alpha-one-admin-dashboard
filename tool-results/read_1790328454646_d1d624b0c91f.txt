   107→function resolvePhase(id: string | undefined): {
   108→  phase: ChallengePhaseConfig;
   109→  index: number;
   110→} {
   111→  const all = getChallengePhaseConfigs();
   112→  if (!id) return { phase: all[0], index: 0 };
   113→  const numeric = Number(id);
   114→  if (Number.isInteger(numeric) && numeric >= 0 && numeric < all.length) {
   115→    return { phase: all[numeric], index: numeric };
   116→  }
   117→  const byId = all.findIndex((p) => p.id === id);
   118→  if (byId >= 0) return { phase: all[byId], index: byId };
   119→  return { phase: all[0], index: 0 };
   120→}

   159→function generateChangeHistory(phaseId: string, index: number): PhaseChangeEntry[] {
   160→  // Deterministic per phase — use index as seed, no Math.random
   161→  const count = 8 + (index % 3); // 8, 9, or 10 entries
   162→  const out: PhaseChangeEntry[] = [];
   163→  const baseTs = new Date("2026-01-15T10:30:00.000Z").getTime();
   164→  for (let i = 0; i < count; i++) {
   165→    // Per-phase offset to keep entries distinct across phases
   166→    const t = new Date(baseTs - i * 36 * 3600 * 1000 - index * 3600 * 1000);
   167→    const tpl = FIELD_TEMPLATES[(i + index) % FIELD_TEMPLATES.length];
   168→    out.push({
   169→      id: `${phaseId}-ch-${i}`,
   170→      timestamp: t.toISOString(),
   171→      user: ACTORS[(i + index) % ACTORS.length],
   172→      field: tpl.field,
   173→      oldValue: tpl.oldV,
   174→      newValue: tpl.newV,
   175→      reason: REASONS[(i + index) % REASONS.length],
   176→    });
   177→  }
   178→  return out;
   179→}

   181→function generatePlatformIds(phase: ChallengePhaseConfig, index: number) {
   182→  // Deterministic mock per phase index
   183→  const slug =
   184→    phase.phaseName.toLowerCase().replace(/\s+/g, "") ?? `phase${index}`;
   185→  return {
   186→    mt5Group: `funderblu\\${slug}`,
   187→    mt4Group: index % 2 === 0 ? "" : `funderblu\\${slug}-mt4`,
   188→    dxTradeGroup: index % 3 === 0 ? `dx\\${slug}` : "",
   189→    bridgeStatus: (index % 4 === 0 ? "disconnected" : "connected") as
   190→      | "connected"
   191→      | "disconnected",
   192→    lastSync: new Date(Date.now() - (index + 1) * 1800 * 1000).toISOString(),
   193→  };
   194→}

   786→
   787→function ChangeHistoryTab({
   788→  phase,
   789→  index,
   790→  navigate,
   791→}: {
   792→  phase: ChallengePhaseConfig;
   793→  index: number;
   794→  navigate: (view: string, params?: Record<string, string>) => void;
   795→}) {
   796→  const history = useMemo(
   797→    () => generateChangeHistory(phase.id, index),
   798→    [phase.id, index],
   799→  );
   800→
   801→  // Local filter state
   802→  const [search, setSearch] = useState("");
   803→  const [fieldFilter, setFieldFilter] = useState<string>("all");
   804→
   805→  const fieldOptions = useMemo(() => {
   806→    const set = new Set<string>();
   807→    history.forEach((h) => set.add(h.field));
   808→    return Array.from(set).sort();
   809→  }, [history]);
   810→
   811→  const filtered = useMemo(() => {
   812→    const q = search.trim().toLowerCase();
   813→    return history.filter((h) => {
   814→      if (fieldFilter !== "all" && h.field !== fieldFilter) return false;
   815→      if (!q) return true;
   816→      return (
   817→        h.user.toLowerCase().includes(q) ||
   818→        h.field.toLowerCase().includes(q) ||
   819→        h.oldValue.toLowerCase().includes(q) ||
   820→        h.newValue.toLowerCase().includes(q) ||
   821→        h.reason.toLowerCase().includes(q)
   822→      );
   823→    });
   824→  }, [history, search, fieldFilter]);
   825→
   826→  const exportCsv = () => {
   827→    const header = ["Date/Time", "User", "Field Changed", "Old Value", "New Value", "Reason"];
   828→    const rows = filtered.map((h) => [
   829→      new Date(h.timestamp).toISOString(),
   830→      h.user,
   831→      h.field,
   832→      h.oldValue,
   833→      h.newValue,
   834→      h.reason,
   835→    ]);
   836→    const csv = [header, ...rows]
   837→      .map((r) => r.map((c) => `"${String(c).replace(/"/g, '""')}"`).join(","))
   838→      .join("\n");
   839→    const blob = new Blob([csv], { type: "text/csv;charset=utf-8" });
   840→    const url = URL.createObjectURL(blob);
   841→    const a = document.createElement("a");
   842→    a.href = url;
   843→    a.download = `${phase.id}-change-history.csv`;
   844→    document.body.appendChild(a);
   845→    a.click();
   846→    document.body.removeChild(a);
   847→    URL.revokeObjectURL(url);
   848→    toast({
   849→      title: "CSV exported",
   850→      description: `${filtered.length} change record${filtered.length === 1 ? "" : "s"} exported.`,
   851→    });
   852→  };
   853→
   854→  const columns: Column<PhaseChangeEntry>[] = [
   855→    {
   856→      key: "timestamp",
   857→      header: "Date / Time",
   858→      cell: (h) => (
   859→        <span className="font-mono text-xs text-muted-foreground">
   860→          {new Date(h.timestamp).toLocaleString()}
   861→        </span>
   862→      ),
   863→      sortValue: (h) => h.timestamp,
   864→    },
   865→    {
   866→      key: "user",
   867→      header: "User",
   868→      cell: (h) => (
   869→        <div className="flex items-center gap-1.5">
   870→          <span className="text-foreground">{h.user}</span>
   871→          {h.user === "System" || h.user === "AI Engine" ? (
   872→            <Badge variant="outline" className="text-[10px] text-muted-foreground">
   873→              Automated
   874→            </Badge>
   875→          ) : null}
   876→        </div>
   877→      ),
   878→      sortValue: (h) => h.user,
   879→    },
   880→    {
   881→      key: "field",
   882→      header: "Field Changed",
   883→      cell: (h) => <span className="font-medium text-foreground">{h.field}</span>,
   884→      sortValue: (h) => h.field,
   885→    },
   886→    {
   887→      key: "oldValue",
   888→      header: "Old Value",
   889→      cell: (h) => (
   890→        <span className="text-rose-700 line-through dark:text-rose-400">
   891→          {h.oldValue}
   892→        </span>
   893→      ),
   894→    },
   895→    {
   896→      key: "newValue",
   897→      header: "New Value",
   898→      cell: (h) => (
   899→        <span className="font-medium text-emerald-700 dark:text-emerald-400">
   900→          {h.newValue}
   901→        </span>
   902→      ),
   903→    },
   904→    {
   905→      key: "reason",
   906→      header: "Reason",
   907→      cell: (h) => <span className="text-xs text-muted-foreground">{h.reason}</span>,
   908→      sortValue: (h) => h.reason,
   909→    },
   910→  ];
   911→
   912→  return (
   913→    <div className="flex flex-col gap-4">
   914→      <div className="flex flex-col items-start justify-between gap-3 md:flex-row md:items-center">
   915→        <div>
   916→          <h2 className="text-sm font-semibold text-foreground">
   917→            Change History for {phase.phaseName}
   918→          </h2>
   919→          <p className="mt-0.5 text-xs text-muted-foreground">
   920→            Per-object audit trail (§28) — every change made to this phase
   921→            configuration. Strikethrough red shows the prior value, green shows
   922→            the value it was changed to.
   923→          </p>
   924→        </div>
   925→        <div className="flex flex-wrap items-center gap-2">
   926→          <Button
   927→            size="sm"
   928→            variant="outline"
   929→            onClick={() => navigate("challenge-edit", { id: "" })}
   930→          >
   931→            <ArrowLeft className="mr-1 h-3.5 w-3.5" /> Back to Challenge
   932→          </Button>
   933→          <Button
   934→            size="sm"
   935→            variant="outline"
   936→            onClick={exportCsv}
   937→            disabled={filtered.length === 0}
   938→          >
   939→            <Download className="mr-1 h-3.5 w-3.5" /> Export CSV
   940→          </Button>
   941→        </div>
   942→      </div>
   943→
   944→      <div className="flex flex-col gap-2 md:flex-row md:items-center md:justify-between">
   945→        <div className="relative w-full md:max-w-xs">
   946→          <Search className="absolute left-2.5 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   947→          <Input
   948→            value={search}
   949→            onChange={(e) => setSearch(e.target.value)}
   950→            placeholder="Search by user, field, value…"
   951→            className="pl-8"
   952→          />
   953→        </div>
   954→        <div className="flex items-center gap-2">
   955→          <span className="text-xs text-muted-foreground">Filter by field</span>
   956→          <Select value={fieldFilter} onValueChange={setFieldFilter}>
   957→            <SelectTrigger size="sm" className="h-8 w-56 text-xs">
   958→              <SelectValue placeholder="All fields" />
   959→            </SelectTrigger>
   960→            <SelectContent>
   961→              <SelectItem value="all">All fields</SelectItem>
   962→              {fieldOptions.map((f) => (
   963→                <SelectItem key={f} value={f}>
   964→                  {f}
   965→                </SelectItem>
   966→              ))}
   967→            </SelectContent>
   968→          </Select>
   969→        </div>
   970→      </div>
   971→
   972→      <DataTable
   973→        columns={columns}
   974→        data={filtered}
   975→        rowKey={(h) => h.id}
   976→        pageSize={10}
   977→        emptyTitle="No change records"
   978→        emptyDescription="No changes match the current search or filter."
   979→      />
   980→    </div>
   981→  );
   982→}
