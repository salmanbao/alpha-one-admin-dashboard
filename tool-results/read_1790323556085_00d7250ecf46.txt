   254→const STATUS_TONE: Record<InvoiceStatus, "muted" | "info" | "success" | "danger"> = {
   255→  draft: "muted",
   256→  sent: "info",
   257→  paid: "success",
   258→  overdue: "danger",
   259→  cancelled: "muted",
   260→};
   261→
   262→function invoiceStatusTone(s: InvoiceStatus) {
   263→  return STATUS_TONE[s];
   264→}
   265→
   266→function fmtDate(iso: string): string {
   267→  const d = new Date(iso);
   268→  if (Number.isNaN(d.getTime())) return iso;
   269→  return d.toLocaleDateString("en-US", { year: "numeric", month: "short", day: "numeric" });
   270→}
   271→
   272→function daysBetween(a: string, b: string): number {
   273→  const d1 = new Date(a).getTime();
   274→  const d2 = new Date(b).getTime();
   275→  if (Number.isNaN(d1) || Number.isNaN(d2)) return 0;
   276→  return Math.round((d2 - d1) / (1000 * 60 * 60 * 24));
   277→}

   317→
   318→  // ----- Derived KPIs (deterministic — pure reduce over constant array) -----
   319→  const kpis = useMemo(() => {
   320→    const outstanding = INVOICES.filter(
   321→      (i) => i.status === "sent" || i.status === "overdue",
   322→    ).reduce((s, i) => s + i.total, 0);
   323→    const paidThisMonth = INVOICES.filter((i) => i.status === "paid").reduce(
   324→      (s, i) => s + i.total,
   325→      0,
   326→    );
   327→    const overdue = INVOICES.filter((i) => i.status === "overdue");
   328→    const avgDaysToPay = (() => {
   329→      const paid = INVOICES.filter((i) => i.status === "paid");
   330→      if (!paid.length) return 0;
   331→      const total = paid.reduce((s, i) => s + daysBetween(i.issueDate, i.dueDate), 0);
   332→      return Math.round(total / paid.length);
   333→    })();
   334→    return { outstanding, paidThisMonth, overdueCount: overdue.length, avgDaysToPay };
   335→  }, []);
   336→
   337→  // ----- Filtered rows (status + date range + DataTable's own search) -----
   338→  const filtered = useMemo(() => {
   339→    const now = Date.now();
   340→    const rangeMs = dateRange === "all" ? null : Number(dateRange) * 86_400_000;
   341→    return INVOICES.filter((i) => {
   342→      if (statusFilter !== "all" && i.status !== statusFilter) return false;
   343→      if (rangeMs !== null) {
   344→        const issued = new Date(i.issueDate).getTime();
   345→        if (Number.isNaN(issued) || now - issued > rangeMs) return false;
   346→      }
   347→      return true;
   348→    });
   349→  }, [statusFilter, dateRange]);

   387→      });
   388→      return;
   389→    }
   390→    toast({
   391→      title: "Draft saved",
   392→      description: `New invoice draft for ${draftTrader} saved (demo).`,
   393→    });
   394→    setCreating(false);
   395→    setDraftTrader("");
   396→    setDraftEmail("");
   397→    setDraftNotes("");
   398→  }
