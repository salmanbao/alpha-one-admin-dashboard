   213→/** Simple deterministic 32-bit string hash for index-seeded data. */
   214→function hashStr(s: string): number {
   215→  let n = 5381;
   216→  for (let i = 0; i < s.length; i++) {
   217→    n = ((n << 5) + n + s.charCodeAt(i)) >>> 0;
   218→  }
   219→  return n;
   220→}

   222→/** Deterministic IP address. */
   223→function ipFor(i: number): string {
   224→  const a = 10 + ((i * 13) % 240);
   225→  const b = 1 + ((i * 17) % 254);
   226→  const c = 1 + ((i * 23) % 254);
   227→  const d = (i * 31) % 256;
   228→  return `${a}.${b}.${c}.${d}`;
   229→}

   284→/** Build the rich-text description per event type with metric snapshots. */
   285→function buildDescription(
   286→  eventType: EnhancedEventType,
   287→  i: number,
   288→  accountId: string,
   289→): string {
   290→  // Deterministic money/percent values derived from i — always rendered with
   291→  // fixed 2-decimal precision for visual consistency.
   292→  const equity = (4500 + (i * 13) % 1500).toFixed(2);
   293→  const balance = (4900 + (i * 7) % 1500).toFixed(2);
   294→  const openPnl = -(50 + (i * 3) % 100).toFixed(2);
   295→  const positions = 132413231 + i;
   296→  const dailyLimit = "250.00";
   297→  const dailyCurrent = (280 + (i * 5) % 100).toFixed(2);
   298→  const dailyUsedPct = Math.round((parseFloat(dailyCurrent) / 250) * 100);
   299→  const target = "800.00";
   300→  const targetCurrent = (820 + (i * 7) % 100).toFixed(2);
   301→  const targetPct = Math.round((parseFloat(targetCurrent) / 800) * 100);
   302→  const oldAccountId = `33338${50000 + (i * 11) % 40000}`;
   303→  const newAccountId = accountId;
   304→  const payoutAmount = (750 + (i * 11) % 200).toFixed(2);
   305→  const veriffRef = `veriff-${hashStr(`${i}-veriff`).toString(16).slice(0, 8)}`;
   306→  const ticket = 132413231 + i;
   307→  const symbol = ["EURUSD", "GBPUSD", "XAUUSD", "BTCUSD", "USDJPY"][i % 5];
   308→  const lots = (0.1 + (i % 5) * 0.5).toFixed(1);
   309→  const price = (1.05 + (i % 50) * 0.001).toFixed(4);
   310→  const device = ["Chrome on macOS", "Safari on iPhone", "Edge on Windows", "Firefox on Linux"][i % 4];
   311→  const plan = ["1-Step Evaluation", "2-Step Gen Z", "Standard 100K"][i % 3];
   312→  const startBalance = (5000 + (i % 5) * 5000).toLocaleString("en-US");
   313→
   314→  switch (eventType) {
   315→    case "ACCOUNT_CREATED":
   316→      return `Account registered via signup flow. Initial balance: $${startBalance}. Plan: ${plan}.`;
   317→    case "KYC_COMPLETED":
   318→      return `KYC verification completed via Veriff with manual admin intervention.`;
   319→    case "KYC_REJECTED":
   320→      return `KYC rejected: document failed liveness check. Veriff reference: ${veriffRef}.`;
   321→    case "ORDER_CREATED":
   322→      return `Buy order placed on ${symbol}, ${lots} lots @ ${price}. Ticket: ${ticket}.`;
   323→    case "PAYOUT_REQUESTED":
   324→      return `Payout of $${payoutAmount} requested via bank transfer. Reference: wd-${accountId}.`;
   325→    case "PAYOUT_COMPLETED":
   326→      return `Payout of $${payoutAmount} sent via bank transfer. Reference: wd-${accountId}.`;
   327→    case "CHALLENGE_STARTED":
   328→      return `Challenge ${plan} started with $${startBalance} starting balance.`;
   329→    case "CHALLENGE_PASSED":
   330→      return `Challenge passed: profit target of $${target} reached without breaching risk limits.`;
   331→    case "CHALLENGE_FAILED":
   332→      return `Challenge failed: max drawdown breached before reaching profit target. Final PnL: -$${dailyCurrent}.`;
   333→    case "BREACH_DETECTED":
   334→      return `Breach detected: max drawdown reached 10.1% of starting balance. Account disabled.`;
   335→    case "FLOATING_PNL_BREACHED":
   336→      return `Floating PnL breached: ${accountId} Equity: $${equity}; Balance: $${balance}; Open PnL: $${openPnl}; Positions: ${positions}...`;
   337→    case "DAILY_DRAWDOWN_BREACHED":
   338→      return `Daily Drawdown breached: ${accountId} Daily limit: $${dailyLimit}; Current: $${dailyCurrent}; Used: ${dailyUsedPct}%`;
   339→    case "TARGET_PROFIT_REACHED":
   340→      return `Target Profit reached: ${accountId} Target: $${target}; Current: $${targetCurrent}; Progress: ${targetPct}%`;
   341→    case "PHASE_UPGRADED":
   342→      return `Account: ${oldAccountId} upgraded from Phase 2 to 3. New account: ${newAccountId}`;
   343→    case "LOGIN":
   344→      return `User logged in from ${ipFor(i)} (${device}).`;
   345→    case "PASSWORD_CHANGED":
   346→      return `Password changed via account settings. MFA still active.`;
   347→    default:
   348→      return `Event of type ${eventType}.`;
   349→  }
   350→}

   352→/** Build 168 deterministic events covering all 16 types (each ≥ 8 reps). */
   353→function buildEnhancedEvents(traderPool: Trader[]): EnhancedUserEvent[] {
   354→  const pool = traderPool.length > 0 ? traderPool : allTraders;
   355→  const totalEvents = 168;
   356→  const out: EnhancedUserEvent[] = [];
   357→  for (let i = 0; i < totalEvents; i++) {
   358→    const trader = pool[i % pool.length];
   359→    const eventType = ALL_EVENT_TYPES[i % ALL_EVENT_TYPES.length];
   360→    const accountId = accountIdFor(i);
   361→    const phaseType = phaseTypeFor(i);
   362→    const challengeName = challengeNameFor(i);
   363→    out.push({
   364→      id: `eue-${i + 1}`,
   365→      timestamp: new Date(
   366→        Date.now() - i * 90 * 60 * 1000, // every 90 minutes apart, newest first
   367→      ).toISOString(),
   368→      traderId: trader.id,
   369→      userEmail: trader.email,
   370→      phaseType,
   371→      challengeName,
   372→      accountId,
   373→      eventType,
   374→      description: buildDescription(eventType, i, accountId),
   375→      ipAddress: ipFor(i),
   376→      source: sourceFor(eventType, i),
   377→    });
   378→  }
   379→  return out;
   380→}
