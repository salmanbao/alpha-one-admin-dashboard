   257→function emailStatusTone(status: EmailStatus): "default" | "success" | "warning" | "danger" | "info" | "muted" {
   258→  switch (status) {
   259→    case "draft": return "muted";
   260→    case "scheduled": return "info";
   261→    case "sending": return "warning";
   262→    case "sent": return "success";
   263→    case "completed": return "success";
   264→    case "paused": return "warning";
   265→    case "failed": return "danger";
   266→    default: return "muted";
   267→  }
   268→}

   943→function audienceFor(audience: string | undefined, defaultLabel: string): string {
   944→  if (!audience) return "all";
   945→  if (audience.startsWith("All")) return "all";
   946→  if (audience.startsWith("Active")) return "active";
   947→  if (audience.startsWith("Funded")) return "funded";
   948→  if (audience.startsWith("Failed")) return "failed";
   949→  if (audience.startsWith("Inactive") || audience.startsWith("Phase")) return "segment";
   950→  return "all";
   951→}

   953→function estimatedReach(audience: string): number {
   954→  switch (audience) {
   955→    case "all": return 8420;
   956→    case "active": return 3840;
   957→    case "funded": return 642;
   958→    case "failed": return 1210;
   959→    case "segment": return 0;
   960→    default: return 0;
   961→  }
   962→}
