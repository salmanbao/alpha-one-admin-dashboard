   133→/* ------------------------------------------------------------------ */
   134→
   135→interface DualListProps {
   136→  availableLabel: string;
   137→  selectedLabel: string;
   138→  options: readonly string[];
   139→  selected: string[];
   140→  onChange: (next: string[]) => void;
   141→  availableIcon?: React.ComponentType<{ className?: string }>;
   142→}
   143→
   144→function DualListBox({
   145→  availableLabel,
   146→  selectedLabel,
   147→  options,
   148→  selected,
   149→  onChange,
   150→  availableIcon: AvailableIcon,
   151→}: DualListProps) {
   152→  const available = options.filter((o) => !selected.includes(o));
   153→
   154→  const add = (value: string) => {
   155→    if (!selected.includes(value)) onChange([...selected, value]);
   156→  };
   157→  const remove = (value: string) => {
   158→    onChange(selected.filter((v) => v !== value));
   159→  };
   160→  const addAll = () => onChange([...selected, ...available]);
   161→  const removeAll = () => onChange([]);
   162→
   163→  return (
   164→    <div className="grid grid-cols-1 gap-3 md:grid-cols-[1fr_auto_1fr]">
   165→      {/* Available */}
   166→      <div className="rounded-md border bg-background">
   167→        <div className="flex items-center justify-between gap-2 border-b bg-muted/30 px-3 py-2">
   168→          <span className="flex items-center gap-1.5 text-xs font-medium uppercase tracking-wide text-muted-foreground">
   169→            {AvailableIcon ? <AvailableIcon className="h-3 w-3" /> : null}
   170→            {availableLabel}
   171→          </span>
   172→          <Badge variant="outline" className="text-[10px]">{available.length}</Badge>
   173→        </div>
   174→        <div className="max-h-60 overflow-auto p-1">
   175→          {available.length === 0 ? (
   176→            <p className="px-2 py-3 text-xs text-muted-foreground">No items available.</p>
   177→          ) : (
   178→            available.map((value) => (
   179→              <label
   180→                key={value}
   181→                className="flex items-center gap-2 rounded px-2 py-1.5 text-sm hover:bg-muted/40"
   182→              >
   183→                <input
   184→                  type="checkbox"
   185→                  checked={false}
   186→                  onChange={() => add(value)}
   187→                  className="h-3.5 w-3.5 accent-emerald-600"
   188→                />
   189→                <span className="text-foreground">{value}</span>
   190→              </label>
   191→            ))
   192→          )}
   193→        </div>
   194→      </div>
   195→
   196→      {/* Controls */}
   197→      <div className="flex flex-row items-center justify-center gap-2 md:flex-col">
   198→        <Button
   199→          type="button"
   200→          size="sm"
   201→          variant="outline"
   202→          onClick={addAll}
   203→          disabled={available.length === 0}
   204→          aria-label="Add all"
   205→        >
   206→          <ChevronRight className="h-3.5 w-3.5" />
   207→          <span className="sr-only md:not-sr-only md:ml-1">Add all</span>
   208→        </Button>
   209→        <Button
   210→          type="button"
   211→          size="sm"
   212→          variant="outline"
   213→          onClick={removeAll}
   214→          disabled={selected.length === 0}
   215→          aria-label="Remove all"
   216→        >
   217→          <ChevronLeft className="h-3.5 w-3.5" />
   218→          <span className="sr-only md:not-sr-only md:ml-1">Remove all</span>
   219→        </Button>
   220→      </div>
   221→
   222→      {/* Selected */}
   223→      <div className="rounded-md border bg-background">
   224→        <div className="flex items-center justify-between gap-2 border-b bg-muted/30 px-3 py-2">
   225→          <span className="flex items-center gap-1.5 text-xs font-medium uppercase tracking-wide text-muted-foreground">
   226→            <Filter className="h-3 w-3" />
   227→            {selectedLabel}
   228→          </span>
   229→          <Badge variant="outline" className="text-[10px]">{selected.length}</Badge>
   230→        </div>
   231→        <div className="max-h-60 overflow-auto p-1">
   232→          {selected.length === 0 ? (
   233→            <p className="px-2 py-3 text-xs text-muted-foreground">
   234→              No selection — applies to all.
   235→            </p>
   236→          ) : (
   237→            selected.map((value) => (
   238→              <label
   239→                key={value}
   240→                className="flex items-center gap-2 rounded px-2 py-1.5 text-sm hover:bg-muted/40"
   241→              >
   242→                <input
   243→                  type="checkbox"
   244→                  checked
   245→                  onChange={() => remove(value)}
   246→                  className="h-3.5 w-3.5 accent-rose-500"
   247→                />
   248→                <span className="text-foreground">{value}</span>
   249→              </label>
   250→            ))
   251→          )}
   252→        </div>
   253→      </div>
   254→    </div>
   255→  );
   256→}

   285→          ) : null}
   286→        </div>
   287→      </div>
   288→      {children}
   289→    </section>
   290→  );

   328→  const [segment, setSegment] = useState<SegmentRules>(EMPTY_SEGMENT);

   607→              {selectedCountries.map((c) => (
   608→                <Badge key={c} variant="outline" className="text-[10px]">{c}</Badge>
   609→              ))}
   610→            </div>
   611→          )}
   612→        </SectionCard>
   613→
   614→        {/* Challenge Targeting */}
   615→        <SectionCard
   616→          title="Challenge Targeting"
   617→          description="Restrict this offer to specific challenge types. Empty selection applies to all challenges."
   618→          icon={Layers}
   619→        >
   620→          <DualListBox
   621→            availableLabel="Available Challenges"
   622→            selectedLabel="Selected Challenges"
   623→            options={ALL_CHALLENGES}
   624→            selected={selectedChallenges}
   625→            onChange={setSelectedChallenges}
   626→            availableIcon={Layers}
   627→          />
   628→        </SectionCard>
   629→
   630→        {/* User Segment — collapsible (§12 progressive disclosure) */}
   631→        <Collapsible open={segmentOpen} onOpenChange={setSegmentOpen}>
   632→          <section className="rounded-lg border bg-card p-4">
   633→            <CollapsibleTrigger asChild>
   634→              <button
   635→                type="button"
   636→                className="flex w-full items-center justify-between gap-2 text-left"
   637→              >
   638→                <div className="flex items-start gap-2">
   639→                  <div className="mt-0.5 rounded-md bg-muted/60 p-1.5">
   640→                    <Settings2 className="h-4 w-4 text-foreground" />
   641→                  </div>
   642→                  <div>
   643→                    <h2 className="text-sm font-semibold text-foreground">
   644→                      User Segment
   645→                    </h2>
   646→                    <p className="mt-0.5 text-xs text-muted-foreground">
   647→                      Optional advanced targeting — narrow the audience by trader
   648→                      attributes and account properties.
   649→                    </p>
   650→                  </div>
   651→                </div>
   652→                {segmentOpen ? (
   653→                  <ChevronDown className="h-4 w-4 text-muted-foreground" />
   654→                ) : (
   655→                  <ChevronRight className="h-4 w-4 text-muted-foreground" />
   656→                )}
   657→              </button>
   658→            </CollapsibleTrigger>
   659→            <CollapsibleContent>
   660→              <div className="mt-4 space-y-4">
   661→                <div className="grid gap-4 md:grid-cols-2">
   662→                  <div className="space-y-1.5">
   663→                    <Label htmlFor="seg-name">Segment Name</Label>
   664→                    <Input
   665→                      id="seg-name"
   666→                      value={segment.name}
   667→                      onChange={(e) => setSegmentField("name", e.target.value)}
   668→                      placeholder="e.g. New funded traders — US/EU"
   669→                    />
   670→                  </div>
   671→                  <div className="flex items-center justify-between rounded-md border bg-muted/30 px-3 py-2">
   672→                    <div className="flex flex-col">
   673→                      <Label htmlFor="seg-active" className="cursor-pointer text-sm font-medium">
   674→                        Segment Active
   675→                      </Label>
   676→                      <span className="text-xs text-muted-foreground">
   677→                        Toggle off to disable this segment temporarily.
   678→                      </span>
   679→                    </div>
   680→                    <Switch
   681→                      id="seg-active"
   682→                      checked={segment.active}
   683→                      onCheckedChange={(v) => setSegmentField("active", v)}
   684→                    />
   685→                  </div>
   686→                </div>
   687→
   688→                <Separator />
   689→
   690→                <div className="grid gap-3 md:grid-cols-2">
   691→                  <TriStateSelect
   692→                    label="Account Purchased"
   693→                    help="Restrict to traders who have or have not purchased a challenge."
   694→                    value={segment.accountPurchased}
   695→                    onChange={(v) => setSegmentField("accountPurchased", v)}
   696→                  />
   697→                  <TriStateSelect
   698→                    label="Competition User"
   699→                    help="Restrict to traders who have or have not participated in a competition."
   700→                    value={segment.competitionUser}
   701→                    onChange={(v) => setSegmentField("competitionUser", v)}
   702→                  />
   703→                  <TriStateSelect
   704→                    label="Has Approved Payout"
   705→                    help="Restrict to traders who have or have not received an approved payout."
   706→                    value={segment.hasApprovedPayout}
   707→                    onChange={(v) => setSegmentField("hasApprovedPayout", v)}
   708→                  />
   709→                  <TriStateSelect
   710→                    label="Fund Accounts Only"
   711→                    help="Restrict to traders currently holding a funded account."
   712→                    value={segment.fundAccountsOnly}
   713→                    onChange={(v) => setSegmentField("fundAccountsOnly", v)}
   714→                  />
   715→                  <TriStateSelect
   716→                    label="Has Failed Accounts"
   717→                    help="Restrict to traders who have or have not failed any challenge."
   718→                    value={segment.hasFailedAccounts}
   719→                    onChange={(v) => setSegmentField("hasFailedAccounts", v)}
   720→                  />
   721→                  <div className="flex flex-col gap-1.5">
   722→                    <LabelWithHelp help="Minimum account size in USD the trader must hold to qualify. Blank = no minimum.">
   723→                      Account Size Min (USD)
   724→                    </LabelWithHelp>
   725→                    <Input
   726→                      type="number"
   727→                      min={0}
   728→                      value={segment.accountSizeMin}
   729→                      onChange={(e) => setSegmentField("accountSizeMin", e.target.value)}
   730→                      placeholder="e.g. 5000"
   731→                    />
   732→                  </div>
   733→                  <div className="flex flex-col gap-1.5">
   734→                    <LabelWithHelp help="Maximum account size in USD the trader may hold to qualify. Blank = no maximum.">
   735→                      Account Size Max (USD)
   736→                    </LabelWithHelp>
   737→                    <Input
   738→                      type="number"
   739→                      min={0}
   740→                      value={segment.accountSizeMax}
   741→                      onChange={(e) => setSegmentField("accountSizeMax", e.target.value)}
   742→                      placeholder="e.g. 100000"
   743→                    />
   744→                  </div>
   745→                </div>
   746→
   747→                <div className="rounded-md border bg-muted/20 p-3 text-xs text-muted-foreground">
   748→                  <span className="font-medium text-foreground">Segment preview:</span>{" "}
   749→                  {segment.name || "Untitled segment"} —{" "}
   750→                  <StatusBadge tone={segment.active ? "success" : "muted"}>
   751→                    {segment.active ? "Active" : "Inactive"}
   752→                  </StatusBadge>
   753→                </div>
   754→              </div>
   755→            </CollapsibleContent>
   756→          </section>
   757→        </Collapsible>
   758→
   759→        {/* Action bar — one primary action (§13) */}
   760→        <div className="flex flex-wrap items-center justify-between gap-2 rounded-lg border bg-card p-3">
   761→          <div className="text-xs text-muted-foreground">
   762→            {isNew ? (
   763→              <span className="flex items-center gap-1">
   764→                <Calendar className="h-3 w-3" /> Creating a new offer.
   765→              </span>
   766→            ) : (
   767→              <span className="flex items-center gap-1">
   768→                <Calendar className="h-3 w-3" /> Last updated{" "}
   769→                {existing?.createdAt
   770→                  ? new Date(existing.createdAt).toLocaleDateString()
   771→                  : "—"}
   772→                .
   773→              </span>
   774→            )}
   775→          </div>
   776→          <div className="flex flex-wrap items-center gap-2">
   777→            {!isNew ? (
   778→              <AlertDialog open={deleteOpen} onOpenChange={setDeleteOpen}>
   779→                <AlertDialogTrigger asChild>
   780→                  <Button size="sm" variant="destructive">
   781→                    <Trash2 className="mr-1 h-3.5 w-3.5" /> Delete
   782→                  </Button>
   783→                </AlertDialogTrigger>
   784→                <AlertDialogContent>
   785→                  <AlertDialogHeader>
   786→                    <AlertDialogTitle>Delete this offer?</AlertDialogTitle>
   787→                    <AlertDialogDescription>
   788→                      This will permanently remove <span className="font-medium text-foreground">{title || "this offer"}</span> and stop all
   789→                      coupon redemptions using{" "}
   790→                      <span className="font-mono">{couponCode || "(no coupon)"}</span>.
   791→                      Traders who already redeemed the coupon will keep their discount.
   792→                      This action cannot be undone.
   793→                    </AlertDialogDescription>
   794→                  </AlertDialogHeader>
   795→                  <AlertDialogFooter>
   796→                    <AlertDialogCancel>Cancel</AlertDialogCancel>
   797→                    <AlertDialogAction
   798→                      className="bg-rose-600 hover:bg-rose-700"
   799→                      onClick={onDelete}
   800→                    >
   801→                      Delete offer
   802→                    </AlertDialogAction>
   803→                  </AlertDialogFooter>
   804→                </AlertDialogContent>
   805→              </AlertDialog>
   806→            ) : null}
   807→            <Button
   808→              size="sm"
   809→              variant="outline"
   810→              onClick={onSaveAndAdd}
   811→            >
   812→              <Plus className="mr-1 h-3.5 w-3.5" /> Save and add another
   813→            </Button>
   814→            <Button
   815→              size="sm"
   816→              variant="outline"
   817→              onClick={onSaveAndContinue}
   818→            >
   819→              <Save className="mr-1 h-3.5 w-3.5" /> Save and continue editing
   820→            </Button>
   821→            <Button size="sm" onClick={onSave}>
   822→              <Save className="mr-1 h-3.5 w-3.5" /> Save
   823→            </Button>
   824→          </div>
   825→        </div>
   826→      </PageContent>
   827→    </Page>
   828→  );
   829→}

   835→function TriStateSelect({
   836→  label,
   837→  help,
   838→  value,
   839→  onChange,
   840→}: {
   841→  label: string;
   842→  help: string;
   843→  value: TriState;
   844→  onChange: (v: TriState) => void;

   847→    <div className="flex flex-col gap-1.5">
   848→      <LabelWithHelp help={help}>{label}</LabelWithHelp>
   849→      <Select value={value} onValueChange={(v) => onChange(v as TriState)}>
   850→        <SelectTrigger className="w-full">
   851→          <SelectValue />
   852→        </SelectTrigger>
   853→        <SelectContent>
   854→          {TRI_STATE.map((s) => (
   855→            <SelectItem key={s} value={s}>
   856→              {triStateLabel(s)}
   857→            </SelectItem>
   858→          ))}
   859→        </SelectContent>
   860→      </Select>
   861→    </div>
   862→  );
   863→}
