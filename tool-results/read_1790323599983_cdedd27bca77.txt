   153→/** Deterministic pseudo-random generator from a numeric seed. */
   154→function seededRandom(seed: number): number {
   155→  const x = Math.sin(seed * 9999.1) * 10000;
   156→  return x - Math.floor(x);
   157→}
   158→
   159→/** Hash a string id to a stable integer seed. */
   160→function hashSeed(id: string): number {
   161→  let h = 0;
   162→  for (let i = 0; i < id.length; i++) {
   163→    h = (h * 31 + id.charCodeAt(i)) | 0;
   164→  }
   165→  return Math.abs(h) || 1;
   166→}

   168→/**
   169→ * Generate a single deterministic closed position from an id. The shape
   170→ * mirrors `generateClosedPositions` on the list page so opening a row
   171→ * there would show matching data.
   172→ */
   173→function generateClosedPositionDetail(
   174→  id: string,
   175→  tenantId: string,
   176→): ClosedPositionDetail {
   177→  const accounts = getTenantAccounts(tenantId);
   178→  const traders = getTenantTraders(tenantId);
   179→  const openPositions = getTenantPositions(tenantId);
   180→
   181→  const seed = hashSeed(id);
   182→  const idx = seed % Math.max(openPositions.length, 1);
   183→  const base = openPositions[idx] ?? openPositions[0];
   184→
   185→  const [sym, basePrice, mult, description] =
   186→    BASE_SYMBOLS[(seed - 1) % BASE_SYMBOLS.length];
   187→
   188→  const r1 = seededRandom(seed + 0.1);
   189→  const r2 = seededRandom(seed + 0.5);
   190→  const r3 = seededRandom(seed + 1.7);
   191→  const r4 = seededRandom(seed + 3.3);
   192→
   193→  const side: ClosedSide = base.side;
   194→  const entry = base.entryPrice;
   195→  const movePct = (r1 - 0.5) * 0.04; // ±2%
   196→  const close = entry * (1 + movePct);
   197→  const pnlPerUnit = side === "buy" ? close - entry : entry - close;
   198→  const volume = base.volume;
   199→  const grossPnl = pnlPerUnit * mult * volume;
   200→  const commission = Math.round((volume * basePrice * 0.0004) * 100) / 100;
   201→  const swap = Math.round(Math.sin(seed) * 5 * 100) / 100;
   202→  const profit = Math.round((grossPnl - commission - swap) * 100) / 100;
   203→  const netProfit = Math.round((profit - commission - swap) * 100) / 100;
   204→
   205→  // Determine close reason from P&L direction
   206→  let closeReason: CloseReason;
   207→  if (profit >= Math.abs(grossPnl) * 0.5 && profit > 0) closeReason = "TP";
   208→  else if (profit <= -Math.abs(grossPnl) * 0.3 && profit < 0) closeReason = "SL";
   209→  else if (r4 > 0.92) closeReason = "Liquidation";
   210→  else if (r4 > 0.8) closeReason = "System";
   211→  else closeReason = "Manual";
   212→
   213→  // SL / TP — derive from side and entry
   214→  const slDistance = entry * 0.012;
   215→  const tpDistance = entry * 0.024;
   216→  const stopLoss =
   217→    side === "buy" ? entry - slDistance : entry + slDistance;
   218→  const takeProfit =
   219→    side === "buy" ? entry + tpDistance : entry - tpDistance;
   220→  const rrRatio =
   221→    Math.abs(takeProfit - entry) > 0 && Math.abs(entry - stopLoss) > 0
   222→      ? Math.round(
   223→          (Math.abs(takeProfit - entry) / Math.abs(entry - stopLoss)) * 100,
   224→        ) / 100
   225→      : null;
   226→
   227→  // Open / close timestamps
   228→  const now = Date.now();
   229→  const openedMsAgo = Math.floor((1 + r2 * 90 * 24 * 60)) * 60 * 1000;
   230→  const durationMs = Math.floor((r3 * 48 * 60 + 5) * 60 * 1000);
   231→  const closedMsAgo = Math.max(openedMsAgo - durationMs, 60 * 1000);
   232→  const openTime = new Date(now - openedMsAgo).toISOString();
   233→  const closeTime = new Date(now - closedMsAgo).toISOString();
   234→
   235→  const trader = traders.find((t) => t.id === base.traderId);
   236→  const account = accounts.find((a) => a.id === base.accountId);
   237→
   238→  return {
   239→    id,
   240→    uid: `uid-${seed.toString(16).padStart(8, "0")}`,
   241→    direction: side,
   242→    state: "CLOSED",
   243→    symbol: sym,
   244→    symbolDescription: description,
   245→    positionType: "Market",
   246→    entryType: "In",
   247→    volume,
   248→    openPrice: Math.round(entry * 100000) / 100000,
   249→    closePrice: Math.round(close * 100000) / 100000,
   250→    currentPrice: Math.round(close * 100000) / 100000,
   251→    openTime,
   252→    closeTime,
   253→    durationMs,
   254→    openOrderId: `ord-${(seed % 90000) + 1000}`,
   255→    closeOrderId: `ord-${(seed % 90000) + 1000 + 7}`,
   256→    profit,
   257→    commission,
   258→    swap,
   259→    netProfit,
   260→    stopLoss: Math.round(stopLoss * 100000) / 100000,
   261→    takeProfit: Math.round(takeProfit * 100000) / 100000,
   262→    rrRatio,
   263→    isPartial: r4 > 0.85,
   264→    closeReason,
   265→    traderId: trader?.id ?? "trader-unknown",
   266→    traderName: trader?.name ?? "Unknown trader",
   267→    accountId: account?.id ?? base.accountId,
   268→    accountLogin: account?.login ?? "—",
   269→  };
   270→}

   276→function formatDuration(ms: number): string {
   277→  const totalMinutes = Math.round(ms / 60000);
   278→  const h = Math.floor(totalMinutes / 60);
   279→  const m = totalMinutes % 60;
   280→  if (h >= 24) {
   281→    const d = Math.floor(h / 24);
   282→    const rh = h % 24;
   283→    return `${d}d ${rh}h`;
   284→  }
   285→  if (h === 0) return `${m}m`;
   286→  return `${h}h ${m}m`;
   287→}
   288→
   289→function reasonTone(reason: CloseReason): "success" | "danger" | "warning" | "muted" {
   290→  switch (reason) {
   291→    case "TP":
   292→      return "success";
   293→    case "SL":
   294→    case "Liquidation":
   295→      return "danger";
   296→    case "System":
   297→      return "warning";
   298→    case "Manual":
   299→    default:
   300→      return "muted";
   301→  }
   302→}

   311→  const currency = runtime.tenant?.currency ?? "USD";
   312→
   313→  const id = router.params.id ?? "cpos-unknown";
   314→  const seed = useMemo(
   315→    () => generateClosedPositionDetail(id, tid),
   316→    [id, tid],
   317→  );
   318→
   319→  // Working copy — local state so edits don't mutate mock data.
   320→  const [working, setWorking] = useState<ClosedPositionDetail>(seed);
   321→  const [editing, setEditing] = useState(false);
   322→  const [deleteOpen, setDeleteOpen] = useState(false);
   323→
   324→  // Recompute working when the seed changes (navigating to another id).
   325→  // Use a key-based remount of the inner form below to flush stale edits.
   326→  const update = (patch: Partial<ClosedPositionDetail>) =>
   327→    setWorking((w) => ({ ...w, ...patch }));
   328→
   329→  const onToggleEdit = () => {
   330→    if (editing) {
   331→      // Discard local edits on toggle-off.
   332→      setWorking(seed);
   333→    }
   334→    setEditing((e) => !e);
   335→  };

   337→  const onSave = () => {
   338→    setEditing(false);
   339→    toast({
   340→      title: "Position updated",
   341→      description: `Changes to ${working.id} were saved.`,
   342→    });
   343→  };

   345→  const onSaveAndContinue = () => {
   346→    toast({
   347→      title: "Changes saved",
   348→      description: `Position ${working.id} updated. Continuing edits.`,
   349→    });

   352→  const onDelete = () => {
   353→    setDeleteOpen(false);
   354→    toast({
   355→      title: "Closed position deleted",
   356→      description: `${working.id} was permanently deleted along with its P&L and audit trail.`,

   383→            <BreadcrumbPage className="font-mono text-xs">
   384→              {working.id}
   385→            </BreadcrumbPage>
   386→          </BreadcrumbItem>
   387→        </BreadcrumbList>
   388→      </Breadcrumb>
   389→
   390→      {/* Header row */}
   391→      <div className="flex flex-col gap-3 md:flex-row md:items-start md:justify-between">

   393→          <div className="rounded-lg border bg-muted p-2">
