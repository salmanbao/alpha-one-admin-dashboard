     3→/**
     4→ * Challenge Wizard Page (spec §14, §15, §16, §12, §23)
     5→ *
     6→ * Guided 7-step workflow to create a new challenge template. Each step
     7→ * represents a meaningful mental concept:
     8→ *
     9→ *   1. Select challenge type        — what kind of evaluation is this?
    10→ *   2. Phase 1 config               — risk / target / time limits
    11→ *   3. Phase 2 config (if applicable)— verification phase
    12→ *   4. Trading rules                 — what's allowed?
    13→ *   5. Payout rules                  — profit split / frequency / methods
    14→ *   6. Risk rules                    — daily / overall / trailing
    15→ *   7. Review                        — summary + Create Challenge
    16→ *
    17→ * Smart defaults (spec §15): selecting a challenge type auto-fills the
    18→ * phase config forms from getChallengePhaseConfigs(typeId). Templates over
    19→ * blank forms (spec §16).
    20→ *
    21→ * Step indicator pattern mirrors OnboardingWizard (done/active/inactive
    22→ * circles + connecting progress bars).
    23→ *
    24→ * All form state is local useState. No persistence — on Create, we show
    25→ * a toast and reset.
    26→ */

   161→/* ------------------------------------------------------------------ */
   162→/* Component                                                          */
   163→/* ------------------------------------------------------------------ */
   164→
   165→export function ChallengeWizardPage() {
   166→  const [state, setState] = useState<WizardState>(INITIAL_STATE);
   167→  const [step, setStep] = useState(0);
   168→
   169→  const challengeTypes = getChallengeTypes();
   170→  const selectedType = useMemo(
   171→    () => challengeTypes.find((t) => t.id === state.challengeTypeId) ?? null,
   172→    [challengeTypes, state.challengeTypeId],
   173→  );
   174→  const hasPhase2 = (selectedType?.phases ?? 0) >= 2;
   175→
   176→  /* ---- Smart defaults: select type → fill phases ---- */
   177→  const selectChallengeType = (typeId: string) => {
   178→    const configs = getChallengePhaseConfigs(typeId);
   179→    // Find first non-funded phase as Phase 1, second non-funded as Phase 2
   180→    const evaluationPhases = configs.filter((p) => !p.isFunded);
   181→    const phase1Cfg = evaluationPhases[0] ?? configs[0];
   182→    const phase2Cfg = evaluationPhases[1] ?? configs[1];
   183→    setState((s) => ({
   184→      ...s,
   185→      challengeTypeId: typeId,
   186→      phase1: phase1Cfg ? phaseConfigFromTemplate(phase1Cfg) : { ...EMPTY_PHASE },
   187→      phase2: phase2Cfg ? phaseConfigFromTemplate(phase2Cfg) : { ...EMPTY_PHASE },
   188→    }));
   189→  };
   190→
   191→  /* ---- Validation per step ---- */
   192→  const stepValid = (i: number): boolean => {
   193→    switch (i) {
   194→      case 0:
   195→        return state.challengeTypeId !== null;
   196→      case 1:
   197→        return phaseValid(state.phase1);
   198→      case 2:
   199→        return !hasPhase2 || phaseValid(state.phase2);
   200→      case 3:
   201→      case 4:
   202→        return true;
   203→      case 5:
   204→        return (
   205→          Number(state.riskRules.maxDailyLossPct) > 0 &&
   206→          Number(state.riskRules.maxOverallLossPct) > 0
   207→        );
   208→      case 6:
   209→        return true;
   210→      default:
   211→        return false;
   212→    }
   213→  };
   214→
   215→  const canNext = stepValid(step) && step < STEPS.length - 1;
   216→  const canBack = step > 0;
   217→
   218→  const next = () => setStep((s) => Math.min(STEPS.length - 1, s + 1));
   219→  const back = () => setStep((s) => Math.max(0, s - 1));
   220→
   221→  const createChallenge = () => {
   222→    toast({
   223→      title: "Challenge created",
   224→      description: `${selectedType?.name ?? "Challenge"} template saved and ready to assign to traders.`,
   225→    });
   226→    setState(INITIAL_STATE);
   227→    setStep(0);
   228→  };
   229→
   230→  /* ---- Step content renderer ---- */
   231→  const renderStep = () => {
   232→    switch (step) {
   233→      case 0:
   234→        return (
   235→          <StepType
   236→            types={challengeTypes}
   237→            selectedId={state.challengeTypeId}
   238→            onSelect={selectChallengeType}
   239→          />
   240→        );
   241→      case 1:
   242→        return (
   243→          <PhaseForm
   244→            title="Phase 1 — Evaluation"
   245→            description="Initial evaluation phase. Traders must reach the profit target while staying within risk limits."
   246→            value={state.phase1}
   247→            onChange={(next) => setState((s) => ({ ...s, phase1: next }))}
   248→          />
   249→        );
   250→      case 2:
   251→        if (!hasPhase2) {
   252→          return (
   253→            <NoticeCard
   254→              title="Phase 2 not applicable"
   255→              description={`"${selectedType?.name ?? "This challenge"}" only has one phase. You can skip this step and continue to Trading Rules.`}
   256→            />
   257→          );
   258→        }
   259→        return (
   260→          <PhaseForm
   261→            title="Phase 2 — Verification"
   262→            description="Second evaluation phase. Typically a lower profit target with the same risk limits as Phase 1."
   263→            value={state.phase2}
   264→            onChange={(next) => setState((s) => ({ ...s, phase2: next }))}
   265→          />
   266→        );
   267→      case 3:
   268→        return (
   269→          <TradingRulesForm
   270→            value={state.tradingRules}
   271→            onChange={(next) => setState((s) => ({ ...s, tradingRules: next }))}
   272→          />
   273→        );
   274→      case 4:
   275→        return (
   276→          <PayoutRulesForm
   277→            value={state.payoutRules}
   278→            onChange={(next) => setState((s) => ({ ...s, payoutRules: next }))}
   279→          />
   280→        );
   281→      case 5:
   282→        return (
   283→          <RiskRulesForm
   284→            value={state.riskRules}
   285→            onChange={(next) => setState((s) => ({ ...s, riskRules: next }))}
   286→          />
   287→        );
   288→      case 6:
   289→        return (
   290→          <ReviewStep
   291→            state={state}
   292→            selectedType={selectedType}
   293→            hasPhase2={hasPhase2}
   294→            onCreate={createChallenge}
   295→          />
   296→        );
   297→      default:
   298→        return null;
   299→    }
   300→  };
   301→
   302→  return (
   303→    <Page>
   304→      <PageHeader
   305→        title="Create Challenge"
   306→        description="Guided wizard to configure a new evaluation challenge template."
   307→        icon={Sparkles}
   308→      />
   309→      <PageContent>
   310→        {/* Step indicator — done / active / inactive + connecting bars (mirrors OnboardingWizard) */}
   311→        <div className="rounded-lg border bg-card p-4">
   312→          <div className="flex items-center justify-between">
   313→            {STEPS.map((s, i) => {
   314→              const Icon = s.icon;
   315→              const done = i < step;
   316→              const active = i === step;
   317→              return (
   318→                <div key={s.id} className="flex flex-1 flex-col items-center gap-1">
   319→                  <div className="flex w-full items-center">
   320→                    {i > 0 ? (
   321→                      <div className={cn("h-0.5 flex-1", i <= step ? "bg-primary" : "bg-muted")} />
   322→                    ) : null}
   323→                    <div
   324→                      className={cn(
   325→                        "flex h-8 w-8 shrink-0 items-center justify-center rounded-full border-2 transition-all",
   326→                        done
   327→                          ? "border-primary bg-primary text-primary-foreground"
   328→                          : active
   329→                            ? "border-primary text-primary"
   330→                            : "border-muted text-muted-foreground",
   331→                      )}
   332→                    >
   333→                      {done ? <Check className="h-4 w-4" /> : <Icon className="h-4 w-4" />}
   334→                    </div>
   335→                    {i < STEPS.length - 1 ? (
   336→                      <div className={cn("h-0.5 flex-1", i < step ? "bg-primary" : "bg-muted")} />
   337→                    ) : null}
   338→                  </div>
   339→                  <span
   340→                    className={cn(
   341→                      "text-[10px] font-medium",
   342→                      active ? "text-foreground" : "text-muted-foreground",
   343→                    )}
   344→                  >
   345→                    {s.label}
   346→                  </span>
   347→                </div>
   348→              );
   349→            })}
   350→          </div>
   351→        </div>
   352→
   353→        {/* Step content */}
   354→        <div className="rounded-lg border bg-card p-4 md:p-6">
   355→          <div className="min-h-[320px]">{renderStep()}</div>
   356→
   357→          <Separator className="my-4" />
   358→
   359→          {/* Footer nav — one primary action per step (spec §23) */}
   360→          <div className="flex items-center justify-between">
   361→            <Button variant="ghost" size="sm" disabled={!canBack} onClick={back} className="gap-1">
   362→              <ChevronLeft className="h-4 w-4" /> Back
   363→            </Button>
   364→            <span className="text-xs text-muted-foreground">
   365→              Step {step + 1} of {STEPS.length}
   366→            </span>
   367→            {step < STEPS.length - 1 ? (
   368→              <Button size="sm" disabled={!canNext} onClick={next} className="gap-1">
   369→                Next <ChevronRight className="h-4 w-4" />
   370→              </Button>
   371→            ) : (
   372→              <Button size="sm" onClick={createChallenge} className="gap-1.5">
   373→                <Check className="h-4 w-4" /> Create Challenge
   374→              </Button>
   375→            )}
   376→          </div>
   377→        </div>
   378→      </PageContent>
   379→    </Page>
   380→  );
   381→}

   386→
   387→function phaseConfigFromTemplate(cfg: {
   388→  accountSize: number;
   389→  profitTargetPct: number;
   390→  maxDrawdownPct: number;
   391→  dailyDrawdownPct: number;
   392→  minTradingDays: number;
   393→  maxDays: number;
   394→}): PhaseConfig {
   395→  return {
   396→    accountSize: String(cfg.accountSize),
   397→    profitTargetPct: String(cfg.profitTargetPct),
   398→    maxDrawdownPct: String(cfg.maxDrawdownPct),
   399→    dailyDrawdownPct: String(cfg.dailyDrawdownPct),
   400→    minTradingDays: String(cfg.minTradingDays),
   401→    maxDays: String(cfg.maxDays),
   402→  };
   403→}
   404→
   405→function phaseValid(p: PhaseConfig): boolean {
   406→  return (
   407→    Number(p.accountSize) > 0 &&
   408→    Number(p.profitTargetPct) >= 0 &&
   409→    Number(p.maxDrawdownPct) > 0 &&
   410→    Number(p.dailyDrawdownPct) > 0 &&
   411→    Number(p.minTradingDays) >= 0 &&
   412→    Number(p.maxDays) >= 0
   413→  );
   414→}
   415→
   416→function FieldRow({
   417→  label,
   418→  hint,
   419→  children,
   420→}: {
   421→  label: string;
   422→  hint?: string;
   423→  children: React.ReactNode;
   424→}) {
   425→  return (
   426→    <div className="flex flex-col gap-1.5">
   427→      <Label className="text-xs font-medium text-muted-foreground">{label}</Label>
   428→      {children}
   429→      {hint ? <p className="text-[10px] text-muted-foreground/80">{hint}</p> : null}
   430→    </div>
   431→  );
   432→}
   433→
   434→const REQUIRED_ASTERISK = <span className="text-rose-500" aria-hidden>*</span>;
   435→
   436→function RequiredLabel({ children }: { children: React.ReactNode }) {
   437→  return (
   438→    <Label className="text-xs font-medium text-muted-foreground">
   439→      {children} {REQUIRED_ASTERISK}
   440→    </Label>
   441→  );
   442→}

   447→
   448→function StepType({
   449→  types,
   450→  selectedId,
   451→  onSelect,
   452→}: {
   453→  types: ChallengeType[];
   454→  selectedId: string | null;
   455→  onSelect: (id: string) => void;
   456→}) {
   457→  return (
   458→    <div className="space-y-3">
   459→      <div>
   460→        <RequiredLabel>Select challenge type</RequiredLabel>
   461→        <p className="text-xs text-muted-foreground">
   462→          Choose the evaluation structure. Smart defaults will auto-fill the next steps; you can edit
   463→          them later.
   464→        </p>
   465→      </div>
   466→      <div className="grid grid-cols-1 gap-3 sm:grid-cols-2 lg:grid-cols-3">
   467→        {types.map((t) => {
   468→          const Icon = TYPE_ICONS[t.icon] ?? Target;
   469→          const selected = t.id === selectedId;
   470→          return (
   471→            <button
   472→              key={t.id}
   473→              type="button"
   474→              onClick={() => onSelect(t.id)}
   475→              className={cn(
   476→                "flex flex-col items-start gap-2 rounded-lg border p-4 text-left transition-all hover:shadow-sm",
   477→                selected
   478→                  ? "border-primary bg-primary/5 ring-1 ring-primary/20"

   480→              )}
   481→              aria-pressed={selected}
   482→            >
   483→              <div className="flex items-center gap-2">
   484→                <div className="rounded-md bg-muted p-1.5">
   485→                  <Icon className="h-4 w-4 text-foreground" />
   486→                </div>
   487→                <span className="text-sm font-medium text-foreground">{t.name}</span>
   488→              </div>
   489→              <p className="text-xs text-muted-foreground">{t.description}</p>
   490→              <div className="flex flex-wrap items-center gap-1.5 pt-1">
   491→                <Badge variant="outline" className="text-[9px]">{t.phases} phase{t.phases !== 1 ? "s" : ""}</Badge>
   492→                {t.hasFreeTrial ? <Badge variant="secondary" className="text-[9px]">Free trial</Badge> : null}
   493→                {t.isCompetition ? <Badge variant="secondary" className="text-[9px]">Competition</Badge> : null}
   494→                {!t.active ? <Badge variant="outline" className="text-[9px] text-muted-foreground">Inactive</Badge> : null}
   495→              </div>
   496→              {selected ? (
   497→                <div className="mt-1 inline-flex items-center gap-1 text-[10px] font-medium text-primary">
   498→                  <Check className="h-3 w-3" /> Selected — defaults applied
   499→                </div>
   500→              ) : null}
   501→            </button>
   502→          );
   503→        })}
   504→      </div>
   505→    </div>
   506→  );
   507→}

   513→function PhaseForm({
   514→  title,
   515→  description,
   516→  value,
   517→  onChange,
   518→}: {
   519→  title: string;
   520→  description: string;
   521→  value: PhaseConfig;

   562→            min={0}
   563→            step={0.1}
   564→          />
   565→        </div>
   566→        <div className="flex flex-col gap-1.5">
   567→          <RequiredLabel>Daily drawdown (%)</RequiredLabel>
   568→          <Input
   569→            type="number"
   570→            value={value.dailyDrawdownPct}
   571→            onChange={(e) => update("dailyDrawdownPct", e.target.value)}
   572→            placeholder="5"
   573→            min={0}
   574→            step={0.1}

   576→        </div>

   579→            type="number"
   580→            value={value.minTradingDays}
   581→            onChange={(e) => update("minTradingDays", e.target.value)}

   585→        </FieldRow>

   588→            type="number"
   589→            value={value.maxDays}
   590→            onChange={(e) => update("maxDays", e.target.value)}
   591→            placeholder="30"
   592→            min={0}
   593→          />
   594→        </FieldRow>
   595→      </div>
   596→      <p className="text-xs text-muted-foreground">
   597→        Smart defaults are pulled from the selected challenge type's phase templates.
   598→      </p>
   599→    </div>
   600→  );
   601→}

   603→/* ------------------------------------------------------------------ */
   604→/* Step 4 — Trading rules                                             */

   606→
   607→function TradingRulesForm({
   608→  value,
   609→  onChange,

   934→  options,

   936→  onChange,
   937→}: {
   938→  label: string;

   972→function Switch({ checked, onChange }: { checked: boolean; onChange: (v: boolean) => void }) {
   973→  // Lightweight inline switch (avoids importing a radix switch wrapper just for one row)
   974→  return (
   975→    <button
   976→      type="button"
   977→      role="switch"
   978→      aria-checked={checked}
   979→      onClick={() => onChange(!checked)}
   980→      className={cn(
   981→        "relative inline-flex h-5 w-9 shrink-0 items-center rounded-full border border-transparent transition-colors",
   982→        checked ? "bg-primary" : "bg-input",
   983→      )}
   984→    >
   985→      <span
   986→        className={cn(
   987→          "pointer-events-none block h-4 w-4 rounded-full bg-background shadow transition-transform",
   988→          checked ? "translate-x-4" : "translate-x-0.5",
   989→        )}
   990→      />
   991→    </button>
   992→  );
