   463→      </div>
   464→    </div>
   465→  );

   468→/* ------------------------------------------------------------------ */
   469→/* Tab 2 — Modules (KEY feature: toggle tenant entitlements)         */
   470→/* ------------------------------------------------------------------ */
   471→
   472→function ModulesTab({
   473→  tenant,
   474→  onChange,
   475→  onPersist,
   476→}: {
   477→  tenant: TenantContext;
   478→  onChange: (t: TenantContext) => void;
   479→  onPersist: (t: TenantContext) => void;
   480→}) {
   481→  const allModules = moduleRegistry.getAll();
   482→
   483→  const toggle = (moduleId: string) => {
   484→    const enabled = new Set(tenant.enabledModules);
   485→    const willEnable = !enabled.has(moduleId);
   486→    if (willEnable) enabled.add(moduleId);
   487→    else enabled.delete(moduleId);
   488→    const updated: TenantContext = {
   489→      ...tenant,
   490→      enabledModules: Array.from(enabled),
   491→    };
   492→    onChange(updated);
   493→    onPersist(updated);
   494→    const mod = allModules.find((m) => m.manifest.id === moduleId);
   495→    toast({
   496→      title: willEnable ? "Module enabled" : "Module disabled",
   497→      description: `${mod?.manifest.name ?? moduleId} is now ${willEnable ? "accessible" : "hidden"} for ${tenant.name}.`,
   498→    });
   499→  };
   500→
   501→  return (
   502→    <div className="space-y-3">
   503→      <div className="flex items-center justify-between rounded-lg border bg-card p-3">
   504→        <div>
   505→          <p className="text-sm font-medium">
   506→            <LabelWithHelp help="Toggle modules to control which features this tenant can access. Disabled modules disappear from the tenant's sidebar and dashboard instantly.">
   507→              Module entitlements
   508→            </LabelWithHelp>
   509→          </p>
   510→          <p className="text-xs text-muted-foreground">
   511→            {tenant.enabledModules.length} of {allModules.length} modules enabled
   512→          </p>
   513→        </div>
   514→        <Progress
   515→          value={(tenant.enabledModules.length / allModules.length) * 100}
   516→          className="h-2 w-32"
   517→        />
   518→      </div>
   519→      <div className="grid gap-3 md:grid-cols-2 lg:grid-cols-3">
   520→        {allModules.map((m) => {
   521→          const enabled = tenant.enabledModules.includes(m.manifest.id);
   522→          const Icon = m.manifest.icon;
   523→          const accent = m.manifest.accentColor ?? tenant.branding.primaryColor;
   524→          return (
   525→            <Card key={m.manifest.id} className={cn("gap-4 py-4", enabled && "ring-1 ring-primary/20")}>
   526→              <CardContent className="space-y-3">
   527→                <div className="flex items-start justify-between">
   528→                  <div className="flex items-start gap-2.5">
   529→                    <div className="rounded-md p-2" style={{ background: `${accent}1a`, color: accent }}>
   530→                      {Icon ? <Icon className="h-4 w-4" /> : null}
   531→                    </div>
   532→                    <div>
   533→                      <p className="text-sm font-semibold leading-tight">{m.manifest.name}</p>
   534→                      <p className="text-[10px] uppercase tracking-wide text-muted-foreground">
   535→                        {m.manifest.category ?? "module"}
   536→                      </p>
   537→                    </div>
   538→                  </div>
   539→                  <Switch
   540→                    checked={enabled}
   541→                    onCheckedChange={() => toggle(m.manifest.id)}
   542→                    aria-label={`Toggle ${m.manifest.name}`}
   543→                  />
   544→                </div>
   545→                <p className="text-xs text-muted-foreground leading-relaxed">
   546→                  {m.manifest.description ?? "No description available."}
   547→                </p>
