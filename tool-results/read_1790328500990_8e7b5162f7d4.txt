   122→const BASE_SYMBOLS: Array<[string, number, number]> = [

   134→/** Deterministic pseudo-random from a seed. */
   135→function seededRandom(seed: number): number {
   136→  const x = Math.sin(seed * 9999.1) * 10000;
   137→  return x - Math.floor(x);
   138→}
   139→
   140→/** Return the most recent Saturday at noon UTC, offset by weeks ago. */
   141→function saturdayAtNoon(weeksAgo: number): Date {
   142→  const now = new Date();
   143→  // Day-of-week: Sun=0, Sat=6
   144→  const dayOfWeek = now.getUTCDay();
   145→  const daysSinceSaturday = (dayOfWeek + 1) % 7; // days since last Saturday
   146→  const d = new Date(now);
   147→  d.setUTCDate(d.getUTCDate() - daysSinceSaturday - weeksAgo * 7);
   148→  d.setUTCHours(12, 0, 0, 0);
   149→  return d;
   150→}
   151→
   152→/** Build a deterministic list of weekend trades from open positions. */
   153→function buildWeekendTrades(tenantId: string): WeekendTrade[] {
   154→  const positions = getTenantPositions(tenantId);
   155→  const accounts = getTenantAccounts(tenantId);
   156→  const accountById = new Map(accounts.map((a) => [a.id, a]));
   157→
   158→  return positions.map((p: Position, idx) => {
   159→    const [sym, basePrice, mult] = BASE_SYMBOLS[idx % BASE_SYMBOLS.length];
   160→    const seed = idx + 1;
   161→    const r1 = seededRandom(seed);
   162→    const r2 = seededRandom(seed + 0.5);
   163→    const r3 = seededRandom(seed + 1.7);
   164→    const r4 = seededRandom(seed + 3.3);
   165→
   166→    // Open on a Saturday morning, close on Saturday afternoon / Sunday.
   167→    const openWeeksAgo = idx % 8;
   168→    const openTime = saturdayAtNoon(openWeeksAgo);
   169→    openTime.setUTCHours(11 + (idx % 4), (idx * 7) % 60, 0, 0); // Sat 11:00-14:00
   170→    const holdHours = Math.floor(2 + r2 * 22); // 2-24 hours
   171→    const closeTime = new Date(openTime.getTime() + holdHours * 60 * 60 * 1000);
   172→    // If close spills past Sunday midnight, that's fine — still weekend.
   173→
   174→    // Entry/close prices
   175→    const entry = p.entryPrice;
   176→    const movePct = (r1 - 0.5) * 0.03; // ±1.5%
   177→    const close = entry * (1 + movePct);
   178→    const pnlPerUnit = p.side === "buy" ? close - entry : entry - close;
   179→    const volume = p.volume;
   180→    const grossPnl = pnlPerUnit * mult * volume;
   181→    const commission = Math.round((volume * basePrice * 0.0004) * 100) / 100;
   182→    const swap = Math.round(Math.sin(seed) * 5 * 100) / 100;
   183→    const profit = Math.round((grossPnl - commission - swap) * 100) / 100;
   184→
   185→    // SL / TP prices — symmetrical ±0.5% / ±1% around entry.
   186→    const sl = p.side === "buy" ? entry * 0.995 : entry * 1.005;
   187→    const tp = p.side === "buy" ? entry * 1.01 : entry * 0.99;
   188→    const risk = Math.abs(entry - sl);
   189→    const reward = Math.abs(tp - entry);
   190→    const rrRatio = risk > 0 ? Math.round((reward / risk) * 100) / 100 : 0;
   191→
   192→    // Close reason from P&L direction
   193→    let closeReason: CloseReason;
   194→    if (profit >= Math.abs(grossPnl) * 0.5 && profit > 0) closeReason = "TP";
   195→    else if (profit <= -Math.abs(grossPnl) * 0.3 && profit < 0) closeReason = "SL";
   196→    else if (r3 > 0.7) closeReason = "System";
   197→    else closeReason = "Manual";
   198→
   199→    const acct = accountById.get(p.accountId);
   200→    void r4;
   201→
   202→    return {
   203→      id: `wt-${tenantId}-${idx + 1}`,
   204→      uid: `wt-${tenantId}-${idx + 1}`,
   205→      tenantId,
   206→      accountId: p.accountId,
   207→      accountLogin: acct?.login ?? p.accountId,
   208→      traderId: p.traderId,
   209→      traderName: acct?.traderName ?? "Unknown",
   210→      direction: p.side,
   211→      symbol: sym,
   212→      volume,
   213→      openTime: openTime.toISOString(),
   214→      openPrice: Math.round(entry * 100) / 100,
   215→      closeTime: closeTime.toISOString(),
   216→      closePrice: Math.round(close * 100) / 100,
   217→      closeVolume: volume, // closed-out fully
   218→      profit,
   219→      commission,
   220→      swap,
   221→      sl: Math.round(sl * 100) / 100,
   222→      tp: Math.round(tp * 100) / 100,
   223→      rrRatio,
   224→      closeReason,
   225→      state: "CLOSED" as const,
   226→      holdMs: closeTime.getTime() - openTime.getTime(),
   227→      openOrderId: `o-${100000 + idx * 7}`,
   228→      closeOrderId: `c-${200000 + idx * 11}`,
   229→    };
   230→  });
   231→}

   237→function formatDuration(ms: number): string {
   238→  const totalMinutes = Math.round(ms / 60000);
   239→  const h = Math.floor(totalMinutes / 60);
   240→  const m = totalMinutes % 60;
   241→  if (h >= 24) {
   242→    const d = Math.floor(h / 24);
   243→    return `${d}d ${h % 24}h`;
   244→  }
   245→  if (h === 0) return `${m}m`;
   246→  return `${h}h ${m}m`;
   247→}
   248→
   249→function formatWeekday(iso: string): string {
   250→  const d = new Date(iso);
   251→  const weekday = d.toLocaleDateString("en-US", { weekday: "short" });
   252→  return `${weekday} ${d.toLocaleString()}`;
   253→}

   680→        {/* Inline detail panel */}

   689→            onClose={() => setSelectedId(null)}
   690→          />
   691→        ) : null}
   692→      </PageContent>
   693→    </Page>
   694→  );
   695→}
