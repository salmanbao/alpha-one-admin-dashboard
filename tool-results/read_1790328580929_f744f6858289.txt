   255→
   256→function complianceColor(value: number): string {
   257→  if (value >= 95) return COMPLIANCE_COLORS.green;
   258→  if (value >= 80) return COMPLIANCE_COLORS.amber;
   259→  return COMPLIANCE_COLORS.rose;
   260→}
   261→
   262→function complianceTone(value: number): "success" | "warning" | "danger" {
   263→  if (value >= 95) return "success";
   264→  if (value >= 80) return "warning";
   265→  return "danger";
   266→}
   267→
   268→function complianceLabel(value: number): string {
   269→  if (value >= 95) return "On Target";
   270→  if (value >= 80) return "At Risk";
   271→  return "Off Target";
   272→}

   291→/* ------------------------------------------------------------------ */
   292→/* Derive total tickets 30d deterministically from tenant id           */
   293→/* ------------------------------------------------------------------ */
   294→
   295→function deriveTotalTickets30d(tenantId: string): number {
   296→  // Deterministic 80–180 range based on a stable tenant hash.
   297→  // Each tenant sees a consistent number — super-admin (platform) sees 142.
   298→  if (!tenantId || tenantId === "platform") return 142;
   299→  const h = Math.abs(hashStr(tenantId)) % 101;
   300→  return 80 + h;
   301→}
